4impl Simplify<'_> {
5    /// Heavyweight binary-op rewrites. Cheap canonicalization (constant
6    /// folding, null propagation, boolean-constant simplification,
7    /// literal-on-right swap) runs in `fold::expr_binary_op` before this
8    /// is reached, so heavyweight rules see operands in canonical form
9    /// (no `(Value, Value)`, no `(Value, _)` ahead of `(_, Value)`).
10    ///
11    /// App-level rewrites on eq/ne operands (`Reference::Model` →
12    /// primary-key field, `BelongsTo` → foreign-key field) fire in the
13    /// pre-lowering `lower::expr_eq_operand::RewriteEqOperand` pass, not
14    /// here.
15    pub(super) fn simplify_expr_binary_op(
16        &mut self,
17        op: stmt::BinaryOp,
18        lhs: &mut stmt::Expr,
19        rhs: &mut stmt::Expr,
20    ) -> Option<stmt::Expr> {
21        if (op.is_eq() || op.is_ne())
22            && let (Some(lhs_len), Some(rhs_len)) = (lhs.record_len(), rhs.record_len())
23            && lhs_len != rhs_len
24        {
25            return Some(op.is_ne().into());
26        }
27
28        let result = match (&mut *lhs, &mut *rhs) {
29            // Self-comparison, e.g.,
30            //
31            //  - `x = x` → `true`
32            //  - `x != x` → `false`
33            //
34            // Only applied for non-nullable field references.
35            (Expr::Reference(lhs), Expr::Reference(rhs))
36                if lhs == rhs && (op.is_eq() || op.is_ne()) =>
37            {
38                if lhs.is_field() {
39                    let field = self.cx.resolve_expr_reference(lhs).as_field_unwrap();
40                    if !field.nullable() {
41                        return Some(op.is_eq().into());
42                    }
43                }
44                None
45            }
46            // Tuple decomposition,
47            //
48            //  - `(a, b) = (x, y)` → `a = x and b = y`
49            //  - `(a, b) != (x, y)` → `a != x or b != y`
50            (Expr::Record(lhs_rec), Expr::Record(rhs_rec))
51                if (op.is_eq() || op.is_ne()) && lhs_rec.len() == rhs_rec.len() =>
52            {
53                let comparisons: Vec<_> = std::mem::take(&mut lhs_rec.fields)
54                    .into_iter()
55                    .zip(std::mem::take(&mut rhs_rec.fields))
56                    .map(|(l, r)| record_field_comparison(l, op, r))
57                    .collect();
58
59                let mut comparison = if op.is_eq() {
60                    Expr::and_from_vec(comparisons)
61                } else {
62                    Expr::or_from_vec(comparisons)
63                };
64                self.visit_expr_mut(&mut comparison);
65                Some(comparison)
66            }
67            // Tuple decomposition with a Value::Record on one side,
68            //
69            //  - `(a, b) = Value::Record([x, y])` → `a = x and b = y`
70            //
71            // This arises after match elimination produces `Record([col1, col2]) == Value::Record([1, "alice"])`.
72            (Expr::Record(rec), Expr::Value(stmt::Value::Record(val_rec)))
73            | (Expr::Value(stmt::Value::Record(val_rec)), Expr::Record(rec))
74                if (op.is_eq() || op.is_ne()) && rec.len() == val_rec.len() =>
75            {
76                let comparisons: Vec<_> = std::mem::take(&mut rec.fields)
77                    .into_iter()
78                    .zip(std::mem::take(&mut val_rec.fields))
79                    .map(|(expr, val)| record_field_comparison(expr, op, Expr::from(val)))
80                    .collect();
81
82                let mut comparison = if op.is_eq() {
83                    Expr::and_from_vec(comparisons)
84                } else {
85                    Expr::or_from_vec(comparisons)
86                };
87                self.visit_expr_mut(&mut comparison);
88                Some(comparison)
89            }
90            // Match elimination: distribute binary op into match arms as OR
91            //
92            //   Match(subj, [p1 => e1, p2 => e2]) <op> rhs
93            //   → OR(subj == p1 AND e1 <op> rhs, subj == p2 AND e2 <op> rhs)
94            //
95            // Each arm is fully simplified inline. Arms that fold to false/null
96            // are pruned. Comparison ops only — distributing arithmetic this
97            // way would produce a malformed boolean from a non-boolean term.
98            (Expr::Match(m), _) if !op.is_arithmetic() && m.subject.is_stable() => {
99                let match_expr = lhs.take();
100                let other = rhs.take();
101                Some(self.eliminate_match_in_binary_op(op, match_expr, other, true))
102            }
103            (_, Expr::Match(m)) if !op.is_arithmetic() && m.subject.is_stable() => {
104                let other = lhs.take();
105                let match_expr = rhs.take();
106                Some(self.eliminate_match_in_binary_op(op, match_expr, other, false))
107            }
108            // Decode-cast stripping on comparisons with a constant. A stored
109            // column whose type differs from the model type decodes through
110            // `cast(col, <model ty>)` (see `map_table_column_to_model`). When
111            // such a cast surfaces in a comparison — e.g. out of an enum
112            // decode `Match` arm, which only unfolds via match elimination
113            // after lowering has run — move the conversion onto the constant
114            // side so the driver compares the stored form directly. Other
115            // cast shapes are handled during lowering
116            // (`lower_expr_binary_op`); this rule fires only on the
117            // post-lower column shape.
118            (Expr::Cast(cast), Expr::Value(value)) | (Expr::Value(value), Expr::Cast(cast))
119                if (op.is_eq() || op.is_ne()) && cast.from.is_none() && cast.expr.is_column() =>
120            {
121                self.strip_decode_cast_comparison(op, cast, value)
122            }
123            // Decode-cast stripping on column-versus-column comparisons, such
124            // as one embedded relation compared to another after both sides
125            // substitute to key columns. The eligibility check proves that
126            // comparing the stored forms preserves decoded equality.
127            (Expr::Cast(lhs_cast), Expr::Cast(rhs_cast))
128                if self.can_strip_decode_cast_column_comparison(op, lhs_cast, rhs_cast) =>
129            {
130                self.strip_decode_cast_column_comparison(op, lhs_cast, rhs_cast)
131            }
132            // Self-comparison with projections, e.g.,
133            //
134            //  - `address.city = address.city` → `true`
135            //  - `address.city != address.city` → `false`
136            //
137            // By this point, constant projections and record projections have been simplified.
138            // What remains are projections with opaque bases (e.g., field references).
139            // `lhs.base.is_stable()` keeps this sound: a projection through a
140            // non-deterministic base would evaluate the base twice and could
141            // yield different values each time.
142            (Expr::Project(lhs), Expr::Project(rhs))
143                if lhs == rhs && lhs.base.is_stable() && (op.is_eq() || op.is_ne()) =>
144            {
145                // TODO: Check if the projected value is nullable
146                Some(Expr::from(op.is_eq()))
147            }
148            _ => None,
149        };
150
151        if result.is_some() {
152            return result;
153        }
154
155        // Null propagation for derived VALUES columns.
156        //
157        // If either operand is a column reference into a derived VALUES
158        // table where every row has NULL at that column position, the
159        // binary op can never produce a non-null result.
160        if self.is_always_null_derived_column(lhs) || self.is_always_null_derived_column(rhs) {
161            return Some(Expr::null());
162        }
163
164        // Relation-path-comparison and IN-subquery lifting fire in the
165        // pre-lowering `lower::lift_in_subquery::*` pass, not here.
166        None
167    }

169    /// Rewrites `cast(col, T) <eq/ne> const` so the conversion happens on
170    /// the constant instead of the column: converts `const` to `col`'s
171    /// stored type once, here, and emits `col <eq/ne> const'` so the driver
172    /// compares the bare column against the converted constant rather than
173    /// casting the column on every row.
174    ///
175    /// The target type comes from the referenced column, not from
176    /// `Capability::native_type_for(T)`: a `#[column(type = ...)]` override
177    /// can store the value in a different type than the backend's default
178    /// (e.g. a text-stored UUID on SQLite, whose default UUID storage is a
179    /// blob). Bails on anything that does not resolve to a physical column.
180    fn strip_decode_cast_comparison(
181        &mut self,
182        op: stmt::BinaryOp,
183        cast: &mut stmt::ExprCast,
184        value: &mut stmt::Value,
185    ) -> Option<Expr> {
186        let expr_reference = cast.expr.as_expr_reference()?;
187
188        let ResolvedRef::Column(column) = self.cx.resolve_expr_reference(expr_reference) else {
189            return None;
190        };
191
192        let value = column
193            .ty
194            .cast(self.cx.schema(), value.take())
195            .expect("failed to cast value");
196        Some(Expr::binary_op(cast.expr.take(), op, value))
197    }

199    /// Returns whether `cast(col_a, T) <eq/ne> cast(col_b, T)` can compare the
200    /// stored columns directly without changing the result.
201    fn can_strip_decode_cast_column_comparison(
202        &self,
203        op: stmt::BinaryOp,
204        lhs: &stmt::ExprCast,
205        rhs: &stmt::ExprCast,
206    ) -> bool {
207        if !(op.is_eq() || op.is_ne())
208            || lhs.from.is_some()
209            || rhs.from.is_some()
210            || !lhs.expr.is_column()
211            || !rhs.expr.is_column()
212            || lhs.ty != rhs.ty
213        {
214            return false;
215        }
216
217        let Some(lhs_reference) = lhs.expr.as_expr_reference() else {
218            return false;
219        };
220        let Some(rhs_reference) = rhs.expr.as_expr_reference() else {
221            return false;
222        };
223
224        let ResolvedRef::Column(lhs_column) = self.cx.resolve_expr_reference(lhs_reference) else {
225            return false;
226        };
227        let ResolvedRef::Column(rhs_column) = self.cx.resolve_expr_reference(rhs_reference) else {
228            return false;
229        };
230
231        lhs_column.ty == rhs_column.ty && lhs_column.ty.cast_preserves_equality(&lhs.ty)
232    }

234    /// Rewrites `cast(col_a, T) <eq/ne> cast(col_b, T)` to
235    /// `col_a <eq/ne> col_b` after
236    /// [`Self::can_strip_decode_cast_column_comparison`] proves the rewrite is
237    /// valid.
238    fn strip_decode_cast_column_comparison(
239        &mut self,
240        op: stmt::BinaryOp,
241        lhs: &mut stmt::ExprCast,
242        rhs: &mut stmt::ExprCast,
243    ) -> Option<Expr> {
244        Some(Expr::binary_op(lhs.expr.take(), op, rhs.expr.take()))
245    }

247    /// Returns `true` if `expr` is a column reference that resolves to a
248    /// derived VALUES table where every row has NULL at the referenced column.
249    fn is_always_null_derived_column(&self, expr: &Expr) -> bool {
250        let Expr::Reference(expr_ref) = expr else {
251            return false;
252        };
253
254        match self.cx.resolve_expr_reference(expr_ref) {
255            ResolvedRef::Derived(derived_ref) => derived_ref.is_column_always_null(),
256            _ => false,
257        }
258    }

260    /// Distributes a binary op over match arms, producing an OR of guarded
261    /// comparisons. Each arm becomes `(subject == pattern) AND (arm_expr <op> other)`.
262    /// Dead branches (false/null) are pruned after inline simplification.
263    fn eliminate_match_in_binary_op(
264        &mut self,
265        op: stmt::BinaryOp,
266        match_expr: Expr,
267        other: Expr,
268        match_on_lhs: bool,
269    ) -> Expr {
270        self.eliminate_match(match_expr, |arm| {
271            if match_on_lhs {
272                Expr::binary_op(arm, op, other.clone())
273            } else {
274                Expr::binary_op(other.clone(), op, arm)
275            }
276        })
277    }

279    /// Distributes a predicate over match arms, producing an OR of guarded
280    /// terms. `term` builds the predicate for one arm from that arm's
281    /// expression; each arm becomes `(subject == pattern) AND term(arm_expr)`,
282    /// and the else branch is guarded by the negation of every pattern. Dead
283    /// branches (false/null/error) are pruned after inline simplification.
284    pub(super) fn eliminate_match(
285        &mut self,
286        match_expr: Expr,
287        term: impl Fn(Expr) -> Expr,
288    ) -> Expr {
289        let Expr::Match(match_expr) = match_expr else {
290            unreachable!()
291        };
292
293        let mut operands = Vec::new();
294
295        // Collect arm patterns before consuming the arms (needed for the else guard).
296        let patterns: Vec<_> = match_expr.arms.iter().map(|a| a.pattern.clone()).collect();
297
298        for arm in match_expr.arms {
299            // Unit enum variants return the discriminant subject. The arm's
300            // pattern is the value of that subject within this branch.
301            let arm_expr = if arm.expr == *match_expr.subject {
302                Expr::from(arm.pattern.clone())
303            } else {
304                arm.expr
305            };
306            let guard = Expr::binary_op(
307                (*match_expr.subject).clone(),
308                stmt::BinaryOp::Eq,
309                Expr::from(arm.pattern),
310            );
311
312            let mut term = Expr::and_from_vec(vec![guard, term(arm_expr)]);
313            self.visit_expr_mut(&mut term);
314
315            // Prune dead branches
316            if is_dead_filter_term(&term) {
317                continue;
318            }
319
320            operands.push(term);
321        }
322
323        // Include the else branch with a guard that negates all arm patterns.
324        {
325            let guards: Vec<Expr> = patterns
326                .into_iter()
327                .map(|pattern| {
328                    Expr::not(Expr::binary_op(
329                        (*match_expr.subject).clone(),
330                        stmt::BinaryOp::Eq,
331                        Expr::from(pattern),
332                    ))
333                })
334                .collect();
335
336            let mut else_operands = guards;
337            else_operands.push(term(*match_expr.else_expr));
338            let mut term = Expr::and_from_vec(else_operands);
339            self.visit_expr_mut(&mut term);
340
341            // Prune dead branches
342            if !is_dead_filter_term(&term) {
343                operands.push(term);
344            }
345        }
346
347        Expr::or_from_vec(operands)
348    }
349}

351/// Returns `true` when a distributed-comparison term can never be `true`, so it
352/// adds nothing to the surrounding `OR` and can be dropped.
353///
354/// Three shapes qualify:
355///
356/// 1. The term is `false` or a bare `NULL`.
357///
358/// 2. The term is an `AND` with a `NULL` conjunct, such as `x AND NULL`. This
359///    comes from comparing against a decode `Match`'s `null` else branch — for
360///    example `eq(option_embed, Some(..))` on a `None` row. `x AND NULL` never
361///    holds in three-valued logic, and DynamoDB rejects the bare `NULL`
362///    placeholder it would otherwise serialize to.
363///
364/// 3. The term compares against an `Expr::Error`, the placeholder a total enum
365///    decode puts in its unreachable else branch. For a two-variant enum,
366///    distributing `field == "x"` over the decode produces a term like
367///    `disc != 1 AND disc != 2 AND Error == "x"`. A surrounding variant gate
368///    usually contradicts the `disc != k` guards and folds this away, but
369///    factoring a shared predicate out from under its gates removes that gate
370///    (issue #1061). Since `Error` marks a branch that never runs, the
371///    comparison can never hold. Dropping the term also keeps `Error` out of
372///    the SQL serializer, which cannot render it.
373fn is_dead_filter_term(term: &Expr) -> bool {
374    if term.is_unsatisfiable() {
375        return true;
376    }
377    if contains_error(term) {
378        return true;
379    }
380    matches!(
381        term,
382        Expr::And(and) if and.operands.iter().any(Expr::is_value_null)
383    )
384}

386/// Returns `true` if `expr` contains an `Expr::Error` node anywhere in its
387/// subtree — the unreachable-branch placeholder produced by enum decode.
388fn contains_error(expr: &Expr) -> bool {
389    use toasty_core::stmt::Visit;
390
391    struct FindError(bool);
392
393    impl Visit for FindError {
394        fn visit_expr_error(&mut self, _: &stmt::ExprError) {
395            self.0 = true;
396        }
397    }
398
399    let mut find = FindError(false);
400    find.visit_expr(expr);
401    find.0
402}

404/// Uses null checks for record fields, including empty relation slots.
405fn record_field_comparison(lhs: Expr, op: stmt::BinaryOp, rhs: Expr) -> Expr {
406    let other = if lhs.is_value_null() {
407        rhs
408    } else if rhs.is_value_null() {
409        lhs
410    } else {
411        return Expr::binary_op(lhs, op, rhs);
412    };
413    if op.is_eq() {
414        Expr::is_null(other)
415    } else {
416        Expr::is_not_null(other)
417    }
