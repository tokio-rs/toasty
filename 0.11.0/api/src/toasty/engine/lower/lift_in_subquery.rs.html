1//! App-level rewrites that lift relation references out of `IN` subqueries
2//! into direct foreign-key comparisons.
3//!
4//! [`LiftInSubquery`] runs as a whole-statement pre-pass before the main
5//! lowering walk.  The visitor overrides `visit_expr_mut` to fire two
6//! rewrites pre-children:
7//!
8//! - [`lift_in_subquery`] fires on `Expr::InSubquery` where the LHS is a
9//!   relation field reference (`BelongsTo`/`HasOne`/`HasMany`).  For
10//!   `BelongsTo` it tries to lift the subquery's filter into FK comparisons
11//!   on the parent, falling back to a re-targeted IN subquery on the
12//!   foreign-key column.  For `HasOne`/`HasMany` it always rewrites to a
13//!   foreign-key IN subquery against the related table.
14//!
15//! - [`try_lift_relation_path_predicate`] fires on a comparison
16//!   (`Expr::BinaryOp`), a `LIKE`/`ILIKE` (`Expr::Like`), or a variant
17//!   check (`Expr::IsVariant`) whose subject walks a relation field. For
18//!   example, filtering profiles by their user's name
19//!   (`profile.user.name = 'alice'`) rewrites to:
20//!
21//!   ```text
22//!   Profile.user_id IN (SELECT User.id FROM User WHERE User.name = 'alice')
23//!   ```
24//!
25//!   Simplification combines compatible subqueries after this pass, before
26//!   lowering extracts subqueries into separate statements.
27//!
28//! The relation a path walks may be a field of the model or a `belongs_to`
29//! inside one of its embedded types (`v.human.name = 'alice'`, where
30//! `human` is a field of the `Human` variant of `v`). [`resolve_relation_path`]
31//! resolves either to a [`Relation`]; the rewrites above are the same for
32//! both, differing only in how the relation's foreign-key `IN` subquery
33//! names the key on the host side. A predicate *at* an embedded relation
34//! (`v.human = alice`, `v.human IN list`) compares keys and is not lifted:
35//! `LowerStatement`'s operand hooks substitute the key projection.
36//!
37//! A pre-pass is necessary (rather than folding into
38//! `LowerStatement::visit_expr_mut` per #823's pattern) because not every
39//! expression that contains an `IN` subquery flows through
40//! `LowerStatement::visit_expr_mut`; `ApplyInsertScope::apply_expr` in
41//! particular walks insert-scope constraint expressions through its own
42//! recursion and would panic on an unlifted relation `IN` subquery.
43//!
44//! The free functions [`lift_in_subquery`] and
45//! [`try_lift_relation_path_predicate`] are exposed on `&ExprContext` so
46//! the visitor and the unit tests can both call them without constructing
47//! a `LiftInSubquery`.
48

56/// Pre-lowering pass that lifts relation references out of `IN`-subquery
57/// and projection comparisons into direct foreign-key forms.  Runs as a
58/// whole-statement visitor before the main lowering walk: code paths that
59/// process expressions outside the lowering walk (notably
60/// `ApplyInsertScope::apply_expr`) see the already-lifted form.
61pub(super) struct LiftInSubquery<'a> {
62    cx: ExprContext<'a>,
63    exclude_nulls: bool,
64}
65
66impl<'a> LiftInSubquery<'a> {
67    pub(super) fn new(cx: ExprContext<'a>, exclude_nulls: bool) -> Self {
68        Self { cx, exclude_nulls }
69    }
70
71    pub(super) fn rewrite(&mut self, stmt: &mut stmt::Statement) {
72        self.visit_mut(stmt);
73    }
74
75    fn scope<'scope>(&'scope self, target: impl IntoExprTarget<'scope>) -> LiftInSubquery<'scope> {
76        LiftInSubquery {
77            cx: self.cx.scope(target),
78            exclude_nulls: self.exclude_nulls,
79        }
80    }
81
82    fn exclude_nulls(&self, expr: &mut stmt::Expr) {
83        if !self.exclude_nulls {
84            return;
85        }
86
87        let stmt::Expr::InSubquery(expr) = expr else {
88            return;
89        };
90        let select = expr.query.body.as_select_mut_unwrap();
91        let target = select.source.model_id_unwrap();
92        let returning = select.returning.as_project_unwrap().clone();
93
94        for field in returning
95            .as_record()
96            .map_or(std::slice::from_ref(&returning), |record| &record.fields)
97        {
98            let stmt::Expr::Reference(stmt::ExprReference::Field { index, .. }) = field else {
99                unreachable!();
100            };
101            if self.cx.schema().app.field(target.field(*index)).nullable {
102                select.add_filter(stmt::Expr::is_not_null(field.clone()));
103            }
104        }
105    }
106}

108impl VisitMut for LiftInSubquery<'_> {
109    fn visit_expr_mut(&mut self, expr: &mut stmt::Expr) {
110        // Apply lifts pre-children: the rewrites pattern-match on app-
111        // level `Reference::Field` to a relation, and the children walk
112        // would not introduce relation references that were not there.
113        let lifted = match expr {
114            // Normalization expands `NOT IN` to `Not(InSubquery(..))`, so the
115            // lift below fires on the inner membership and keeps the
116            // negation outside.
117            stmt::Expr::InSubquery(e) if !e.negated => {
118                lift_in_subquery(&self.cx, &e.expr, &e.query)
119            }
120            stmt::Expr::BinaryOp(_) | stmt::Expr::Like(_) | stmt::Expr::IsVariant(_) => {
121                try_lift_relation_path_predicate(&self.cx, expr)
122            }
123            _ => None,
124        };
125
126        if let Some(mut lifted) = lifted {
127            self.exclude_nulls(&mut lifted);
128            *expr = lifted;
129        }
130
131        // Walk children (which may themselves be expressions needing
132        // lifts on subtrees).
133        stmt::visit_mut::visit_expr_mut(self, expr);
134    }
135
136    fn visit_stmt_delete_mut(&mut self, stmt: &mut stmt::Delete) {
137        self.visit_source_mut(&mut stmt.from);
138
139        let mut s = self.scope(&stmt.from);
140
141        s.visit_filter_mut(&mut stmt.filter);
142
143        if let Some(returning) = &mut stmt.returning {
144            s.visit_returning_mut(returning);
145        }
146    }
147
148    fn visit_stmt_insert_mut(&mut self, stmt: &mut stmt::Insert) {
149        self.visit_insert_target_mut(&mut stmt.target);
150
151        let mut s = self.scope(&stmt.target);
152
153        s.visit_stmt_query_mut(&mut stmt.source);
154
155        if let Some(returning) = &mut stmt.returning {
156            s.visit_returning_mut(returning);
157        }
158    }
159
160    fn visit_stmt_select_mut(&mut self, stmt: &mut stmt::Select) {
161        self.visit_source_mut(&mut stmt.source);
162
163        let mut s = self.scope(&stmt.source);
164
165        s.visit_filter_mut(&mut stmt.filter);
166        s.visit_returning_mut(&mut stmt.returning);
167    }
168
169    fn visit_stmt_update_mut(&mut self, stmt: &mut stmt::Update) {
170        self.visit_update_target_mut(&mut stmt.target);
171
172        let mut s = self.scope(&stmt.target);
173
174        s.visit_assignments_mut(&mut stmt.assignments);
175        s.visit_filter_mut(&mut stmt.filter);
176
177        if let Some(expr) = &mut stmt.condition.expr {
178            s.visit_expr_mut(expr);
179        }
180
181        if let Some(returning) = &mut stmt.returning {
182            s.visit_returning_mut(returning);
183        }
184    }
185}

196/// Lift `expr IN (subquery)` into a foreign-key-based comparison when
197/// `expr` is a relation field reference and `subquery` targets the
198/// relation's target model.
199///
200/// Returns `None` when the LHS is not a relation field reference or when
201/// the lift cannot apply.  When the lift succeeds, the returned
202/// expression may itself contain unlowered references and the caller is
203/// expected to re-visit it through the lowering walk.
204pub(super) fn lift_in_subquery(
205    cx: &ExprContext,
206    expr: &stmt::Expr,
207    query: &stmt::Query,
208) -> Option<stmt::Expr> {
209    // The expression is a path expression referencing a relation.
210    match expr {
211        // `Project(Ref(rel), [head, ...tail])` — the path traverses through a
212        // relation field (`rel`) before reaching the relation the subquery
213        // targets, or reaches a relation inside an embedded type. Re-root
214        // the path at the relation's target model and rebuild as a nested
215        // IN-subquery on that model, then recurse to lift the outer hop.
216        stmt::Expr::Project(_) | stmt::Expr::Variant(_) => {
217            lift_projection_in_subquery(cx, expr, query)
218        }
219        stmt::Expr::Reference(expr_reference @ stmt::ExprReference::Field { .. }) => {
220            let field = cx.resolve_expr_reference(expr_reference).as_field_unwrap();
221            lift_relation_in_subquery(cx, &Relation::Field(field), query)
222        }
223        _ => None,
224    }
225}

227/// A resolved relation with the expressions needed for subquery lifting.
228struct RelationPath<'a> {
229    relation: Relation<'a>,
230    target: ModelId,
231    target_expr: Option<Expr>,
232}
233
234/// The relation a path walks. Both lift to a foreign-key `IN` subquery on
235/// the target model; they differ in how the host side names the key.
236enum Relation<'a> {
237    /// A relation field of the model: `BelongsTo`, `Has`, or `Via`.
238    Field(&'a app::Field),
239
240    /// A `belongs_to` inside an embedded type, with the expression
241    /// projecting its key field(s) out of the host row.
242    Embedded {
243        belongs_to: &'a BelongsTo,
244        key_expr: Expr,
245    },
246}
247
248fn resolve_relation_path<'a>(cx: &ExprContext<'a>, expr: &Expr) -> Option<RelationPath<'a>> {
249    let resolved = relation_expr::resolve(cx, expr)?;
250    let relation = match resolved.embedded() {
251        Some(belongs_to) => Relation::Embedded {
252            belongs_to,
253            key_expr: resolved.key_expr()?,
254        },
255        None => Relation::Field(resolved.field),
256    };
257    Some(RelationPath {
258        relation,
259        target: resolved.target,
260        target_expr: resolved.target_expr(),
261    })
262}

264/// Lift `relation IN (subquery)` for a resolved relation.
265///
266/// Direct relations lift through their paired foreign keys. A `via` has no
267/// single pair, so normalize its relation chain into a projected path and
268/// use the projection lift. An embedded relation's foreign key is projected
269/// out of the host row. Returns `None` when the field is not a relation.
270fn lift_relation_in_subquery(
271    cx: &ExprContext,
272    relation: &Relation<'_>,
273    query: &stmt::Query,
274) -> Option<stmt::Expr> {
275    match relation {
276        Relation::Field(field) => match &field.ty {
277            FieldTy::BelongsTo(belongs_to) => lift_belongs_to_in_subquery(cx, belongs_to, query),
278            FieldTy::Has(has) => {
279                lift_has_n_in_subquery(has.target, has.pair(&cx.schema().app), query)
280            }
281            FieldTy::Via(via) => lift_via_in_subquery(cx, via, query),
282            _ => None,
283        },
284        Relation::Embedded {
285            belongs_to,
286            key_expr,
287        } => lift_fk_in_subquery(
288            belongs_to.target,
289            key_expr.clone(),
290            super::key_field_refs(0, belongs_to.foreign_key.fields.iter().map(|fk| fk.target)),
291            query,
292        ),
293    }
294}

296/// Lift an `IN` subquery whose left side names a `via` relation.
297///
298/// For `User.groups.any(Group.name == "Rust")`, where `groups` follows
299/// `User.memberships.group`, this builds:
300///
301/// ```text
302/// User.id IN (
303///     SELECT Membership.user_id FROM Membership
304///     WHERE Membership.group_id IN (
305///         SELECT Group.id FROM Group WHERE Group.name == "Rust"
306///     )
307/// )
308/// ```
309///
310/// The path is expanded into direct relation fields, converted to the same
311/// projected expression produced by an explicit relation chain, and then
312/// handled by the regular projection lift.
313fn lift_via_in_subquery(
314    cx: &ExprContext,
315    via: &app::Via,
316    query: &stmt::Query,
317) -> Option<stmt::Expr> {
318    if via.is_scalar() {
319        return None;
320    }
321
322    let fields = super::relation_path::flatten_via_path(cx.schema(), via)?;
323    let target = fields
324        .last()
325        .and_then(|field| cx.schema().app.field(*field).relation_target_id())?;
326
327    if target != via.target || target != query.body.as_select_unwrap().source.model_id_unwrap() {
328        return None;
329    }
330
331    let (base, projection) = fields.split_first()?;
332    let base = stmt::Expr::ref_self_field(*base);
333    let path = if projection.is_empty() {
334        base
335    } else {
336        let projection = projection
337            .iter()
338            .map(|field| field.index)
339            .collect::<Vec<_>>();
340        stmt::Expr::project(base, projection.as_slice())
341    };
342
343    lift_in_subquery(cx, &path, query)
344}

346/// Lifts an `IN`-subquery whose left side is a path through a relation field.
347///
348/// `.any()` on a relation chain — for example, `Release.project.topics.any(name
349/// == "rust")` — builds an `InSubquery` whose LHS projects through the
350/// chain:
351///
352/// ```text
353/// InSubquery {
354///     expr:  Project(Ref(Release.project), [Project.topics_idx]),
355///     query: SELECT FROM Topic WHERE name == "rust",
356/// }
357/// ```
358///
359/// Two paths handle this:
360///
361/// **Fused.** [`try_fuse_paired_relations`] recognizes the common
362/// `BelongsTo → Has` chain over a shared primary key and emits a single
363/// FK-on-FK `IN` that bypasses the intermediate model:
364///
365/// ```text
366/// Release.project_id IN (SELECT Topic.project_id FROM Topic WHERE name == "rust")
367/// ```
368///
369/// **General.** For chains the fast path doesn't match (multi-hop
370/// projections, non-paired relations), re-root the projection at the
371/// relation's target model and wrap the original subquery in a nested `IN`
372/// against that model, then recurse on the outer hop. For the same input,
373/// the fallback produces:
374///
375/// ```text
376/// Release.project IN (
377///     SELECT FROM Project
378///     WHERE Project.topics IN (SELECT FROM Topic WHERE name == "rust")
379/// )
380/// ```
381///
382/// The recursive call lifts the outer `Release.project` hop via the standard
383/// `BelongsTo` branch; the inner `Project.topics` hop is lifted on the next
384/// visitor pass when [`LiftInSubquery`]'s children walk reaches it.
385///
386/// A path may instead reach a relation inside an embedded type
387/// (`v.human` or `v.human.posts`). The steps through the embed resolve to
388/// that relation, whose foreign key is projected out of the host row; any
389/// steps past it are re-rooted at its target as above.
390fn lift_projection_in_subquery(
391    cx: &ExprContext,
392    path: &stmt::Expr,
393    query: &stmt::Query,
394) -> Option<stmt::Expr> {
395    let RelationPath {
396        relation,
397        target,
398        target_expr,
399    } = resolve_relation_path(cx, path)?;
400
401    let Some(inner_lhs) = target_expr else {
402        // The path ends at the relation (`v.human IN (subquery)`).
403        return lift_relation_in_subquery(cx, &relation, query);
404    };
405
406    if let Some(index) = inner_lhs.as_self_field_index()
407        && let Relation::Field(field) = relation
408        && let Some(direct) = try_fuse_paired_relations(cx, field, target, index, query)
409    {
410        return Some(direct);
411    }
412
413    let new_subquery = stmt::Query::new_select(
414        stmt::Source::from(target),
415        Expr::in_subquery(inner_lhs, query.clone()),
416    );
417
418    lift_relation_in_subquery(cx, &relation, &new_subquery)
419}

421/// Fuses a `BelongsTo → Has` chain into a single FK-on-FK `IN` when both
422/// relations meet at the same primary key.
423///
424/// # Example
425///
426/// Given the schema:
427///
428/// ```text
429/// Todo     { category_id: Category.id, ... }   // BelongsTo  Todo.category
430/// Category { todos: HasMany Todo, ... }        // Has, paired with Todo.category
431/// ```
432///
433/// The path `Todo.category.todos` lifts as follows:
434///
435/// ```text
436/// // Input
437/// InSubquery {
438///     expr:  Project(Ref(Todo.category), [Category.todos_idx]),
439///     query: SELECT FROM Todo WHERE title == "salad",
440/// }
441///
442/// // Output
443/// InSubquery {
444///     expr:  Ref(Todo.category_id),
445///     query: SELECT Todo.category_id FROM Todo WHERE title == "salad",
446/// }
447/// ```
448///
449/// The outer relation's FK source columns become the LHS; the inner
450/// relation's paired-BelongsTo FK source columns become the subquery's
451/// returning list. The user's original filter is preserved verbatim.
452///
453/// # Why the fusion is sound
454///
455/// Composing the two hops without fusion routes through the intermediate
456/// model:
457///
458/// ```text
459/// Todo.category_id IN (
460///     SELECT Category.id FROM Category
461///     WHERE Category.id IN (SELECT Todo.category_id FROM Todo WHERE title == "salad")
462/// )
463/// ```
464///
465/// Both FKs target `Category.id`, so every value the innermost subquery
466/// returns exists in `Category.id` under FK integrity (which the engine
467/// already assumes). The middle filter therefore admits every row the inner
468/// returns, and the middle `SELECT Category.id` simply re-emits them. The
469/// `Category` scan is a no-op and can be dropped.
470///
471/// Returns `None` when the chain doesn't match the pattern — outer isn't a
472/// `BelongsTo`, inner isn't a `Has`, or the FK columns don't line up.
473fn try_fuse_paired_relations(
474    cx: &ExprContext,
475    outer_field: &app::Field,
476    target_model_id: ModelId,
477    head_idx: usize,
478    query: &stmt::Query,
479) -> Option<stmt::Expr> {
480    let outer_belongs_to = match &outer_field.ty {
481        FieldTy::BelongsTo(rel) => rel,
482        _ => return None,
483    };
484
485    let target_model = cx.schema().app.model(target_model_id).as_root_unwrap();
486    let head_field = target_model.fields.get(head_idx)?;
487    let inner_has = match &head_field.ty {
488        FieldTy::Has(has) => has,
489        _ => return None,
490    };
491    let inner_pair = inner_has.pair(&cx.schema().app);
492
493    // Both FKs must reference the same PK columns in the same order.
494    if outer_belongs_to.foreign_key.fields.len() != inner_pair.foreign_key.fields.len() {
495        return None;
496    }
497    for (outer_fk, inner_fk) in outer_belongs_to
498        .foreign_key
499        .fields
500        .iter()
501        .zip(inner_pair.foreign_key.fields.iter())
502    {
503        if outer_fk.target != inner_fk.target {
504            return None;
505        }
506    }
507
508    lift_fk_in_subquery(
509        inner_has.target,
510        super::key_field_refs(
511            0,
512            outer_belongs_to
513                .foreign_key
514                .fields
515                .iter()
516                .map(|fk| fk.source),
517        ),
518        super::key_field_refs(0, inner_pair.foreign_key.fields.iter().map(|fk| fk.source)),
519        query,
520    )
521}

523/// Rewrites a predicate whose subject walks a relation into a foreign-key
524/// subquery.
525///
526/// `Profile::filter(Profile::fields().user().name().eq("alice"))` starts as
527/// the filter `profile.user.name = 'alice'`, where the left side projects
528/// through the `user` relation. This moves the comparison into a subquery on
529/// the target model and defers to [`lift_relation_in_subquery`]:
530///
531/// ```text
532/// Profile.user_id IN (SELECT User.id FROM User WHERE User.name = 'alice')
533/// ```
534///
535/// The same applies to `LIKE`/`ILIKE` (`profile.user.name LIKE 'al%'`) and
536/// to a variant check on an enum reached through the relation
537/// (`link.item.state IS Selected`, the guard the typed layer fixes next to
538/// `link.item.state.selected.value == x`).
539///
540/// Returns `None` when `expr` is not one of those predicates or its subject
541/// does not walk a relation.
542pub(super) fn try_lift_relation_path_predicate(
543    cx: &ExprContext,
544    expr: &stmt::Expr,
545) -> Option<stmt::Expr> {
546    let (path, filter) = resolve_relation_predicate(cx, expr)?;
547    lift_relation_predicate(cx, &path, filter)
548}

550/// Resolve a predicate whose subject walks a relation to that relation and
551/// the predicate re-rooted at the relation's target model — the subquery
552/// filter [`lift_relation_predicate`] wraps.
553///
554/// A comparison's subject may be either operand: `'alice' = profile.user.name`
555/// resolves with the operator commuted.
556fn resolve_relation_predicate<'a>(
557    cx: &ExprContext<'a>,
558    expr: &stmt::Expr,
559) -> Option<(RelationPath<'a>, stmt::Expr)> {
560    match expr {
561        Expr::BinaryOp(e) => {
562            let comparison = |op: stmt::BinaryOp, subject: &Expr, other: &Expr| {
563                resolve_relation_path_predicate(cx, subject, |target_lhs| {
564                    Expr::binary_op(target_lhs, op, other.clone())
565                })
566            };
567            comparison(e.op, &e.lhs, &e.rhs).or_else(|| comparison(e.op.commute()?, &e.rhs, &e.lhs))
568        }
569        Expr::Like(e) => resolve_relation_path_predicate(cx, &e.expr, |target_lhs| {
570            stmt::ExprLike {
571                expr: Box::new(target_lhs),
572                pattern: e.pattern.clone(),
573                escape: e.escape,
574                case_insensitive: e.case_insensitive,
575            }
576            .into()
577        }),
578        Expr::IsVariant(e) => resolve_relation_path_predicate(cx, &e.expr, |target_lhs| {
579            Expr::is_variant(target_lhs, e.variant)
580        }),
581        _ => None,
582    }
583}

585/// Shared core of [`resolve_relation_predicate`].
586///
587/// `subject` is the side of the predicate that walks a relation — the
588/// `profile.user.name` in `profile.user.name = 'alice'`. Its steps up to the
589/// relation name the `user` relation on `Profile` (through an embedded type,
590/// when the relation sits in one); the remaining steps index into `User`.
591/// This re-roots the remainder at the target model (so it reads `User.name`)
592/// and hands it to `make_filter`, the only difference between predicate
593/// forms.
594///
595/// Returns `None` when `subject` does not walk into a relation's target —
596/// including a path that ends *at* the relation, which is a key comparison
597/// rather than a predicate on the target.
598fn resolve_relation_path_predicate<'a>(
599    cx: &ExprContext<'a>,
600    subject: &stmt::Expr,
601    make_filter: impl FnOnce(stmt::Expr) -> stmt::Expr,
602) -> Option<(RelationPath<'a>, stmt::Expr)> {
603    let path = resolve_relation_path(cx, subject)?;
604
605    let filter = make_filter(path.target_expr.clone()?);
606
607    Some((path, filter))
608}

610/// Lift `filter`, a predicate on the target model of `path`'s relation,
611/// into a foreign-key `IN` subquery on that relation.
612fn lift_relation_predicate(
613    cx: &ExprContext,
614    path: &RelationPath<'_>,
615    filter: stmt::Expr,
616) -> Option<stmt::Expr> {
617    let subquery = stmt::Query::new_select(stmt::Source::from(path.target), filter);
618    lift_relation_in_subquery(cx, &path.relation, &subquery)
619}

621/// Build a foreign-key `IN` subquery for one direct relation edge.
622///
623/// `lhs` references key fields on the current model. `returning` references
624/// the matching key fields on `target`, which is also the subquery source.
625fn lift_fk_in_subquery(
626    target: ModelId,
627    lhs: stmt::Expr,
628    returning: stmt::Expr,
629    query: &stmt::Query,
630) -> Option<stmt::Expr> {
631    if target != query.body.as_select_unwrap().source.model_id_unwrap() {
632        return None;
633    }
634
635    let mut subquery = query.clone();
636    subquery.body.as_select_mut_unwrap().returning = stmt::Returning::Project(returning);
637
638    Some(stmt::Expr::in_subquery(lhs, subquery))
639}
640
641/// BelongsTo branch: try to lift the subquery's filter into direct FK
642/// comparisons.  When the filter references only FK-mapped fields, return
643/// the AND of per-FK equalities.  Otherwise, fall back to an IN subquery
644/// on the foreign-key column(s) — a tuple-form IN for composite FKs.
645///
646/// Returns `None` when the subquery does not target the BelongsTo's
647/// target model.
648fn lift_belongs_to_in_subquery(
649    cx: &ExprContext,
650    belongs_to: &BelongsTo,
651    query: &stmt::Query,
652) -> Option<stmt::Expr> {
653    if belongs_to.target != query.body.as_select_unwrap().source.model_id_unwrap() {
654        return None;
655    }
656
657    let select = query.body.as_select_unwrap();
658
659    let mut lift = LiftBelongsTo {
660        cx: cx.scope(&select.source),
661        belongs_to,
662        fk_field_matches: vec![false; belongs_to.foreign_key.fields.len()],
663        operands: vec![],
664        fail: false,
665    };
666
667    lift.visit_filter(&select.filter);
668
669    // Fall back to the IN-subquery form whenever we couldn't account for
670    // every FK column with a direct equality from the filter. This covers
671    // both `fail=true` (a binary op referenced something that isn't on the
672    // FK) and the case where the filter contained no liftable binary ops at
673    // all — e.g. when it's a nested `InSubquery` that the LiftBelongsTo
674    // visitor deliberately skips (see `visit_expr_in_subquery`).
675    let all_fks_matched = lift.fk_field_matches.iter().all(|m| *m);
676
677    if lift.fail || !all_fks_matched {
678        lift_fk_in_subquery(
679            belongs_to.target,
680            super::key_field_refs(0, belongs_to.foreign_key.fields.iter().map(|fk| fk.source)),
681            super::key_field_refs(0, belongs_to.foreign_key.fields.iter().map(|fk| fk.target)),
682            query,
683        )
684    } else {
685        Some(if lift.operands.len() == 1 {
686            lift.operands.into_iter().next().unwrap()
687        } else {
688            stmt::ExprAnd {
689                operands: lift.operands,
690            }
691            .into()
692        })
693    }
694}
695
696/// HasOne/HasMany branch: rewrite to a foreign-key IN subquery against
697/// the related table. Single-column FKs produce a scalar IN; composite
698/// FKs produce a tuple-form IN that the SQL serializer renders as
699/// `(a, b) IN (SELECT a, b FROM ...)`.
700fn lift_has_n_in_subquery(
701    target: ModelId,
702    pair: &BelongsTo,
703    query: &stmt::Query,
704) -> Option<stmt::Expr> {
705    lift_fk_in_subquery(
706        target,
707        super::key_field_refs(0, pair.foreign_key.fields.iter().map(|fk| fk.target)),
708        super::key_field_refs(0, pair.foreign_key.fields.iter().map(|fk| fk.source)),
709        query,
710    )
711}
712
713impl Visit for LiftBelongsTo<'_> {
714    fn visit_expr_in_subquery(&mut self, _i: &stmt::ExprInSubquery) {
715        // Stop the walk at a nested IN-subquery boundary. Field references
716        // inside resolve in the nested query's own scope, but the visitor's
717        // `cx` is scoped to the current (outer) subquery's source — recursing
718        // would misresolve inner refs as outer-model fields and produce
719        // bogus FK matches. The main LiftInSubquery walker handles the
720        // nested subquery on its own pass, with the correct scope.
721    }
722
723    fn visit_expr_binary_op(&mut self, i: &stmt::ExprBinaryOp) {
724        match (&*i.lhs, &*i.rhs) {
725            (stmt::Expr::Reference(expr_reference), other)
726            | (other, stmt::Expr::Reference(expr_reference)) => {
727                assert!(i.op.is_eq() || i.op.is_ne());
728
729                if i.op.is_eq() || i.op.is_ne() {
730                    let field = self
731                        .cx
732                        .resolve_expr_reference(expr_reference)
733                        .as_field_unwrap();
734
735                    self.lift_fk_constraint(field.id, i.op, other);
736                } else {
737                    self.fail = true;
738                }
739            }
740            // Constraints we can't lift to a direct FK comparison (e.g. a
741            // projection through an embedded field).  Bail to the IN-subquery
742            // form so the filter is preserved verbatim; without this, the
743            // empty `operands` list silently produced an empty AND (= true)
744            // and the subquery returned every row.
745            _ => {
746                self.fail = true;
747            }
748        }
749    }
750}
751
752impl LiftBelongsTo<'_> {
753    fn lift_fk_constraint(&mut self, field: FieldId, op: stmt::BinaryOp, expr: &stmt::Expr) {
754        for (i, fk_field) in self.belongs_to.foreign_key.fields.iter().enumerate() {
755            if fk_field.target == field {
756                if self.fk_field_matches[i] {
757                    todo!("not handled");
758                }
759
760                self.operands.push(stmt::Expr::binary_op(
761                    stmt::Expr::ref_self_field(fk_field.source),
762                    op,
763                    expr.clone(),
764                ));
765                self.fk_field_matches[i] = true;
766
767                return;
768            }
769        }
770
771        self.fail = true;
772    }
773}
