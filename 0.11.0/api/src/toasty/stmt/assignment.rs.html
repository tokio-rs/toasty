5/// Apply a field mutation to an update statement's [`Assignments`] map.
6///
7/// Every update builder setter accepts `impl Assign<T>`. All types that
8/// implement [`IntoExpr<T>`] also implement `Assign<T>` with set semantics,
9/// so update setters accept the same types as create setters.
10///
11/// For has-many fields, arrays and [`Vec`]s implement
12/// `Assign<List<T>>` with set (replace) semantics. Use [`insert`],
13/// [`remove`], or [`apply`] for incremental mutations.
14///
15/// [`Assignments`]: toasty_core::stmt::Assignments
16pub trait Assign<T> {
17    /// Convert into an [`Assignment<T>`] value.
18    fn into_assignment(self) -> Assignment<T>;
19
20    /// Record one or more assignments into the given [`Assignments`] map at
21    /// the specified projection.
22    fn assign(self, assignments: &mut stmt::Assignments, projection: stmt::Projection)
23    where
24        Self: Sized,
25    {
26        self.into_assignment().kind.apply(assignments, projection);
27    }
28}

30/// A typed assignment produced by the [`insert`], [`remove`], [`set`], and
31/// [`patch`] combinators.
32///
33/// `Assignment<T>` implements `Assign<T>`, so it can be passed directly
34/// to any update builder setter that accepts `impl Assign<T>`.
35pub struct Assignment<T> {
36    kind: AssignmentKind,
37    _p: PhantomData<T>,
38}
39
40enum AssignmentKind {
41    Set(stmt::Expr),
42    Insert(stmt::Expr),
43    Remove(stmt::Expr),
44    Append(stmt::Expr),
45    Pop,
46    RemoveAt(stmt::Expr),
47    Add(stmt::Expr),
48    Subtract(stmt::Expr),
49    Patch {
50        path_projection: stmt::Projection,
51        inner: Box<AssignmentKind>,
52    },
53    Apply(Vec<AssignmentKind>),
54    /// An operation the builder API accepts but Toasty does not support.
55    /// Applying it records the reason on the [`Assignments`] map instead of
56    /// an entry; the statement then fails verification with an
57    /// `unsupported_feature` error before anything is written.
58    ///
59    /// [`Assignments`]: toasty_core::stmt::Assignments
60    Unsupported(String),
61}
62
63impl AssignmentKind {
64    fn apply(self, assignments: &mut stmt::Assignments, projection: stmt::Projection) {
65        match self {
66            AssignmentKind::Set(expr) => assignments.set(projection, expr),
67            AssignmentKind::Insert(expr) => assignments.insert(projection, expr),
68            AssignmentKind::Remove(expr) => assignments.remove(projection, expr),
69            AssignmentKind::Append(expr) => assignments.append(projection, expr),
70            AssignmentKind::Pop => assignments.pop(projection),
71            AssignmentKind::RemoveAt(expr) => assignments.remove_at(projection, expr),
72            AssignmentKind::Add(expr) => assignments.add(projection, expr),
73            AssignmentKind::Subtract(expr) => assignments.subtract(projection, expr),
74            AssignmentKind::Patch {
75                path_projection,
76                inner,
77            } => {
78                let mut projection = projection;
79                for &step in path_projection.as_slice() {
80                    projection.push(step);
81                }
82                inner.apply(assignments, projection);
83            }
84            AssignmentKind::Apply(ops) => {
85                for op in ops {
86                    op.apply(assignments, projection.clone());
87                }
88            }
89            AssignmentKind::Unsupported(reason) => assignments.reject_unsupported(reason),
90        }
91    }
92}

101/// Helper macro: generates `impl Assign<$target> for $source` with set
102/// semantics by delegating to `IntoExpr`. Used alongside every `IntoExpr`
103/// impl to keep the two traits in sync.
104macro_rules! impl_assign_via_expr {
105    // Simple: impl Assign<T> for S
106    ($source:ty => $target:ty) => {
107        impl super::assignment::Assign<$target> for $source {
108            fn into_assignment(self) -> super::assignment::Assignment<$target> {
109                $crate::stmt::set(
110                    super::IntoExpr::<$target>::into_expr(self),
111                )
112            }
113        }
114    };
115    // Generic: impl<generics> Assign<Target> for Source where bounds
116    // Uses { } instead of [ ] to avoid parsing ambiguity with array types.
117    ({ $($gen:tt)* } $source:ty => $target:ty) => {
118        impl<$($gen)*> super::assignment::Assign<$target> for $source {
119            fn into_assignment(self) -> super::assignment::Assignment<$target> {
120                $crate::stmt::set(
121                    super::IntoExpr::<$target>::into_expr(self),
122                )
123            }
124        }
125    };
126}

131/// Insert a value into a collection field.
132///
133/// Takes an expression of `T` (a single item) and produces an assignment for
134/// `List<T>` (the collection). The returned [`Assignment`] can be passed to any
135/// update builder setter that accepts `impl Assign<List<T>>`.
136///
137/// [`Assignments`]: toasty_core::stmt::Assignments
138///
139/// # Examples
140///
141/// ```ignore
142/// user.update()
143///     .todos(stmt::insert(Todo::create().title("Buy groceries")))
144///     .exec(&mut db)
145///     .await?;
146/// ```
147pub fn insert<T>(expr: impl IntoExpr<T>) -> Assignment<List<T>> {
148    Assignment {
149        kind: AssignmentKind::Insert(expr.into_expr().untyped),
150        _p: PhantomData,
151    }
152}

154/// Remove a value from a collection field.
155///
156/// Takes an expression of `T` (the item to remove) and produces an assignment
157/// for `List<T>` (the collection).
158///
159/// What "remove" means depends on the field kind:
160/// - **`Vec<scalar>` field**: every element equal to the value is removed,
161///   atomically against the existing column value. Backends that advertise
162///   [`Capability::vec_remove`](toasty_core::driver::Capability::vec_remove)
163///   support this — currently PostgreSQL only; other backends return an
164///   error.
165/// - **Has-many relation**: the related record is dissociated. With an
166///   optional foreign key the FK is set to `NULL`; with a required foreign
167///   key the related record is deleted.
168///
169/// # Examples
170///
171/// ```ignore
172/// // Vec<String> field
173/// user.update()
174///     .tags(stmt::remove("admin"))
175///     .exec(&mut db)
176///     .await?;
177///
178/// // Has-many relation
179/// user.update()
180///     .todos(stmt::remove(&todo_a))
181///     .exec(&mut db)
182///     .await?;
183/// ```
184pub fn remove<T>(expr: impl IntoExpr<T>) -> Assignment<List<T>> {
185    Assignment {
186        kind: AssignmentKind::Remove(expr.into_expr().untyped),
187        _p: PhantomData,
188    }
189}

191/// Replace a field's value entirely.
192///
193/// For collection fields, `set` replaces the entire collection: all current
194/// members are disassociated (following the same optional/required foreign key
195/// rules as [`remove`]), then the new set is associated.
196///
197/// Pass an empty slice to clear the collection:
198///
199/// ```ignore
200/// user.update()
201///     .todos(stmt::set::<List<Todo>>([]))
202///     .exec(&mut db)
203///     .await?;
204/// ```
205///
206/// For scalar fields, `set` is equivalent to passing a plain value (the
207/// setter already defaults to set semantics).
208///
209/// # Examples
210///
211/// ```ignore
212/// // Replace all todos
213/// user.update()
214///     .todos(stmt::set([
215///         Todo::create().title("Only todo"),
216///     ]))
217///     .exec(&mut db)
218///     .await?;
219/// ```
220pub fn set<T>(expr: impl IntoExpr<T>) -> Assignment<T> {
221    Assignment {
222        kind: AssignmentKind::Set(expr.into_expr().untyped),
223        _p: PhantomData,
224    }
225}

227/// Append one element to an ordered collection field (e.g. `Vec<scalar>`).
228///
229/// Takes an expression of `T` (the element to append) and produces an
230/// assignment for `List<T>` (the collection). The append is atomic
231/// against the existing column value on every supported backend.
232///
233/// After `.exec()`, the instance's field reflects the post-update value
234/// (old contents followed by the appended element).
235///
236/// # Examples
237///
238/// ```ignore
239/// user.update()
240///     .tags(stmt::push("admin"))
241///     .exec(&mut db)
242///     .await?;
243/// ```
244pub fn push<T>(expr: impl IntoExpr<T>) -> Assignment<List<T>> {
245    let element = expr.into_expr().untyped;
246    Assignment {
247        kind: AssignmentKind::Append(stmt::Expr::list([element])),
248        _p: PhantomData,
249    }
250}

254/// Takes a list-shaped expression (anything that converts to `List<T>`

271    Assignment {
272        kind: AssignmentKind::Append(items.into_expr().untyped),
273        _p: PhantomData,
274    }

353/// Partially update a sub-field of an embedded struct.
354///
355/// Takes a [`Path<T, U>`] (identifying which sub-field to update) and a
356/// value (`impl Assign<U>` — either a plain value or a nested
357/// [`Assignment<U>`] for deeper patching). Returns an [`Assignment<T>`]
358/// that can be passed to the parent field's setter.
359///
360/// The path may step through embedded structs, and it may end at an
361/// embedded enum field, which the patch then replaces as a whole. It may
362/// not enter an enum variant: setting or incrementing a field inside a
363/// variant is not supported on any backend. `patch` itself cannot fail, so
364/// the update statement carrying such a patch is rejected as a whole with an
365/// `unsupported_feature` error when it is executed, before any write. Change
366/// a variant's field by assigning a whole enum value instead.
367///
368/// # Examples
369///
370/// ```ignore
371/// // Update a single sub-field
372/// user.update()
373///     .address(stmt::patch(Address::fields().city(), "Portland"))
374///     .exec(&mut db)
375///     .await?;
376///
377/// // Nested patching
378/// company.update()
379///     .headquarters(stmt::patch(
380///         Office::fields().address().into(),
381///         stmt::patch(Address::fields().city(), "Boston"),
382///     ))
383///     .exec(&mut db)
384///     .await?;
385///
386/// // Replace an enum field nested inside a struct
387/// task.update()
388///     .meta(stmt::patch(
389///         Meta::fields().status().into(),
390///         Status::Failed { reason: "flaky".to_string() },
391///     ))
392///     .exec(&mut db)
393///     .await?;
394/// ```
395pub fn patch<T, U>(path: Path<T, U>, value: impl Assign<U>) -> Assignment<T> {
396    let inner = value.into_assignment();
397
398    // A variant root is dropped below along with the rest of the path root,
399    // which would leave the variant-local steps to name whichever field the
400    // parent type has at those indices. Reject the patch instead.
401    let kind = match path.untyped.root {
402        stmt::PathRoot::Model(_) => AssignmentKind::Patch {
403            path_projection: path.untyped.projection,
404            inner: Box::new(inner.kind),
405        },
406        stmt::PathRoot::Variant { .. } => AssignmentKind::Unsupported(
407            "patching a field inside an enum variant is not supported; \
408             assign a whole enum value instead"
409                .to_string(),
410        ),
411    };
412
413    Assignment {
414        kind,
415        _p: PhantomData,
416    }
417}

419/// Apply multiple operations to a single field.
420///
421/// Takes an array or [`Vec`] of [`Assignment<T>`] and applies each in order.
422///
423/// # Examples
424///
425/// ```ignore
426/// user.update()
427///     .todos(stmt::apply([
428///         stmt::insert(Todo::create().title("Buy groceries")),
429///         stmt::insert(Todo::create().title("Walk the dog")),
430///         stmt::remove(&old_todo),
431///     ]))
432///     .exec(&mut db)
433///     .await?;
434/// ```
435pub fn apply<T>(ops: impl IntoIterator<Item = Assignment<T>>) -> Assignment<T> {
436    let ops: Vec<AssignmentKind> = ops.into_iter().map(|a| a.kind).collect();
437    Assignment {
438        kind: AssignmentKind::Apply(ops),
439        _p: PhantomData,
440    }
441}

488/// Increment a numeric field by one.
489///
490/// The [`Numeric`] bound encodes the `1` literal in a value variant
491/// matching the field type, so `stmt::increment()` works on every
492/// supported numeric column (`i8`–`i64`, `u8`–`u64`, `f32`, `f64`) and
493/// on user newtypes that implement [`Numeric`].

503///     .exec(&mut db)
504///     .await?;
505/// ```
506pub fn increment<T: Numeric>() -> Assignment<T> {
507    Assignment {
508        kind: AssignmentKind::Add(stmt::Expr::Value(T::one())),
509        _p: PhantomData,
510    }

538    /// `field 1` of model 0, then field 0 of variant 0 of the enum at model 1.

540        Path::<(), ()>::field_at(ModelId(0), 1)
541            .into_variant(VariantId {
542                model: ModelId(1),
543                index: 0,
544            })
545            .chain(Path::<(), i64>::field_at(ModelId(1), 0))
546    }

548    #[test]
549    fn patch_into_variant_records_unsupported() {

551        patch(path_into_variant(), increment::<i64>())
552            .assign(&mut assignments, stmt::Projection::from_index(2));
