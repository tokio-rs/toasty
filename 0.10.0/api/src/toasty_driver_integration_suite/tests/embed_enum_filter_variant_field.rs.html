3/// Variant+field filter combined with a partition key so DynamoDB can execute it.
4#[driver_test]
5pub async fn filter_variant_field_with_partition_key(t: &mut Test) -> Result<()> {
6    #[derive(Debug, PartialEq, toasty::Embed)]
7    enum ContactInfo {
8        #[column(variant = 1)]
9        Email { address: String },
10        #[column(variant = 2)]
11        Phone { number: String },
12    }
13
14    #[derive(Debug, toasty::Model)]
15    #[key(partition = group, local = id)]
16    #[allow(dead_code)]
17    struct User {
18        #[auto]
19        id: uuid::Uuid,
20        group: String,
21        name: String,
22        contact: ContactInfo,
23    }
24
25    let mut db = t.setup_db(models!(User)).await;
26
27    User::create()
28        .group("eng")
29        .name("Alice")
30        .contact(ContactInfo::Email {
31            address: "alice@example.com".to_string(),
32        })
33        .exec(&mut db)
34        .await?;
35
36    User::create()
37        .group("eng")
38        .name("Bob")
39        .contact(ContactInfo::Phone {
40            number: "555-1234".to_string(),
41        })
42        .exec(&mut db)
43        .await?;
44
45    User::create()
46        .group("eng")
47        .name("Carol")
48        .contact(ContactInfo::Email {
49            address: "carol@example.com".to_string(),
50        })
51        .exec(&mut db)
52        .await?;
53
54    // Partition key + variant field filter
55    let results = User::filter(
56        User::fields().group().eq("eng").and(
57            User::fields()
58                .contact()
59                .email()
60                .matches(|e| e.address().eq("alice@example.com")),
61        ),
62    )
63    .exec(&mut db)
64    .await?;
65
66    assert_eq!(results.len(), 1);
67    assert_eq!(results[0].name, "Alice");
68
69    Ok(())
70}

72/// OR of two variant-rooted field predicates — one per variant. Each predicate
73/// on its own works; only the `OR` used to panic in the SQL serializer with
74/// "unexpected enum discriminant" (issue #1061).
75#[driver_test(requires(scan))]
76pub async fn cross_variant_or(t: &mut Test) -> Result<()> {
77    #[derive(Debug, PartialEq, toasty::Embed)]
78    enum Contact {
79        #[column(variant = 1)]
80        Email { address: String },
81        #[column(variant = 2)]
82        Phone { number: String },
83    }
84
85    #[derive(Debug, toasty::Model)]
86    #[allow(dead_code)]
87    struct User {
88        #[key]
89        #[auto]
90        id: uuid::Uuid,
91        contact: Contact,
92    }
93
94    let mut db = t.setup_db(models!(User)).await;
95
96    User::create()
97        .contact(Contact::Email {
98            address: "x".to_string(),
99        })
100        .exec(&mut db)
101        .await?;
102    User::create()
103        .contact(Contact::Phone {
104            number: "x".to_string(),
105        })
106        .exec(&mut db)
107        .await?;
108
109    let rows = User::filter(
110        User::fields()
111            .contact()
112            .email()
113            .address()
114            .eq("x")
115            .or(User::fields().contact().phone().number().eq("x")),
116    )
117    .exec(&mut db)
118    .await?;
119
120    assert_eq!(rows.len(), 2);
121    Ok(())
122}

124/// A variant field whose model type differs from its stored column type
125/// (`uuid::Uuid`, stored as a string) is filterable. The decode cast the
126/// enum's `Match` arm wraps around the column used to reach the SQL
127/// serializer unchanged and panic; simplify now moves the conversion onto
128/// the constant side.
129#[driver_test]
130pub async fn filter_variant_field_with_cast_storage(t: &mut Test) -> Result<()> {
131    #[derive(Debug, PartialEq, toasty::Embed)]
132    enum Owner {
133        #[column(variant = 1)]
134        Human { id: uuid::Uuid },
135        #[column(variant = 2)]
136        Animal { tag: uuid::Uuid },
137    }
138
139    #[derive(Debug, toasty::Model)]
140    struct Object {
141        #[key]
142        #[auto]
143        id: uuid::Uuid,
144        owner: Owner,
145    }
146
147    let mut db = t.setup_db(models!(Object)).await;
148
149    let target = uuid::Uuid::new_v4();
150    let expected = toasty::create!(Object {
151        owner: Owner::Human { id: target }
152    })
153    .exec(&mut db)
154    .await?;
155    // Same UUID in the other variant's column: must not match.
156    toasty::create!(Object {
157        owner: Owner::Animal { tag: target }
158    })
159    .exec(&mut db)
160    .await?;
161
162    let found = Object::filter(
163        Object::fields()
164            .owner()
165            .human()
166            .matches(|h| h.id().eq(target)),
167    )
168    .exec(&mut db)
169    .await?;
170    assert_eq!(found.len(), 1);
171    assert_eq!(found[0].id, expected.id);
172
173    Ok(())
174}

176/// Same as above, but the variant field overrides its storage type away from
177/// the backend default (`#[column(type = text)]` UUID; SQLite's default UUID
178/// storage is a blob). The comparison constant must be converted to the
179/// referenced column's stored type, not the backend default for the model
180/// type — the latter silently matches zero rows.
181#[driver_test(requires(sql))]
182pub async fn filter_variant_field_with_storage_override(t: &mut Test) -> Result<()> {
183    #[derive(Debug, PartialEq, toasty::Embed)]
184    enum Owner {
185        #[column(variant = 1)]
186        Human {
187            #[column(type = text)]
188            id: uuid::Uuid,
189        },
190        #[column(variant = 2)]
191        Animal { tag: uuid::Uuid },
192    }
193
194    #[derive(Debug, toasty::Model)]
195    struct Object {
196        #[key]
197        #[auto]
198        id: uuid::Uuid,
199        owner: Owner,
200    }
201
202    let mut db = t.setup_db(models!(Object)).await;
203
204    let target = uuid::Uuid::new_v4();
205    let expected = toasty::create!(Object {
206        owner: Owner::Human { id: target }
207    })
208    .exec(&mut db)
209    .await?;
210    toasty::create!(Object {
211        owner: Owner::Human {
212            id: uuid::Uuid::new_v4()
213        }
214    })
215    .exec(&mut db)
216    .await?;
217
218    let found = Object::filter(
219        Object::fields()
220            .owner()
221            .human()
222            .matches(|h| h.id().eq(target)),
223    )

228
229    Ok(())
