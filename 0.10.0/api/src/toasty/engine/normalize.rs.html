15/// Normalizes application-level statements in one recursive traversal.
16///
17/// Node-specific helpers may inspect or mutate the node they receive, but they
18/// must not recurse. The [`VisitMut`] implementation owns traversal so adding a
19/// normalization rule does not add another full AST walk.
20struct Normalize<'a> {
21    schema: &'a Schema,
22    capability: &'a Capability,
23    error: Option<Error>,
24}
25
26impl Engine {
27    pub(super) fn normalize_stmt(&self, stmt: &mut stmt::Statement) -> Result<()> {
28        let mut normalize = Normalize {
29            schema: &self.schema,
30            capability: self.capability,
31            error: None,
32        };
33        normalize.visit_stmt_mut(stmt);
34
35        match normalize.error {
36            Some(error) => Err(error),
37            None => Ok(()),
38        }
39    }
40}
41
42impl Normalize<'_> {
43    fn record(&mut self, error: Error) {
44        if self.error.is_none() {
45            self.error = Some(error);
46        }
47    }
48}
49
50impl VisitMut for Normalize<'_> {
51    fn visit_stmt_insert_mut(&mut self, insert: &mut stmt::Insert) {
52        self.normalize_upsert_defaults(insert);
53        if self.error.is_none() {
54            stmt::visit_mut::visit_stmt_insert_mut(self, insert);
55        }
56    }
57
58    fn visit_stmt_query_mut(&mut self, query: &mut stmt::Query) {
59        self.normalize_cursor_order(query);
60        if self.error.is_none() {
61            stmt::visit_mut::visit_stmt_query_mut(self, query);
62        }
63    }
64}
