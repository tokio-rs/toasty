21/// A dedicated database connection retrieved from a pool.
22///
23/// Holding a `Connection` guarantees that all operations are executed on the
24/// same physical connection. This is useful when multiple statements must
25/// share connection-level state such as temporary tables or session variables.
26///
27/// When dropped, the connection is returned to the pool for reuse.
28pub struct Connection {
29    pub(super) inner: deadpool::managed::Object<Manager>,
30    pub(super) shared: Arc<super::Shared>,
31}
32
33impl Connection {
34    /// Access the underlying connection handle.
35    pub(crate) fn handle(&self) -> &ConnectionHandle {
36        &self.inner
37    }
38
39    /// Returns the compiled schema used by this connection.
40    pub fn schema(&self) -> &Arc<Schema> {
41        &self.shared.engine.schema
42    }
43
44    pub(crate) async fn exec_stmt(
45        &self,
46        stmt: stmt::Statement,
47        in_transaction: bool,
48    ) -> crate::Result<ExecResponse> {
49        // Created on the caller's task so the span parents to the caller's
50        // current span; the worker task enters it while executing.
51        let span = crate::instrument::query_span(&self.shared.engine.schema, &stmt);
52
53        let (tx, rx) = oneshot::channel();
54
55        self.handle()
56            .in_tx
57            .send(ConnectionOperation::ExecStatement {
58                stmt: Box::new(stmt),
59                in_transaction,
60                span: span.clone(),
61                tx,
62            })
63            .unwrap();
64
65        rx.instrument(span).await.unwrap()
66    }
67
68    pub(crate) async fn exec_operation(&self, operation: Operation) -> crate::Result<ExecResponse> {
69        let (tx, rx) = oneshot::channel();
70
71        self.handle()
72            .in_tx
73            .send(ConnectionOperation::ExecOperation {
74                operation: Box::new(operation),
75                span: tracing::Span::current(),
76                tx,
77            })
78            .unwrap();
79

81    }

100    /// Begin a transaction on this connection.
101    ///
102    /// Takes `&mut self` so the `Connection` is exclusively borrowed while
103    /// the transaction is open. This prevents statements from running on the
104    /// connection directly — bypassing the transaction — when they should
105    /// have gone through `&mut tx`.
106    pub async fn transaction(&mut self) -> crate::Result<super::Transaction<'_>> {
107        <Self as super::Executor>::transaction(self).await
108    }
109
110    /// Returns a [`TransactionBuilder`](super::TransactionBuilder) that will
111    /// use this connection.
112    ///
113    /// Like [`transaction`](Self::transaction), this takes `&mut self` so the
114    /// `Connection` stays locked for the lifetime of the transaction.
115    pub fn transaction_builder(&mut self) -> super::TransactionBuilder<'_> {
116        super::TransactionBuilder::new(super::tx::TxSource::Connection(self))
117    }
118
119    /// Creates tables and indices defined in the schema on the database.
120    pub async fn push_schema(&self) -> crate::Result<()> {
121        tracing::info!("pushing schema to database");
122        let (tx, rx) = oneshot::channel();
123        self.handle()
124            .in_tx
125            .send(ConnectionOperation::PushSchema {
126                span: tracing::Span::current(),
127                tx,
128            })
129            .unwrap();
130        rx.await.unwrap()
131    }

146    #[cfg(feature = "migration")]
147    pub(crate) async fn apply_migration(
148        &self,
149        id: u64,
150        name: &str,
151        migration: Migration,
152    ) -> crate::Result<()> {
153        let (tx, rx) = oneshot::channel();
154        self.handle()
155            .in_tx
156            .send(ConnectionOperation::ApplyMigration {
157                id,
158                name: name.to_owned(),
159                migration: Box::new(migration),
160                span: tracing::Span::current(),
161                tx,
162            })
163            .unwrap();
164        rx.await.unwrap()
165    }

168#[async_trait]
169impl super::Executor for Connection {
170    async fn transaction(&mut self) -> crate::Result<Transaction<'_>> {
171        Transaction::begin(ConnRef::Borrowed(self)).await
172    }
173
174    async fn exec_untyped(
175        &mut self,
176        stmt: toasty_core::stmt::Statement,
177    ) -> crate::Result<ExecResponse> {
178        self.exec_stmt(stmt, false).await
179    }
180
181    async fn exec_raw_sql(&mut self, raw: RawSql) -> crate::Result<ExecResponse> {
182        Connection::exec_raw_sql(self, raw).await
183    }
184
185    fn capability(&mut self) -> &Capability {
186        self.shared.engine.capability()
187    }
188
189    fn schema(&mut self) -> &Arc<Schema> {
190        Connection::schema(self)
191    }
192}
