7impl Expand<'_> {
8    pub(super) fn expand_create_builder(&self) -> TokenStream {
9        let toasty = &self.toasty;
10        let vis = &self.model.vis;
11        let model_ident = &self.model.ident;
12        let model_span = model_ident.span();
13        let create_struct_ident = &self.model.kind.as_root_unwrap().create_struct_ident;
14        let create_methods = self.expand_create_methods();
15        let default_stmts = self.expand_create_default_stmts();
16        let conversions = self.expand_create_conversions(
17            create_struct_ident,
18            quote!(self.stmt.into()),
19            quote!(todo!()),
20        );
21
22        // Span the struct definition to the model ident so that "method not
23        // found for this struct" errors point at `struct User`, not the derive
24        // attribute.
25        let struct_def = quote_spanned! { model_span=>
26            #[derive(Clone)]
27            #vis struct #create_struct_ident {
28                stmt: #toasty::stmt::Insert<#model_ident>,
29            }
30        };
31
32        quote! {
33            #struct_def
34
35            impl #create_struct_ident {
36                #create_methods
37
38                #vis async fn exec(self, executor: &mut dyn #toasty::Executor) -> #toasty::Result<#model_ident> {
39                    executor.exec(self.stmt.into()).await
40                }
41            }
42
43            impl #toasty::IntoInsert for #create_struct_ident {
44                type Model = #model_ident;
45
46                fn into_insert(self) -> #toasty::stmt::Insert<#model_ident> {
47                    self.stmt
48                }
49            }
50
51            impl #toasty::IntoStatement for #create_struct_ident {
52                type Returning = #model_ident;
53
54                fn into_statement(self) -> #toasty::Statement<#model_ident> {
55                    self.stmt.into()
56                }
57            }
58
59            #conversions
60
61            impl Default for #create_struct_ident {
62                fn default() -> #create_struct_ident {
63                    let mut s = #create_struct_ident {
64                        stmt: #toasty::stmt::Insert::blank_single(),
65                    };
66                    #default_stmts
67                    s
68                }
69            }
70        }
71    }

73    /// Generates expression and assignment conversions for a construction
74    /// builder, accepting both the model and its nullable form.
75    pub(super) fn expand_create_conversions(
76        &self,
77        builder_ident: &syn::Ident,
78        into_expr: TokenStream,
79        by_ref: TokenStream,
80    ) -> TokenStream {
81        let toasty = &self.toasty;
82        let model_ident = &self.model.ident;
83
84        [
85            (quote!(#model_ident), into_expr, by_ref),
86            (
87                quote!(#toasty::Option<#model_ident>),
88                quote!(<Self as #toasty::IntoExpr<#model_ident>>::into_expr(self).cast()),
89                quote!(<Self as #toasty::IntoExpr<#model_ident>>::by_ref(self).cast()),
90            ),
91        ]
92        .into_iter()
93        .map(|(target, into_expr, by_ref)| {
94            quote! {
95                impl #toasty::IntoExpr<#target> for #builder_ident {
96                    fn into_expr(self) -> #toasty::stmt::Expr<#target> {
97                        #into_expr
98                    }
99
100                    fn by_ref(&self) -> #toasty::stmt::Expr<#target> {
101                        #by_ref
102                    }
103                }
104
105                impl #toasty::Assign<#target> for #builder_ident {
106                    fn into_assignment(self) -> #toasty::stmt::Assignment<#target> {
107                        #toasty::stmt::set(
108                            <Self as #toasty::IntoExpr<#target>>::into_expr(self)
109                        )
110                    }
111                }
112            }
113        })
114        .collect()
115    }

117    fn expand_create_default_stmts(&self) -> TokenStream {
118        let toasty = &self.toasty;
119
120        self.model
121            .fields
122            .iter()
123            .enumerate()
124            .filter_map(|(index, field)| {
125                // #[default] takes priority over #[update] on create
126                let expr = field
127                    .attrs
128                    .default_expr
129                    .as_ref()
130                    .or(field.attrs.update_expr.as_ref())?;
131                let FieldTy::Primitive(ty) = &field.ty else {
132                    return None;
133                };
134                let index_tokenized = util::int(index);
135                Some(quote! {
136                    s.stmt.set(
137                        #index_tokenized,
138                        <#ty as #toasty::IntoExpr<<#ty as #toasty::Field>::ExprTarget>>::into_expr(#expr),
139                    );
140                })
141            })
142            .collect()
143    }
144
145    fn expand_create_methods(&self) -> TokenStream {
146        let toasty = &self.toasty;
147        let vis = &self.model.vis;
148        let model_ident = &self.model.ident;
149
150        self.model
151            .fields
152            .iter()
153            .enumerate()
154            .filter_map(|(index, field)| {
155                let target = self.expand_setter_target(&field.ty)?;
156                let into_expr = if field.ty.is_primitive() {
157                    quote!(IntoExpr)
158                } else {
159                    quote!(#toasty::IntoExpr)
160                };
161                let name = &field.name.ident;
162                let index = util::int(index);
163                let method = if matches!(field.ty, FieldTy::HasMany(_)) {
164                    quote!(insert_all)
165                } else {
166                    quote!(set)
167                };
168                let field_use = matches!(field.ty, FieldTy::BelongsTo(_)).then(|| quote! {
169                    // Silences unused field warning when the field is set on creation.
170                    if false {
171                        let m = <#model_ident as #toasty::Load>::load(Default::default()).unwrap();
172                        let _ = &m.#name;
173                    }
174                });
175
176                Some(quote! {
177                    #vis fn #name(mut self, #name: impl #into_expr<#target>) -> Self {
178                        #field_use
179                        self.stmt.#method(#index, #name.into_expr());
180                        self
181                    }
182                })
183            })
184            .collect()
185    }
