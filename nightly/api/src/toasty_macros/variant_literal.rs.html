1//! Recognizes embedded-enum variant literals in `create!` / `update!` field
2//! values and expands them into construction builder chains.
3
4use crate::model::schema::Name;
5
6use proc_macro2::TokenStream;
7use quote::quote_spanned;
8use syn::spanned::Spanned;
9
10/// Rewrites `Enum::Variant { field: value }` into a chain on the enum's
11/// `EmbedCreate` builder. Other expressions pass through `value` whole.
12/// `create!` returns values unchanged; `update!` hoists each written value.
13///
14/// Recognition requires the last two path segments to start uppercase,
15/// without generic arguments. Unqualified variants, qualified-self paths,
16/// functional updates, and unnamed fields pass through unchanged.
17pub(crate) fn expand_value(
18    expr: &syn::Expr,
19    mut value: impl FnMut(&syn::Expr) -> TokenStream,
20) -> TokenStream {
21    // Validate the whole literal before calling `value`: the update hoister
22    // must not register individual fields if the literal passes through whole.
23    let Some(literal) = VariantLiteral::recognize(expr) else {
24        return value(expr);
25    };
26
27    literal.expand(&mut value)
28}
29
30struct VariantLiteral<'a> {
31    expr: &'a syn::ExprStruct,
32    enum_path: syn::Path,
33    builder_method: syn::Ident,
34}
35
36impl<'a> VariantLiteral<'a> {
37    fn recognize(expr: &'a syn::Expr) -> Option<Self> {
38        let syn::Expr::Struct(expr) = expr else {
39            return None;
40        };
41
42        if expr.qself.is_some() {
43            return None;
44        }
45
46        if expr.rest.is_some() || expr.dot2_token.is_some() {
47            return None;
48        }
49
50        if expr
51            .fields
52            .iter()
53            .any(|field| !matches!(field.member, syn::Member::Named(_)))
54        {
55            return None;
56        }
57
58        let (enum_path, variant) = split_variant_path(&expr.path)?;
59        let builder_method = Name::from_ident(&variant).ident;
60
61        Some(Self {
62            expr,
63            enum_path,
64            builder_method,
65        })
66    }
67
68    fn expand(&self, value: &mut impl FnMut(&syn::Expr) -> TokenStream) -> TokenStream {
69        let enum_path = &self.enum_path;
70        let method = &self.builder_method;
71        let setters = self.expr.fields.iter().map(|field| {
72            let name = &field.member;
73            let value = value(&field.expr);
74            quote_spanned! { name.span()=> .#name(#value) }
75        });
76        let builder = quote_spanned! { enum_path.span()=>
77            <#enum_path as toasty::codegen_support::EmbedCreate>::create()
78        };
79
80        quote_spanned! { self.expr.path.span()=>
81            #builder.#method() #( #setters )*
82        }
83    }
84}
85
86fn split_variant_path(path: &syn::Path) -> Option<(syn::Path, syn::Ident)> {
87    let mut segments = path.segments.iter().rev();
88    let variant = segments.next()?;
89    let enum_name = segments.next()?;
90
91    if !is_type_name(enum_name) || !is_type_name(variant) {
92        return None;
93    }
94
95    let enum_path = syn::Path {
96        leading_colon: path.leading_colon,
97        segments: path
98            .segments
99            .iter()
100            .take(path.segments.len() - 1)
101            .cloned()
102            .collect(),
103    };
104
105    Some((enum_path, variant.ident.clone()))
106}
107
108fn is_type_name(segment: &syn::PathSegment) -> bool {
109    matches!(segment.arguments, syn::PathArguments::None)
110        && segment
111            .ident
112            .to_string()
113            .chars()
114            .next()
115            .is_some_and(|c| c.is_uppercase())
116}
