8/// Collects user-supplied value expressions so they can be evaluated
9/// before the target's `.update()` call.
10///
11/// Instance targets borrow `&mut target` for the whole builder chain, so
12/// a value expression that reads the target (`done: !todo.done`) would
13/// hit E0503 if evaluated inside the chain. Hoisting evaluates every
14/// value first, while the target is still unborrowed.
15///
16/// The values are evaluated as a `match` scrutinee tuple rather than
17/// `let` bindings: temporaries created in a scrutinee live for the whole
18/// `match`, so expressions that borrow from temporaries
19/// (`name: s.to_uppercase().as_str()`) keep compiling exactly as they
20/// did when they were evaluated inline in the chain. Tuple evaluation is
21/// left-to-right, preserving field-order evaluation.
22#[derive(Default)]
23struct Hoist {
24    exprs: Vec<TokenStream>,
25    idents: Vec<syn::Ident>,
26}
27
28impl Hoist {
29    /// Register `expr` for pre-evaluation and return the `__value_N`
30    /// ident that will be bound to it. Both carry the expression's span
31    /// so type errors point at the user's code.
32    fn hoist(&mut self, expr: &syn::Expr) -> TokenStream {
33        let span = expr.span();
34        let ident = format_ident!("__value_{}", self.idents.len(), span = span);
35        self.exprs.push(quote_spanned! { span=> #expr });
36        self.idents.push(ident.clone());
37        quote!(#ident)
38    }
39
40    /// Tokens for a field value.
41    ///
42    /// A variant literal (`Enum::Variant { .. }`) becomes a construction
43    /// builder chain; only its written field values are hoisted, so the
44    /// chain itself runs inside the `match` arm, after the target's
45    /// `.update()`. Any other expression is hoisted whole.
46    fn value(&mut self, expr: &syn::Expr) -> TokenStream {
47        expand_value(expr, |value| self.hoist(value))
48    }
49}

51/// Top-level entry point. Expand an `update!` invocation into the
52/// equivalent update-builder method chain.
53///
54/// Field value expressions are evaluated first, as a `match` scrutinee,
55/// before the target expression is borrowed. This lets values read the
56/// target (`update!(todo { done: !todo.done })`) without tripping over
57/// the `&mut` borrow the builder chain holds.
58///
59/// The target expression is evaluated once at `.update()`, not at
60/// some earlier `let` binding. This is what lets instance updates use
61/// a bare identifier: `update!(user { ... })` expands to
62/// `user.update()`, which auto-borrows `&mut user` the same way the
63/// chain form does, leaving the original binding owned. Query
64/// targets pass straight through and are consumed.
65///
66/// The fields struct needed for embedded-patch paths is recovered
67/// from the bound update builder via the macro-only
68/// `__macro_fields_root()` method.
69///
70/// Field-name validation falls out of the chain expansion: each named
71/// field becomes a method call on the update builder, so a typo
72/// surfaces as a "no method" error at the macro call site without the
73/// macro doing anything specific.
74pub(crate) fn expand(item: &UpdateItem) -> TokenStream {
75    let target = &item.target;
76    let target_span = target.span();
77
78    let fields_path = quote!(__fields);
79    let mut hoist = Hoist::default();
80    let field_calls = expand_field_set(&item.fields, &fields_path, &mut hoist);
81
82    let value_exprs = &hoist.exprs;
83    let value_idents = &hoist.idents;
84
85    // Trailing commas keep the tuple shape valid for zero and one
86    // hoisted values.
87    quote_spanned! { target_span=>
88        match ( #( #value_exprs, )* ) {
89            ( #( #value_idents, )* ) => {
90                let __update = #target.update();
91
92                // Fields path used for embedded-patch path construction.
93                // `unused_variables` is silenced because the update may
94                // have only top-level scalar entries that don't need it.
95                #[allow(unused_variables)]
96                let #fields_path = __update.__macro_fields_root();
97
98                __update #(#field_calls)*
99            }
100        }
101    }
102}

104/// Expand a [`FieldSet`] into a list of method calls.
105fn expand_field_set(
106    fields: &FieldSet,
107    parent_path: &TokenStream,
108    hoist: &mut Hoist,
109) -> Vec<TokenStream> {
110    fields
111        .0
112        .iter()
113        .map(|entry| expand_field_entry(entry, parent_path, hoist))
114        .collect()
115}

117/// Expand a single field entry into a method-call token stream.
118///
119/// `parent_path` is the fields-path expression for the parent type
120/// (e.g. `__target_fields` at the top level, or
121/// `parent_path.embedded_field()` for a nested embedded entry).
122fn expand_field_entry(
123    entry: &FieldEntry,
124    parent_path: &TokenStream,
125    hoist: &mut Hoist,
126) -> TokenStream {
127    let name = &entry.name;
128    let span = name.span();
129
130    match &entry.value {
131        FieldValue::Expr(expr) => {
132            let value = hoist.value(expr);
133            quote_spanned! { span=> .#name(#value) }
134        }
135        FieldValue::Method { combinator, args } => {
136            let args = hoist_args(args, hoist);
137            quote_spanned! { span=> .#name(toasty::stmt::#combinator(#(#args),*)) }
138        }
139        FieldValue::Single(sub_fields) => {
140            // Embedded partial update — expand into
141            // `stmt::apply([stmt::patch(<sub_root>.sub(), val), ...])`.
142            let nested_path = quote_spanned! { span=> #parent_path.#name() };
143            let patches = expand_patch_entries(sub_fields, &nested_path, hoist);
144            quote_spanned! { span=>
145                .#name(toasty::stmt::apply([ #( #patches ),* ]))
146            }
147        }
148        FieldValue::List(items) => {
149            // Two emission shapes, depending on whether any list item is
150            // a `{ ... }` create-builder shorthand:
151            //
152            // - With a builder shorthand (only meaningful for has-many),
153            //   wrap each builder in `stmt::insert(..)` and the whole
154            //   list in `stmt::apply([..])` so the inserts compose with
155            //   any plain `stmt::*` siblings:
156            //
157            //     update!(user {
158            //         todos: [{ title: "x" }, stmt::remove(&old)],
159            //     })
160            //     // →
161            //     // .todos(stmt::apply([
162            //     //     stmt::insert(<child create>),
163            //     //     stmt::remove(&old),
164            //     // ]))
165            //
166            // - With every item a plain expression, pass the array
167            //   straight through. For `Vec<scalar>` this hits the
168            //   setter's `Assign<List<T>>` impl as set semantics:
169            //
170            //     update!(article { tags: ["a", "b"] })
171            //     // → .tags(["a", "b"])
172            let any_builder = items
173                .iter()
174                .any(|item| matches!(item, FieldValue::Single(_)));
175            let nested_path = quote_spanned! { span=> #parent_path.#name() };
176
177            if any_builder {
178                let entries: Vec<_> = items
179                    .iter()
180                    .map(|item| expand_has_many_list_item(item, &nested_path, span, hoist))
181                    .collect();
182                quote_spanned! { span=>
183                    .#name(toasty::stmt::apply([ #( #entries ),* ]))
184                }
185            } else {
186                let entries: Vec<_> = items
187                    .iter()
188                    .map(|item| match item {
189                        FieldValue::Expr(e) => hoist.hoist(e),
190                        // Nested `[ [...] ]` is rejected by the macro.
191                        FieldValue::List(_) => quote_spanned! { span=>
192                            compile_error!("nested lists are not supported in update!")
193                        },
194                        // Single/Method don't appear here because the
195                        // any_builder check above routed us to the other
196                        // branch when a Single was present, and Method only
197                        // appears as a top-level entry shape.
198                        FieldValue::Single(_) | FieldValue::Method { .. } => unreachable!(),
199                    })
200                    .collect();
201                quote_spanned! { span=>
202                    .#name([ #( #entries ),* ])
203                }
204            }
205        }
206    }
207}

209/// Hoist each argument of a method-shorthand call.
210fn hoist_args(
211    args: &syn::punctuated::Punctuated<syn::Expr, syn::Token![,]>,
212    hoist: &mut Hoist,
213) -> Vec<TokenStream> {
214    args.iter().map(|arg| hoist.hoist(arg)).collect()
215}

217/// Expand each entry of an embedded brace block into a
218/// `stmt::patch(<rooted>.sub_field(), value)` invocation.
219fn expand_patch_entries(
220    fields: &FieldSet,
221    parent_path: &TokenStream,
222    hoist: &mut Hoist,
223) -> Vec<TokenStream> {
224    fields
225        .0
226        .iter()
227        .map(|entry| expand_patch_entry(entry, parent_path, hoist))
228        .collect()
229}

231/// Expand a single sub-entry of an embedded brace block.
232fn expand_patch_entry(
233    entry: &FieldEntry,
234    parent_path: &TokenStream,
235    hoist: &mut Hoist,
236) -> TokenStream {
237    let name = &entry.name;
238    let span = name.span();
239
240    // The patch path is rooted at the embedded type itself, obtained
241    // via the `into_root()` inherent method on the parent fields path.
242    // The `.into()` accepts both a plain `Path` (a scalar field) and a
243    // fields handle (an embedded field), which converts to its path.
244    let patch_path = quote_spanned! { span=> #parent_path.into_root().#name().into() };
245
246    match &entry.value {
247        FieldValue::Expr(expr) => {
248            // A variant literal replaces the nested field as a whole: its
249            // builder is handed to `patch` like any other value.
250            let value = hoist.value(expr);
251            quote_spanned! { span=>
252                toasty::stmt::patch(#patch_path, #value)
253            }
254        }
255        FieldValue::Method { combinator, args } => {
256            let args = hoist_args(args, hoist);
257            quote_spanned! { span=>
258                toasty::stmt::patch(
259                    #patch_path,
260                    toasty::stmt::#combinator(#(#args),*),
261                )
262            }
263        }
264        FieldValue::Single(sub_fields) => {
265            // Nested embedded patch — recurse.
266            let nested_path = quote_spanned! { span=> #parent_path.#name() };
267            let inner_patches = expand_patch_entries(sub_fields, &nested_path, hoist);
268            quote_spanned! { span=>
269                toasty::stmt::patch(
270                    #patch_path,
271                    toasty::stmt::apply([ #( #inner_patches ),* ]),
272                )
273            }
274        }
275        FieldValue::List(_) => {
276            quote_spanned! { span=>
277                compile_error!("list values are not supported inside embedded patch blocks")
278            }
279        }
280    }
281}

283/// Expand a single item inside a has-many list literal.
284///
285/// A brace block builds a create builder and is wrapped in
286/// `stmt::insert`. A plain expression passes through as a complete
287/// `Assignment<List<T>>` value (e.g. `stmt::remove(&old)`).
288fn expand_has_many_list_item(
289    item: &FieldValue,
290    parent_path: &TokenStream,
291    span: proc_macro2::Span,
292    hoist: &mut Hoist,
293) -> TokenStream {
294    match item {
295        FieldValue::Single(fields) => {
296            // Build a child create builder via the field-list struct's
297            // `.create()` method. A typo in any setter name surfaces as
298            // a "no method named …" error at the macro call site.
299            let setters = expand_create_setters(fields, hoist);
300            quote_spanned! { span=>
301                toasty::stmt::insert(#parent_path.create() #(#setters)*)
302            }
303        }
304        FieldValue::Expr(e) => hoist.hoist(e),
305        FieldValue::List(_) => quote_spanned! { span=>
306            compile_error!("nested lists are not supported in update!")
307        },
308        FieldValue::Method { .. } => quote_spanned! { span=>
309            compile_error!("method shorthand is not supported as a list item")
310        },
311    }
312}

314/// Expand the setters for a create-builder list item. Same shape as
315/// `expand_field_entry`, but limited to entries that have a
316/// counterpart on a create builder. Deeper nesting inside a has-many
317/// list item is rejected for v1.
318fn expand_create_setters(fields: &FieldSet, hoist: &mut Hoist) -> Vec<TokenStream> {
319    fields
320        .0
321        .iter()
322        .map(|entry| {
323            let name = &entry.name;
324            let span = name.span();
325            match &entry.value {
326                FieldValue::Expr(expr) => {
327                    let value = hoist.value(expr);
328                    quote_spanned! { span=> .#name(#value) }
329                }
330                FieldValue::Method { combinator, args } => {
331                    let args = hoist_args(args, hoist);
332                    quote_spanned! { span=> .#name(toasty::stmt::#combinator(#(#args),*)) }
333                }
334                FieldValue::Single(_) | FieldValue::List(_) => quote_spanned! { span=>
335                    .#name(compile_error!(
336                        "deeper nesting inside a has-many list item is not supported in update!"
337                    ))
338                },
339            }
340        })
341        .collect()
