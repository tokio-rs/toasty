23    /// Converts a MySQL value within a row to a Toasty value.
24    pub(crate) fn from_sql(
25        i: usize,
26        row: &MySqlRow,
27        column: &MySqlColumn,
28        ty: &stmt::Type,
29    ) -> Result<Self, sqlx_core::Error> {
30        if row.try_get_raw(i)?.is_null() {
31            return Ok(Self(CoreValue::Null));
32        }
33
34        let value = match ty {
35            stmt::Type::Bool => decode(row, i, CoreValue::Bool)?,
36            stmt::Type::I8 => decode(row, i, CoreValue::I8)?,
37            stmt::Type::I16 => decode(row, i, CoreValue::I16)?,
38            stmt::Type::I32 => decode(row, i, CoreValue::I32)?,
39            stmt::Type::I64 => decode(row, i, CoreValue::I64)?,
40            stmt::Type::U8 => decode(row, i, CoreValue::U8)?,
41            stmt::Type::U16 => decode(row, i, CoreValue::U16)?,
42            stmt::Type::U32 => decode(row, i, CoreValue::U32)?,
43            stmt::Type::U64 => decode(row, i, CoreValue::U64)?,
44            stmt::Type::F32 => decode(row, i, CoreValue::F32)?,
45            stmt::Type::F64 => decode(row, i, CoreValue::F64)?,
46            stmt::Type::String => decode_string(row, i, column.type_info().name())?,
47            stmt::Type::Bytes => decode(row, i, CoreValue::Bytes)?,
48            stmt::Type::Uuid => decode_uuid(row, i)?,
49            #[cfg(feature = "rust_decimal")]
50            stmt::Type::Decimal => map_string(decode_string(row, i, "DECIMAL")?, |value| {
51                CoreValue::Decimal(value.parse().expect("failed to parse Decimal from MySQL"))
52            }),
53            #[cfg(feature = "bigdecimal")]
54            stmt::Type::BigDecimal => map_string(decode_string(row, i, "DECIMAL")?, |value| {
55                CoreValue::BigDecimal(
56                    value
57                        .parse()
58                        .expect("failed to parse BigDecimal from MySQL"),
59                )
60            }),
61            #[cfg(feature = "jiff")]
62            stmt::Type::Timestamp => map_datetime(decode_datetime(row, i)?, |value| {
63                CoreValue::Timestamp(
64                    value
65                        .to_zoned(jiff::tz::TimeZone::UTC)
66                        .expect("MySQL DATETIME is outside jiff's range")
67                        .into(),
68                )
69            }),
70            #[cfg(feature = "jiff")]
71            stmt::Type::Date => decode_date(row, i)?,
72            #[cfg(feature = "jiff")]
73            stmt::Type::Time => decode_time(row, i)?,
74            #[cfg(feature = "jiff")]
75            stmt::Type::DateTime => decode_datetime(row, i)?,
76            stmt::Type::List(elem) => decode_json(row, i, |bytes| {
77                toasty_sql::json::list_from_slice(bytes, elem)
78                    .expect("MySQL returned non-JSON for a JSON column")
79            })?,
80            stmt::Type::Object => decode_json(row, i, |bytes| {
81                toasty_sql::json::from_slice(bytes, ty)
82                    .expect("MySQL returned non-JSON for a JSON document column")
83            })?,
84            stmt::Type::Null => CoreValue::Null,
85            _ => todo!(
86                "implement SQLx MySQL conversion for `{}` and {ty:#?}",
87                column.type_info().name()
88            ),
89        };
90
91        Ok(Self(value))
92    }

94    /// Converts a MySQL value using the metadata returned by SQLx.
95    pub(crate) fn from_sql_infer(
96        i: usize,
97        row: &MySqlRow,
98        column: &MySqlColumn,
99    ) -> Result<Self, sqlx_core::Error> {
100        if row.try_get_raw(i)?.is_null() {
101            return Ok(Self(CoreValue::Null));
102        }
103
104        let name = column.type_info().name();
105        let value = match name {
106            "TINYINT UNSIGNED" | "SMALLINT UNSIGNED" | "MEDIUMINT UNSIGNED" | "INT UNSIGNED"
107            | "BIGINT UNSIGNED" | "YEAR" | "BIT" => decode(row, i, CoreValue::U64)?,
108            "BOOLEAN" | "TINYINT" | "SMALLINT" | "MEDIUMINT" | "INT" | "BIGINT" => {
109                decode(row, i, CoreValue::I64)?
110            }
111            "FLOAT" => decode(row, i, CoreValue::F32)?,
112            "DOUBLE" => decode(row, i, CoreValue::F64)?,
113            "DATE" => infer_date(row, i)?,
114            "TIME" => infer_time(row, i)?,
115            "TIMESTAMP" | "DATETIME" => infer_datetime(row, i)?,
116            "NULL" => CoreValue::Null,
117            "CHAR" | "VARCHAR" | "BINARY" | "VARBINARY" | "TINYTEXT" | "TEXT" | "MEDIUMTEXT"
118            | "LONGTEXT" | "TINYBLOB" | "BLOB" | "MEDIUMBLOB" | "LONGBLOB" | "ENUM" | "SET"
119            | "DECIMAL" | "JSON" | "GEOMETRY" => infer_bytes(row, i)?,
120            _ => todo!("infer Toasty type for SQLx MySQL type `{name}`"),
121        };
122
123        Ok(Self(value))
124    }

126    /// Adds an owned Toasty value to SQLx's MySQL argument buffer.
127    pub(crate) fn add_to(self, args: &mut MySqlArguments) -> Result<(), sqlx_core::Error> {
128        macro_rules! add {
129            ($value:expr) => {
130                Arguments::add(args, $value).map_err(sqlx_core::Error::Encode)
131            };
132        }
133
134        match self.0 {
135            CoreValue::Bool(value) => add!(value),
136            CoreValue::I8(value) => add!(value),
137            CoreValue::I16(value) => add!(value),
138            CoreValue::I32(value) => add!(value),
139            CoreValue::I64(value) => add!(value),
140            CoreValue::U8(value) => add!(value),
141            CoreValue::U16(value) => add!(value),
142            CoreValue::U32(value) => add!(value),
143            CoreValue::U64(value) => add!(value),
144            CoreValue::F32(value) => add!(value),
145            CoreValue::F64(value) => add!(value),
146            CoreValue::Null => add!(Option::<String>::None),
147            CoreValue::String(value) => add!(value),
148            CoreValue::Bytes(value) => add!(value),
149            CoreValue::Uuid(value) => add!(value.hyphenated().to_string()),
150            #[cfg(feature = "rust_decimal")]
151            CoreValue::Decimal(value) => add!(value.to_string()),
152            #[cfg(feature = "bigdecimal")]
153            CoreValue::BigDecimal(value) => add!(value.to_string()),
154            #[cfg(feature = "jiff")]
155            CoreValue::Timestamp(value) => add!(jiff_datetime_to_time(
156                value.to_zoned(jiff::tz::TimeZone::UTC).datetime()
157            )?),
158            #[cfg(feature = "jiff")]
159            CoreValue::Date(value) => add!(jiff_date_to_time(value)?),
160            #[cfg(feature = "jiff")]
161            CoreValue::Time(value) => add!(jiff_time_to_time(value)?),
162            #[cfg(feature = "jiff")]
163            CoreValue::DateTime(value) => add!(jiff_datetime_to_time(value)?),
164            value @ (CoreValue::List(_) | CoreValue::Object(_)) => {
165                let value = toasty_sql::json::to_vec(&value).expect("serialize MySQL JSON column");
166                add!(String::from_utf8(value).expect("serialized JSON is valid UTF-8"))
167            }
168            value => todo!("encode Toasty value for SQLx MySQL: {value:#?}"),
169        }
170    }
171}

173fn decode<'r, T>(
174    row: &'r MySqlRow,
175    i: usize,
176    constructor: impl FnOnce(T) -> CoreValue,
177) -> Result<CoreValue, sqlx_core::Error>
178where
179    T: Decode<'r, MySql>,
180{
181    row.try_get_unchecked(i).map(constructor)
182}
183
184fn decode_string(row: &MySqlRow, i: usize, mysql_ty: &str) -> Result<CoreValue, sqlx_core::Error> {
185    match mysql_ty {
186        "DATE" => row
187            .try_get_unchecked::<time::Date, _>(i)
188            .map(|value| CoreValue::String(value.to_string())),
189        "TIME" => row
190            .try_get_unchecked::<MySqlTime, _>(i)
191            .map(|value| CoreValue::String(value.to_string())),
192        "TIMESTAMP" | "DATETIME" => row
193            .try_get_unchecked::<time::PrimitiveDateTime, _>(i)
194            .map(|value| CoreValue::String(value.to_string())),
195        _ => row.try_get_unchecked::<Vec<u8>, _>(i).map(|bytes| {
196            CoreValue::String(
197                String::from_utf8(bytes).expect("invalid UTF-8 in MySQL string value"),
198            )
199        }),
200    }
201}
202
203fn decode_uuid(row: &MySqlRow, i: usize) -> Result<CoreValue, sqlx_core::Error> {
204    row.try_get_unchecked::<Vec<u8>, _>(i).map(|bytes| {
205        let uuid = uuid::Uuid::from_slice(&bytes).or_else(|_| {
206            let value = std::str::from_utf8(&bytes).expect("invalid UTF-8 in MySQL UUID value");
207            uuid::Uuid::parse_str(value)
208        });
209        CoreValue::Uuid(uuid.expect("failed to parse UUID from MySQL value"))
210    })
211}

213fn decode_json(
214    row: &MySqlRow,
215    i: usize,
216    constructor: impl FnOnce(&[u8]) -> CoreValue,
217) -> Result<CoreValue, sqlx_core::Error> {
218    row.try_get_unchecked::<Vec<u8>, _>(i)
219        .map(|bytes| constructor(&bytes))
220}

222fn infer_bytes(row: &MySqlRow, i: usize) -> Result<CoreValue, sqlx_core::Error> {
223    row.try_get_unchecked::<Vec<u8>, _>(i).map(|bytes| {
224        String::from_utf8(bytes)
225            .map(CoreValue::String)
226            .unwrap_or_else(|error| CoreValue::Bytes(error.into_bytes()))
227    })
228}

231fn decode_date(row: &MySqlRow, i: usize) -> Result<CoreValue, sqlx_core::Error> {
232    row.try_get_unchecked::<time::Date, _>(i).map(|value| {
233        CoreValue::Date(jiff::civil::Date::constant(
234            value
235                .year()
236                .try_into()
237                .expect("MySQL year outside jiff's range"),
238            u8::from(value.month()) as i8,
239            value.day() as i8,
240        ))
241    })
242}

305#[cfg(feature = "jiff")]
306fn jiff_date_to_time(value: jiff::civil::Date) -> Result<time::Date, sqlx_core::Error> {
307    time::Date::from_calendar_date(
308        value.year().into(),
309        time::Month::try_from(value.month() as u8).map_err(encode_error)?,
310        value.day() as u8,
311    )
312    .map_err(encode_error)
313}

315#[cfg(feature = "jiff")]
316fn jiff_time_to_time(value: jiff::civil::Time) -> Result<time::Time, sqlx_core::Error> {
317    time::Time::from_hms_nano(
318        value.hour() as u8,
319        value.minute() as u8,
320        value.second() as u8,
321        value.subsec_nanosecond() as u32,
322    )
323    .map_err(encode_error)
324}
