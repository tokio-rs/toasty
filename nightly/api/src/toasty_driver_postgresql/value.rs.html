13/// Wrapper for reading string values from PostgreSQL enum columns.
14///
15/// The standard `String::FromSql::accepts()` rejects custom enum types.
16/// This wrapper accepts `Kind::Enum` types and reads the value as UTF-8 text.
17struct EnumString(String);
18
19impl<'a> postgres_types::FromSql<'a> for EnumString {
20    fn from_sql(
21        _ty: &Type,
22        raw: &'a [u8],
23    ) -> std::result::Result<Self, Box<dyn std::error::Error + Sync + Send>> {
24        Ok(EnumString(
25            std::str::from_utf8(raw)
26                .map_err(|e| Box::new(e) as Box<dyn std::error::Error + Sync + Send>)?
27                .to_string(),
28        ))
29    }
30
31    fn accepts(ty: &Type) -> bool {
32        matches!(ty.kind(), Kind::Enum(_))
33    }
34}

36/// Captures the raw wire bytes of a column without interpreting them.
37/// Used by the array decode path to hand the column body to
38/// [`array_from_sql`] directly, bypassing tokio-postgres's
39/// `Vec<Option<T>>` element decoder.
40struct RawBytes<'a>(&'a [u8]);
41
42impl<'a> postgres_types::FromSql<'a> for RawBytes<'a> {
43    fn from_sql(
44        _ty: &Type,
45        raw: &'a [u8],
46    ) -> std::result::Result<Self, Box<dyn std::error::Error + Sync + Send>> {
47        Ok(RawBytes(raw))
48    }
49
50    fn accepts(_ty: &Type) -> bool {
51        true
52    }
53}

70    /// Converts a PostgreSQL value within a row to a Toasty value.
71    pub(crate) fn from_sql(
72        index: usize,
73        row: &Row,
74        column: &Column,
75        expected_ty: &stmt::Type,
76    ) -> Self {
77        // Gets the value from the row as Option<T> and return stmt::Value::Null if the Option is
78        // None.
79        macro_rules! get_or_return_null {
80            ($ty:ty) => {{
81                match row.get::<usize, Option<$ty>>(index) {
82                    Some(inner) => inner,
83                    None => return Self(stmt::Value::Null),
84                }
85            }};
86        }
87
88        // NOTE: unfortunately, the inner representation of the PostgreSQL type enum is not
89        // accessible, so we must manually match each type like so.
90        let core_value = if column.type_() == &Type::TEXT || column.type_() == &Type::VARCHAR {
91            text_to_value(get_or_return_null!(String), expected_ty)
92        } else if column.type_() == &Type::BOOL {
93            stmt::Value::Bool(get_or_return_null!(bool))
94        } else if column.type_() == &Type::INT2 {
95            int2_to_value(get_or_return_null!(i16), expected_ty)
96        } else if column.type_() == &Type::INT4 {
97            int4_to_value(get_or_return_null!(i32), expected_ty)
98        } else if column.type_() == &Type::INT8 {
99            int8_to_value(get_or_return_null!(i64), expected_ty)
100        } else if column.type_() == &Type::UUID {
101            let v = get_or_return_null!(uuid::Uuid);
102            match expected_ty {
103                stmt::Type::Uuid => stmt::Value::Uuid(v),
104                stmt::Type::String => stmt::Value::String(v.to_string()),
105                _ => stmt::Value::Uuid(v),
106            }
107        } else if column.type_() == &Type::BYTEA {
108            let v = get_or_return_null!(Vec<u8>);
109            match expected_ty {
110                stmt::Type::Uuid => stmt::Value::Uuid(v.try_into().expect("invalid uuid bytes")),
111                stmt::Type::Bytes => stmt::Value::Bytes(v),
112                _ => todo!(
113                    "unsupported conversion from {:#?} to {expected_ty:?}",
114                    column.type_()
115                ),
116            }
117        } else if column.type_() == &Type::TIMESTAMPTZ {
118            #[cfg(feature = "jiff")]
119            {
120                stmt::Value::Timestamp(get_or_return_null!(jiff::Timestamp))
121            }
122            #[cfg(not(feature = "jiff"))]
123            {
124                panic!("TIMESTAMPTZ requires jiff feature to be enabled")
125            }
126        } else if column.type_() == &Type::TIMESTAMP {
127            #[cfg(feature = "jiff")]
128            {
129                stmt::Value::DateTime(get_or_return_null!(jiff::civil::DateTime))
130            }
131            #[cfg(not(feature = "jiff"))]
132            {
133                panic!("TIMESTAMP requires jiff feature to be enabled")
134            }
135        } else if column.type_() == &Type::DATE {
136            #[cfg(feature = "jiff")]
137            {
138                stmt::Value::Date(get_or_return_null!(jiff::civil::Date))
139            }
140            #[cfg(not(feature = "jiff"))]
141            {
142                panic!("DATE requires jiff feature to be enabled")
143            }
144        } else if column.type_() == &Type::TIME {
145            #[cfg(feature = "jiff")]
146            {
147                stmt::Value::Time(get_or_return_null!(jiff::civil::Time))
148            }
149            #[cfg(not(feature = "jiff"))]
150            {
151                panic!("TIME requires jiff feature to be enabled")
152            }
153        } else if column.type_() == &Type::CIDR {
154            #[cfg(feature = "net")]
155            {
156                let raw = get_or_return_null!(RawBytes<'_>).0;
157                stmt::Value::Cidr(decode_cidr(raw))
158            }
159            #[cfg(not(feature = "net"))]
160            {
161                panic!("CIDR requires net feature to be enabled")
162            }
163        } else if column.type_() == &Type::INET {
164            #[cfg(feature = "net")]
165            {
166                let raw = get_or_return_null!(RawBytes<'_>).0;
167                stmt::Value::Inet(decode_inet(raw))
168            }
169            #[cfg(not(feature = "net"))]
170            {
171                panic!("INET requires net feature to be enabled")
172            }
173        } else if column.type_() == &Type::MACADDR {
174            #[cfg(feature = "net")]
175            {
176                let raw = get_or_return_null!(RawBytes<'_>).0;
177                stmt::Value::MacAddr(decode_macaddr(raw))
178            }
179            #[cfg(not(feature = "net"))]
180            {
181                panic!("MACADDR requires net feature to be enabled")
182            }
183        } else if column.type_() == &Type::MACADDR8 {
184            #[cfg(feature = "net")]
185            {
186                let raw = get_or_return_null!(RawBytes<'_>).0;
187                stmt::Value::MacAddr8(decode_macaddr8(raw))
188            }
189            #[cfg(not(feature = "net"))]
190            {
191                panic!("MACADDR8 requires net feature to be enabled")
192            }
193        } else if column.type_() == &Type::FLOAT4 {
194            float4_to_value(get_or_return_null!(f32), expected_ty)
195        } else if column.type_() == &Type::FLOAT8 {
196            float8_to_value(get_or_return_null!(f64), expected_ty)
197        } else if column.type_() == &Type::NUMERIC {
198            #[cfg(feature = "rust_decimal")]
199            {
200                stmt::Value::Decimal(get_or_return_null!(rust_decimal::Decimal))
201            }
202            #[cfg(not(feature = "rust_decimal"))]
203            {
204                panic!("NUMERIC requires rust_decimal feature to be enabled")
205            }
206        } else if matches!(column.type_().kind(), Kind::Enum(_)) {
207            // Native database enum types (CREATE TYPE ... AS ENUM) are read as strings.
208            // We use EnumString instead of String because String::FromSql::accepts()
209            // rejects custom enum types.
210            match row.get::<usize, Option<EnumString>>(index) {
211                Some(EnumString(v)) => stmt::Value::String(v),
212                None => return Self(stmt::Value::Null),
213            }
214        } else if column.type_() == &Type::JSONB || column.type_() == &Type::JSON {
215            let raw = match row.get::<usize, Option<RawBytes<'_>>>(index) {
216                Some(RawBytes(raw)) => raw,
217                None => return Self(stmt::Value::Null),
218            };
219            // `jsonb` wire format: a 1-byte version header, then UTF-8 JSON
220            // text. `json` is plain UTF-8 text.
221            let json_bytes = if column.type_() == &Type::JSONB {
222                assert_eq!(raw.first(), Some(&1), "unsupported jsonb wire version");
223                &raw[1..]
224            } else {
225                raw
226            };
227
228            if expected_ty == &stmt::Type::String {
229                // `Json<T>` is opaque to the engine. Preserve the serialized
230                // JSON so `Json<T>::load` can deserialize it directly into T.
231                // `String::from_utf8` reuses this single allocation.
232                stmt::Value::String(
233                    String::from_utf8(json_bytes.to_vec())
234                        .expect("PostgreSQL returned non-UTF-8 JSON"),
235                )
236            } else {
237                // `#[document]` columns are queryable by the engine, so decode
238                // them to their structural `Value::Object` representation.
239                toasty_sql::json::from_slice(json_bytes, expected_ty)
240                    .expect("invalid JSON in document column")
241            }
242        } else if let Kind::Array(_) = column.type_().kind() {
243            // Native array column (e.g. `text[]`, `int8[]`) — decoded
244            // directly into `Vec<stmt::Value>` via `array_from_sql`. The
245            // element type is taken from `expected_ty.as_list_unwrap()` so
246            // the per-element conversion mirrors a column of that scalar
247            // type.
248            let elem_ty = match expected_ty {
249                stmt::Type::List(elem) => elem.as_ref(),
250                other => panic!("array column expected stmt::Type::List, got {other:?}"),
251            };
252            let items = read_array_items(index, row, column, elem_ty);
253            match items {
254                Some(items) => stmt::Value::List(items),
255                None => return Self(stmt::Value::Null),
256            }
257        } else {
258            todo!(
259                "implement PostgreSQL to toasty conversion for `{:#?}`",
260                column.type_()
261            );
262        };
263
264        Value(core_value)
265    }

398// ============================================================================
399// Per-primitive conversions
400// ----------------------------------------------------------------------------
401// These functions translate a single decoded PostgreSQL primitive (the value
402// you get from `Row::get` or from an array element) into a `stmt::Value`,
403// respecting Toasty's expected element type. Sharing them between the column
404// path ([`Value::from_sql`]) and the array path ([`read_array_items`]) keeps
405// the two reading paths consistent: a `text[]` element and a `text` column
406// decode through the same logic.
407// ============================================================================
408
409fn text_to_value(v: String, expected_ty: &stmt::Type) -> stmt::Value {
410    match expected_ty {
411        stmt::Type::String => stmt::Value::String(v),
412        stmt::Type::Uuid => stmt::Value::Uuid(
413            v.parse()
414                .unwrap_or_else(|_| panic!("uuid could not be parsed from text")),
415        ),
416        _ => stmt::Value::String(v),
417    }
418}
419
420fn int2_to_value(v: i16, expected_ty: &stmt::Type) -> stmt::Value {
421    match expected_ty {
422        stmt::Type::I8 => stmt::Value::I8(v as i8),
423        stmt::Type::I16 => stmt::Value::I16(v),
424        stmt::Type::U8 => stmt::Value::U8(
425            u8::try_from(v).unwrap_or_else(|_| panic!("u8 value out of range: {v}")),
426        ),
427        stmt::Type::U16 => stmt::Value::U16(v as u16),
428        _ => panic!("unexpected type for INT2: {expected_ty:#?}"),
429    }
430}
431
432fn int4_to_value(v: i32, expected_ty: &stmt::Type) -> stmt::Value {
433    match expected_ty {
434        stmt::Type::I32 => stmt::Value::I32(v),
435        stmt::Type::U16 => stmt::Value::U16(
436            u16::try_from(v).unwrap_or_else(|_| panic!("u16 value out of range: {v}")),
437        ),
438        stmt::Type::U32 => stmt::Value::U32(v as u32),
439        _ => stmt::Value::I32(v),
440    }
441}
442
443fn int8_to_value(v: i64, expected_ty: &stmt::Type) -> stmt::Value {
444    match expected_ty {
445        stmt::Type::I64 => stmt::Value::I64(v),
446        stmt::Type::U32 => stmt::Value::U32(
447            u32::try_from(v).unwrap_or_else(|_| panic!("u32 value out of range: {v}")),
448        ),
449        stmt::Type::U64 => stmt::Value::U64(
450            u64::try_from(v).unwrap_or_else(|_| panic!("u64 value out of range: {v}")),
451        ),
452        _ => stmt::Value::I64(v),
453    }
454}

456fn float4_to_value(v: f32, expected_ty: &stmt::Type) -> stmt::Value {
457    match expected_ty {
458        stmt::Type::F32 => stmt::Value::F32(v),
459        stmt::Type::F64 => stmt::Value::F64(v as f64),

466        stmt::Type::F32 => stmt::Value::F32(v as f32),
467        stmt::Type::F64 => stmt::Value::F64(v),
468        _ => panic!("unexpected type for FLOAT8: {expected_ty:#?}"),

472/// Decode a PostgreSQL array column into a list of Toasty values. Returns
473/// `None` for SQL NULL. The column body is parsed with [`array_from_sql`]
474/// and each element is decoded straight into a `stmt::Value` via the
475/// per-primitive helpers above — the same ones used by [`Value::from_sql`]
476/// for scalar columns. No intermediate `Vec<Option<T>>` is allocated.
477fn read_array_items(
478    index: usize,
479    row: &Row,
480    column: &Column,
481    elem_ty: &stmt::Type,
482) -> Option<Vec<stmt::Value>> {
483    let elem_pg_ty = match column.type_().kind() {
484        Kind::Array(elem) => elem,
485        _ => panic!(
486            "read_array_items called on non-array column: {:?}",
487            column.type_()
488        ),
489    };
490
491    let RawBytes(raw) = row.get::<usize, Option<RawBytes<'_>>>(index)?;
492    let array = array_from_sql(raw).expect("invalid PostgreSQL array wire format");
493
494    let ndims = array
495        .dimensions()
496        .count()
497        .expect("invalid PostgreSQL array dimensions header");
498    if ndims > 1 {
499        panic!(
500            "multi-dimensional PostgreSQL arrays are not supported \
501             (got {ndims} dimensions). See https://github.com/tokio-rs/toasty/issues/870"
502        );
503    }
504
505    let mut values = array.values();
506    let (cap, _) = values.size_hint();
507    let mut out = Vec::with_capacity(cap);
508    while let Some(elem) = values
509        .next()
510        .expect("invalid PostgreSQL array element framing")
511    {
512        out.push(match elem {
513            None => stmt::Value::Null,
514            Some(bytes) => decode_array_element(elem_pg_ty, bytes, elem_ty),
515        });
516    }
517    Some(out)
518}

520/// Decode a single PostgreSQL array element from its raw wire bytes and
521/// project it into the Toasty value space using `elem_ty`. Reuses the
522/// scalar-column `_to_value` helpers so a `text[]` element and a `text`
523/// column decode through identical logic.
524fn decode_array_element(elem_pg_ty: &Type, bytes: &[u8], elem_ty: &stmt::Type) -> stmt::Value {
525    if elem_pg_ty == &Type::TEXT || elem_pg_ty == &Type::VARCHAR {
526        text_to_value(
527            String::from_sql(elem_pg_ty, bytes).expect("decode TEXT array element"),
528            elem_ty,
529        )
530    } else if elem_pg_ty == &Type::BOOL {
531        stmt::Value::Bool(bool::from_sql(elem_pg_ty, bytes).expect("decode BOOL array element"))
532    } else if elem_pg_ty == &Type::INT2 {
533        int2_to_value(
534            i16::from_sql(elem_pg_ty, bytes).expect("decode INT2 array element"),
535            elem_ty,
536        )
537    } else if elem_pg_ty == &Type::INT4 {
538        int4_to_value(
539            i32::from_sql(elem_pg_ty, bytes).expect("decode INT4 array element"),
540            elem_ty,
541        )
542    } else if elem_pg_ty == &Type::INT8 {
543        int8_to_value(
544            i64::from_sql(elem_pg_ty, bytes).expect("decode INT8 array element"),
545            elem_ty,
546        )
547    } else if elem_pg_ty == &Type::FLOAT4 {
548        float4_to_value(
549            f32::from_sql(elem_pg_ty, bytes).expect("decode FLOAT4 array element"),
550            elem_ty,
551        )
552    } else if elem_pg_ty == &Type::FLOAT8 {
553        float8_to_value(
554            f64::from_sql(elem_pg_ty, bytes).expect("decode FLOAT8 array element"),
555            elem_ty,
556        )
557    } else if elem_pg_ty == &Type::NUMERIC {
558        #[cfg(feature = "rust_decimal")]
559        {
560            stmt::Value::Decimal(
561                rust_decimal::Decimal::from_sql(elem_pg_ty, bytes)
562                    .expect("decode NUMERIC array element"),
563            )
564        }
565        #[cfg(not(feature = "rust_decimal"))]
566        {
567            panic!("NUMERIC array elements require rust_decimal feature to be enabled")
568        }
569    } else if elem_pg_ty == &Type::UUID {
570        stmt::Value::Uuid(
571            uuid::Uuid::from_sql(elem_pg_ty, bytes).expect("decode UUID array element"),
572        )
573    } else if elem_pg_ty == &Type::TIMESTAMPTZ {
574        #[cfg(feature = "jiff")]
575        {
576            stmt::Value::Timestamp(
577                jiff::Timestamp::from_sql(elem_pg_ty, bytes)
578                    .expect("decode TIMESTAMPTZ array element"),
579            )
580        }
581        #[cfg(not(feature = "jiff"))]
582        {
583            panic!("TIMESTAMPTZ requires jiff feature to be enabled")
584        }
585    } else if elem_pg_ty == &Type::TIMESTAMP {
586        #[cfg(feature = "jiff")]
587        {
588            stmt::Value::DateTime(
589                jiff::civil::DateTime::from_sql(elem_pg_ty, bytes)
590                    .expect("decode TIMESTAMP array element"),
591            )
592        }
593        #[cfg(not(feature = "jiff"))]
594        {
595            panic!("TIMESTAMP requires jiff feature to be enabled")
596        }
597    } else if elem_pg_ty == &Type::DATE {
598        #[cfg(feature = "jiff")]
599        {
600            stmt::Value::Date(
601                jiff::civil::Date::from_sql(elem_pg_ty, bytes).expect("decode DATE array element"),
602            )
603        }
604        #[cfg(not(feature = "jiff"))]
605        {
606            panic!("DATE requires jiff feature to be enabled")
607        }
608    } else if elem_pg_ty == &Type::TIME {
609        #[cfg(feature = "jiff")]
610        {
611            stmt::Value::Time(
612                jiff::civil::Time::from_sql(elem_pg_ty, bytes).expect("decode TIME array element"),
613            )
614        }
615        #[cfg(not(feature = "jiff"))]
616        {
617            panic!("TIME requires jiff feature to be enabled")
618        }
619    } else if elem_pg_ty == &Type::CIDR {
620        #[cfg(feature = "net")]
621        {
622            stmt::Value::Cidr(decode_cidr(bytes))
623        }
624        #[cfg(not(feature = "net"))]
625        {
626            panic!("CIDR requires net feature to be enabled")
627        }
628    } else if elem_pg_ty == &Type::INET {
629        #[cfg(feature = "net")]
630        {
631            stmt::Value::Inet(decode_inet(bytes))
632        }
633        #[cfg(not(feature = "net"))]
634        {
635            panic!("INET requires net feature to be enabled")
636        }
637    } else if elem_pg_ty == &Type::MACADDR {
638        #[cfg(feature = "net")]
639        {
640            stmt::Value::MacAddr(decode_macaddr(bytes))
641        }
642        #[cfg(not(feature = "net"))]
643        {
644            panic!("MACADDR requires net feature to be enabled")
645        }
646    } else if elem_pg_ty == &Type::MACADDR8 {
647        #[cfg(feature = "net")]
648        {
649            stmt::Value::MacAddr8(decode_macaddr8(bytes))
650        }
651        #[cfg(not(feature = "net"))]
652        {
653            panic!("MACADDR8 requires net feature to be enabled")
654        }
655    } else if matches!(elem_pg_ty.kind(), Kind::Enum(_)) {
656        // Enum labels are plain UTF-8; `EnumString` accepts `Kind::Enum` where
657        // `String` won't.
658        let EnumString(v) =
659            EnumString::from_sql(elem_pg_ty, bytes).expect("decode ENUM array element");
660        text_to_value(v, elem_ty)
661    } else {
662        todo!(
663            "implement PostgreSQL array decoding for element type `{:#?}`",
664            elem_pg_ty
665        )
666    }
667}

669impl ToSql for Value {
670    fn to_sql(
671        &self,
672        ty: &Type,
673        out: &mut BytesMut,
674    ) -> std::result::Result<IsNull, Box<dyn std::error::Error + Sync + Send>>
675    where
676        Self: Sized,
677    {
678        value_to_sql(&self.0, ty, out)
679    }
680
681    fn accepts(ty: &Type) -> bool {
682        matches!(
683            *ty,
684            Type::BOOL
685                | Type::INT2
686                | Type::INT4
687                | Type::INT8
688                | Type::TEXT
689                | Type::FLOAT4
690                | Type::FLOAT8
691                | Type::VARCHAR
692                | Type::BYTEA
693                | Type::UUID
694                | Type::NUMERIC
695                | Type::TIMESTAMP
696                | Type::TIMESTAMPTZ
697                | Type::DATE
698                | Type::TIME
699                | Type::JSONB
700                | Type::JSON
701                | Type::CIDR
702                | Type::INET
703                | Type::MACADDR
704                | Type::MACADDR8
705        ) || matches!(ty.kind(), Kind::Enum(_) | Kind::Array(_))
706    }
707    to_sql_checked!();
708}
709
710/// Free-fn form of `Value::to_sql` so the array-element closure can call it
711/// with a `&CoreValue` borrowed from the input slice — no per-element clone
712/// or wrapper construction.
713fn value_to_sql(
714    value: &CoreValue,
715    ty: &Type,
716    out: &mut BytesMut,
717) -> std::result::Result<IsNull, Box<dyn std::error::Error + Sync + Send>> {
718    match (value, ty) {
719        (stmt::Value::Bool(value), _) => value.to_sql(ty, out),
720        (stmt::Value::I8(value), &Type::INT2) => (*value as i16).to_sql(ty, out),
721        (stmt::Value::I8(value), &Type::INT4) => (*value as i32).to_sql(ty, out),
722        (stmt::Value::I8(value), &Type::INT8) => (*value as i64).to_sql(ty, out),
723        (stmt::Value::I16(value), &Type::INT2) => value.to_sql(ty, out),
724        (stmt::Value::I16(value), &Type::INT4) => (*value as i32).to_sql(ty, out),
725        (stmt::Value::I16(value), &Type::INT8) => (*value as i64).to_sql(ty, out),
726        (stmt::Value::I32(value), &Type::INT4) => value.to_sql(ty, out),
727        (stmt::Value::I32(value), &Type::INT8) => (*value as i64).to_sql(ty, out),
728        (stmt::Value::I64(value), &Type::INT4) => (*value as i32).to_sql(ty, out),
729        (stmt::Value::I64(value), &Type::INT8) => value.to_sql(ty, out),
730        (stmt::Value::U8(value), &Type::INT2) => (*value as i16).to_sql(ty, out),
731        (stmt::Value::U8(value), &Type::INT4) => (*value as i32).to_sql(ty, out),
732        (stmt::Value::U8(value), &Type::INT8) => (*value as i64).to_sql(ty, out),
733        (stmt::Value::U16(value), &Type::INT4) => (*value as i32).to_sql(ty, out),
734        (stmt::Value::U16(value), &Type::INT8) => (*value as i64).to_sql(ty, out),
735        (stmt::Value::U32(value), &Type::INT8) => (*value as i64).to_sql(ty, out),
736        (stmt::Value::U64(value), &Type::INT8) => {
737            if *value > i64::MAX as u64 {
738                return Err(Box::new(std::io::Error::new(
739                    std::io::ErrorKind::InvalidData,
740                    format!(
741                        "u64 value {} exceeds i64::MAX ({}), cannot store in PostgreSQL BIGINT",
742                        value,
743                        i64::MAX
744                    ),
745                )));
746            }
747            (*value as i64).to_sql(ty, out)
748        }
749        (stmt::Value::F32(value), &Type::FLOAT4) => value.to_sql(ty, out),
750        (stmt::Value::F32(value), &Type::FLOAT8) => (*value as f64).to_sql(ty, out),
751        (stmt::Value::F64(value), &Type::FLOAT4) => (*value as f32).to_sql(ty, out),
752        (stmt::Value::F64(value), &Type::FLOAT8) => value.to_sql(ty, out),
753        (stmt::Value::Null, _) => Ok(IsNull::Yes),
754        // `Json<T>` values are already serialized once by serde. Bind their
755        // bytes directly instead of decoding and re-encoding a JSON tree.
756        (stmt::Value::String(value), &Type::JSONB) => {
757            out.extend_from_slice(&[1]);
758            out.extend_from_slice(value.as_bytes());
759            Ok(IsNull::No)
760        }
761        (stmt::Value::String(value), &Type::JSON) => {
762            out.extend_from_slice(value.as_bytes());
763            Ok(IsNull::No)
764        }
765        // PG enums are wire-encoded as plain UTF-8 text. `String::ToSql::accepts`
766        // rejects `Kind::Enum`, so write the bytes directly.
767        (stmt::Value::String(value), _) if matches!(ty.kind(), Kind::Enum(_)) => {
768            out.extend_from_slice(value.as_bytes());
769            Ok(IsNull::No)
770        }
771        (stmt::Value::String(value), _) => value.to_sql(ty, out),
772        (stmt::Value::Bytes(value), &Type::BYTEA) => value.to_sql(ty, out),
773        (stmt::Value::Uuid(value), &Type::UUID) => value.to_sql(ty, out),
774        #[cfg(feature = "rust_decimal")]
775        (stmt::Value::Decimal(value), _) => value.to_sql(ty, out),
776        #[cfg(feature = "jiff")]
777        (stmt::Value::Timestamp(value), _) => value.to_sql(ty, out),
778        #[cfg(feature = "jiff")]
779        (stmt::Value::Date(value), _) => value.to_sql(ty, out),
780        #[cfg(feature = "jiff")]
781        (stmt::Value::Time(value), _) => value.to_sql(ty, out),
782        #[cfg(feature = "jiff")]
783        (stmt::Value::DateTime(value), _) => value.to_sql(ty, out),
784        #[cfg(feature = "net")]
785        (stmt::Value::Cidr(value), &Type::CIDR) => {
786            inet_to_sql(value.first_address(), value.network_length(), out);
787            Ok(IsNull::No)
788        }
789        #[cfg(feature = "net")]
790        (stmt::Value::Inet(value), &Type::INET) => {
791            inet_to_sql(value.address(), value.network_length(), out);
792            Ok(IsNull::No)
793        }
794        #[cfg(feature = "net")]
795        (stmt::Value::MacAddr(value), &Type::MACADDR) => {
796            out.extend_from_slice(value.as_bytes());
797            Ok(IsNull::No)
798        }
799        #[cfg(feature = "net")]
800        (stmt::Value::MacAddr8(value), &Type::MACADDR8) => {
801            out.extend_from_slice(value.as_bytes());
802            Ok(IsNull::No)
803        }
804        // `#[document]` columns: serialize the structural value (a
805        // `Value::List` of `Value::Object`s) to JSON and bind it. The engine
806        // has already converted positional records to named objects, so this
807        // is a plain structural encode.
808        (value, &Type::JSONB) => {
809            let bytes = toasty_sql::json::to_vec(value)
810                .map_err(|e| Box::new(e) as Box<dyn std::error::Error + Sync + Send>)?;
811            // `jsonb` wire format: a 1-byte version header, then UTF-8 text.
812            out.extend_from_slice(&[1]);
813            out.extend_from_slice(&bytes);
814            Ok(IsNull::No)
815        }
816        (value, &Type::JSON) => {
817            let bytes = toasty_sql::json::to_vec(value)
818                .map_err(|e| Box::new(e) as Box<dyn std::error::Error + Sync + Send>)?;
819            out.extend_from_slice(&bytes);
820            Ok(IsNull::No)
821        }
822        // List → bind as a PostgreSQL array via the streaming `array_to_sql`
823        // primitive: the closure runs per element and writes directly into
824        // `out`, so there's no intermediate `Vec<Option<T>>`. The element PG
825        // type (carried by the prepared statement, see `db::Type::List` →
826        // `to_postgres_type`) drives per-item conversion via this same fn.
827        (stmt::Value::List(items), _) => {
828            let Kind::Array(elem) = ty.kind() else {
829                return Err(format!("Value::List bound to non-array PG type {ty:?}").into());
830            };
831            let len = i32::try_from(items.len())
832                .map_err(|_| format!("array length {} exceeds i32::MAX", items.len()))?;
833            array_to_sql(
834                [ArrayDimension {
835                    len,
836                    lower_bound: 1,
837                }],
838                elem.oid(),
839                items.iter(),
840                |v, buf| match value_to_sql(v, elem, buf)? {
841                    IsNull::No => Ok(postgres_protocol::IsNull::No),
842                    IsNull::Yes => Ok(postgres_protocol::IsNull::Yes),
843                },
844                out,
845            )?;
846            Ok(IsNull::No)
847        }
848        (value, _) => todo!("unsupported Value for PostgreSQL type: {value:#?}, type: {ty:#?}"),
849    }
850}

852#[cfg(feature = "net")]
853fn decode_cidr(raw: &[u8]) -> IpCidr {
854    let value = inet_from_sql(raw).expect("decode PostgreSQL CIDR");
855    IpCidr::new(value.addr(), value.netmask()).expect("PostgreSQL returned invalid CIDR")
856}
857
858#[cfg(feature = "net")]
859fn decode_inet(raw: &[u8]) -> IpInet {
860    let value = inet_from_sql(raw).expect("decode PostgreSQL INET");
861    IpInet::new(value.addr(), value.netmask()).expect("PostgreSQL returned invalid INET")
862}
863
864#[cfg(feature = "net")]
865fn decode_macaddr(raw: &[u8]) -> MacAddr6 {
866    macaddr_from_sql(raw)
867        .map(MacAddr6::from)
868        .expect("decode PostgreSQL MACADDR")
869}
870
871#[cfg(feature = "net")]
872fn decode_macaddr8(raw: &[u8]) -> MacAddr8 {
873    let bytes: [u8; 8] = raw
874        .try_into()
875        .expect("invalid PostgreSQL MACADDR8 wire length");
876    MacAddr8::from(bytes)
