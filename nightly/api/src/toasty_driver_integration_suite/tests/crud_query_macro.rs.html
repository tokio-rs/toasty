3#[driver_test(scenario(crate::scenarios::two_models), requires(scan))]
4pub async fn query_macro_scan_filters(test: &mut Test) -> Result<()> {
5    let mut db = setup(test).await;
6
7    toasty::create!(User::[{ name: "Alice" }, { name: "Bob" }, { name: "Carl" }])
8        .exec(&mut db)
9        .await?;
10
11    // query!(User) expands to User::all()
12    let users = toasty::query!(User).exec(&mut db).await?;
13    assert_struct!(users, #({ name: "Alice" }, { name: "Bob" }, { name: "Carl" }));
14
15    let users = toasty::query!(User filter .name == "Alice")
16        .exec(&mut db)
17        .await?;
18    assert_struct!(users, [{ name: "Alice" }]);
19
20    let users = toasty::query!(User filter .name != "Alice")
21        .exec(&mut db)
22        .await?;
23    assert_struct!(users, #({ name: "Bob" }, { name: "Carl" }));
24
25    let users = toasty::query!(User filter .name == "Alice" or .name == "Bob")
26        .exec(&mut db)
27        .await?;
28    assert_struct!(users, #({ name: "Alice" }, { name: "Bob" }));
29
30    let users = toasty::query!(User filter not .name == "Alice")
31        .exec(&mut db)
32        .await?;
33    assert_struct!(users, #({ name: "Bob" }, { name: "Carl" }));
34
35    let target_name = "Alice";
36    let users = toasty::query!(User filter .name == #target_name)
37        .exec(&mut db)
38        .await?;
39    assert_struct!(users, [{ name: "Alice" }]);
40
41    fn get_name() -> &'static str {
42        "Bob"
43    }
44    let users = toasty::query!(User filter .name == #(get_name()))
45        .exec(&mut db)
46        .await?;
47    assert_struct!(users, [{ name: "Bob" }]);
48
49    let users = toasty::query!(User FILTER .name == "Alice")
50        .exec(&mut db)
51        .await?;
52    assert_struct!(users, [{ name: "Alice" }]);
53
54    let users = toasty::query!(User Filter .name == "Alice" AND .name == "Alice")
55        .exec(&mut db)
56        .await?;
57    assert_struct!(users, [{ name: "Alice" }]);
58
59    let users = toasty::query!(User filter .name IN ["Alice", "Carl"])
60        .exec(&mut db)
61        .await?;
62    assert_struct!(users, #({ name: "Alice" }, { name: "Carl" }));
63
64    let names = vec!["Alice", "Bob"];
65    let users = toasty::query!(User filter .name IN #names)
66        .exec(&mut db)
67        .await?;
68    assert_struct!(users, #({ name: "Alice" }, { name: "Bob" }));
69
70    let users = toasty::query!(User filter .name IN ["Alice", "Bob", "Carl"] and .name != "Bob")
71        .exec(&mut db)
72        .await?;
73    assert_struct!(users, #({ name: "Alice" }, { name: "Carl" }));
74
75    Ok(())
76}

78#[driver_test(scenario(crate::scenarios::user_with_age), requires(scan))]
79pub async fn query_macro_numeric_and_boolean_filters(test: &mut Test) -> Result<()> {
80    let mut db = setup(test).await;
81
82    toasty::create!(User::[
83        { name: "Young", age: 15 },
84        { name: "Adult", age: 25 },
85        { name: "Senior", age: 65 },
86        { name: "Alice", age: 30 },
87        { name: "Bob", age: 30 },
88        { name: "Alice", age: 20 },
89        { name: "Carl", age: 40 },
90        { name: "Diana", age: 10 },
91    ])
92    .exec(&mut db)
93    .await?;
94
95    // Greater than
96    let users = toasty::query!(User filter .age > 20).exec(&mut db).await?;
97    assert_struct!(users, #(
98        { name: "Adult" },
99        { name: "Senior" },
100        { name: "Alice" },
101        { name: "Bob" },
102        { name: "Carl" },
103    ));
104
105    // Greater than or equal
106    let users = toasty::query!(User filter .age >= 25).exec(&mut db).await?;
107    assert_struct!(users, #(
108        { name: "Adult" },
109        { name: "Senior" },
110        { name: "Alice" },
111        { name: "Bob" },
112        { name: "Carl" },
113    ));
114
115    // Less than
116    let users = toasty::query!(User filter .age < 25).exec(&mut db).await?;
117    assert_struct!(users, #(
118        { name: "Young" },
119        { name: "Alice", age: 20 },
120        { name: "Diana" },
121    ));
122
123    // Less than or equal
124    let users = toasty::query!(User filter .age <= 25).exec(&mut db).await?;
125    assert_struct!(users, #(
126        { name: "Young" },
127        { name: "Adult" },
128        { name: "Alice", age: 20 },
129        { name: "Diana" },
130    ));
131
132    let users = toasty::query!(User filter .name == "Alice" and .age == 30)
133        .exec(&mut db)
134        .await?;
135    assert_struct!(users, [{ name: "Alice", age: 30 }]);
136
137    // AND binds tighter than OR, so parentheses change the grouping:
138    // .name == "Alice" AND (.age > 25 OR .age < 15)
139    let users = toasty::query!(User filter .name == "Alice" and (.age > 25 or .age < 15))
140        .exec(&mut db)
141        .await?;
142    assert_struct!(users, [{ name: "Alice" }]);
143
144    // Complex: NOT (.age < 18) AND (.name == "Alice" OR .name == "Carl")
145    let users =
146        toasty::query!(User filter not (.age < 18) and (.name == "Alice" or .name == "Carl"))
147            .exec(&mut db)
148            .await?;
149    assert_struct!(users, #(
150        { name: "Alice", age: 30 },
151        { name: "Alice", age: 20 },
152        { name: "Carl" },
153    ));
154
155    Ok(())
156}

181pub async fn query_macro_ordering_and_pagination(test: &mut Test) -> Result<()> {
182    let mut db = setup(test).await;
183
184    toasty::create!(User::[{ name: "Carl" }, { name: "Alice" }, { name: "Bob" }, { name: "Diana" }])
185        .exec(&mut db)
186        .await?;
187
188    let users = toasty::query!(User ORDER BY .name ASC)
189        .exec(&mut db)
190        .await?;
191    assert_struct!(users, [
192        { name: "Alice" },
193        { name: "Bob" },
194        { name: "Carl" },
195        { name: "Diana" },
196    ]);
197
198    let users = toasty::query!(User ORDER BY .name DESC)
199        .exec(&mut db)
200        .await?;
201    assert_struct!(users, [
202        { name: "Diana" },
203        { name: "Carl" },
204        { name: "Bob" },
205        { name: "Alice" },
206    ]);
207
208    let users = toasty::query!(User ORDER BY .name ASC LIMIT 2)
209        .exec(&mut db)
210        .await?;
211    assert_eq!(users.len(), 2);
212    assert_struct!(users, [{ name: "Alice" }, { name: "Bob" }]);
213
214    // Skip 1, take 2 (ordered by name ascending)
215    let users = toasty::query!(User ORDER BY .name ASC OFFSET 1 LIMIT 2)
216        .exec(&mut db)
217        .await?;
218    assert_struct!(users, [{ name: "Bob" }, { name: "Carl" }]);
219
220    let n = 1;
221    let users = toasty::query!(User ORDER BY .name ASC LIMIT #n)
222        .exec(&mut db)
223        .await?;
224    assert_struct!(users, [{ name: "Alice" }]);
225
226    let users = toasty::query!(User order by .name asc limit 2)
227        .exec(&mut db)
228        .await?;
229    assert_struct!(users, [{ name: "Alice" }, { name: "Bob" }]);
230
231    Ok(())
232}

261pub async fn query_macro_partition_key_filters(test: &mut Test) -> Result<()> {
262    #[derive(Debug, toasty::Model)]
263    #[key(partition = league, local = name)]
264    struct Team {
265        league: String,
266
267        name: String,
268
269        founded: i64,
270    }
271
272    let mut db = test.setup_db(models!(Team)).await;
273
274    for (league, name, founded) in [
275        ("MLS", "Portland Timbers", 2009),
276        ("MLS", "Seattle Sounders FC", 2007),
277        ("EPL", "Arsenal", 1886),
278        ("EPL", "Chelsea", 1905),
279    ] {
280        toasty::create!(Team {
281            league: league,
282            name: name,
283            founded: founded
284        })
285        .exec(&mut db)
286        .await?;
287    }
288
289    // Filter on partition key only
290    let teams = toasty::query!(Team filter .league == "EPL")
291        .exec(&mut db)
292        .await?;
293
294    assert_struct!(teams, #({ name: "Arsenal" }, { name: "Chelsea" }));
295
296    // Filter on partition key + local key
297    let teams = toasty::query!(Team filter .league == "MLS" and .name == "Portland Timbers")
298        .exec(&mut db)
299        .await?;
300
301    assert_struct!(teams, [{ name: "Portland Timbers", founded: 2009 }]);
302
303    let target_league = "MLS";
304    let teams = toasty::query!(Team filter .league == #target_league)
305        .exec(&mut db)
306        .await?;
307    assert_struct!(teams, #({ name: "Portland Timbers" }, { name: "Seattle Sounders FC" }));
308
309    Ok(())

337            timestamp: ts
338        })
339        .exec(&mut db)

444pub async fn query_macro_filter_in_list_by_pk(test: &mut Test) -> Result<()> {
445    let mut db = setup(test).await;
