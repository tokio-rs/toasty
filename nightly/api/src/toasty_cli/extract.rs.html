1//! Schema extraction: build the user's package and read the schema back out
2//! of the artifact. The dump constructor in `toasty::schema_dump` fires when
3//! `TOASTY_DUMP_SCHEMA` is set — before `main` for a bin, during `dlopen` for
4//! a cdylib — writes a JSON [`SchemaDump`] to stdout, and exits.
5
6use crate::cargo::{self, BuildTarget};
7use crate::{Flavor, Project};
8
9use anyhow::{Context, Result, bail};
10use console::style;
11use std::path::Path;
12use std::process::{Command, Stdio};
13use toasty::schema::db;
14use toasty::schema_dump::{
15    DUMP_SCHEMA_ENV, DUMP_TABLE_NAME_PREFIX_ENV, SCHEMA_DUMP_VERSION, SchemaDump,
16};

18/// Builds the selected target of `project` and extracts its schema, lowered
19/// for `flavor`.
20pub fn extract_schema(project: &Project, flavor: Flavor, bin: Option<&str>) -> Result<db::Schema> {
21    // Running the artifact is only safe when `toasty` is linked: its
22    // constructor dumps and exits before `main`. Without it the environment
23    // variable is inert and the user's program would run.
24    if !cargo::links_toasty(
25        &project.workspace_root,
26        &project.package_name,
27        &project.features,
28    )? {
29        bail!(
30            "`{}` does not depend on `toasty`, directly or transitively, so it has no schema \
31             to extract; select the package that does with `-p <package>`",
32            project.package_name
33        );
34    }
35
36    let target = project.build_target(bin)?;
37
38    // Progress goes to stderr so `migrate snapshot` stdout stays parseable.
39    eprintln!(
40        "  {} Compiling {}...",
41        style("→").cyan(),
42        style(&project.package_name).bold()
43    );
44
45    let artifact = cargo::build_artifact(
46        &project.workspace_root,
47        &project.package_name,
48        &target,
49        &project.features,
50    )?;
51
52    let prefix = project.config.migration.table_name_prefix.as_deref();
53    let output = run_dumper(&artifact, &target, flavor, prefix)?;
54    let schema = parse_dump(&output, &project.package_name)?;
55
56    check_not_empty(&schema, project, &target)?;
57
58    Ok(schema)
59}

61/// Runs the built artifact so the dump constructor fires, and captures its
62/// stdout.
63fn run_dumper(
64    artifact: &Path,
65    target: &BuildTarget,
66    flavor: Flavor,
67    table_name_prefix: Option<&str>,
68) -> Result<Vec<u8>> {
69    let mut cmd = match target {
70        BuildTarget::Bin(_) => {
71            let mut cmd = Command::new(artifact);
72            cmd.env(DUMP_SCHEMA_ENV, flavor.as_str());
73            // Don't let an inherited prefix through; `Toasty.toml` is the
74            // only source for it.
75            match table_name_prefix {
76                Some(prefix) => cmd.env(DUMP_TABLE_NAME_PREFIX_ENV, prefix),
77                None => cmd.env_remove(DUMP_TABLE_NAME_PREFIX_ENV),
78            };
79            cmd
80        }
81        BuildTarget::Cdylib => {
82            // A cdylib has no entry point, and its constructor exits the
83            // process that loads it, so the CLI re-execs itself to `dlopen`
84            // it in a child. The subcommand sets the environment variable
85            // itself: set here, a debug build of the CLI (which links
86            // `toasty`) would dump its own empty schema at startup.
87            let current_exe =
88                std::env::current_exe().context("failed to locate the `toasty` executable")?;
89            let mut cmd = Command::new(current_exe);
90            cmd.arg("__load-cdylib")
91                .arg(artifact)
92                .args(["--flavor", flavor.as_str()]);
93            if let Some(prefix) = table_name_prefix {
94                cmd.args(["--table-name-prefix", prefix]);
95            }
96            cmd
97        }
98    };
99
100    let output = cmd
101        .stdout(Stdio::piped())
102        .stderr(Stdio::inherit())
103        .output()
104        .with_context(|| format!("failed to run schema dumper `{}`", artifact.display()))?;
105
106    if !output.status.success() {
107        bail!("the schema dumper exited with {}", output.status);
108    }
109
110    Ok(output.stdout)
111}

113/// Parses the dumper's stdout into a schema.
114///
115/// Other constructors in the user's dependency graph may also write to
116/// stdout, so every line is tried and unrelated JSON with a `version` field
117/// must not mask the real dump. A version mismatch is reported only once no
118/// line decoded.
119fn parse_dump(output: &[u8], package: &str) -> Result<db::Schema> {
120    let text = String::from_utf8_lossy(output);
121    let mut mismatched_version = None;
122    let mut decode_error = None;
123
124    for value in text
125        .lines()
126        .filter_map(|line| serde_json::from_str::<serde_json::Value>(line).ok())
127    {
128        let Some(version) = value.get("version").and_then(serde_json::Value::as_u64) else {
129            continue;
130        };
131
132        if !value
133            .get("schema")
134            .is_some_and(serde_json::Value::is_object)
135        {
136            continue;
137        }
138
139        if version != u64::from(SCHEMA_DUMP_VERSION) {
140            mismatched_version.get_or_insert(version);
141            continue;
142        }
143
144        match serde_json::from_value::<SchemaDump>(value) {
145            Ok(dump) => return Ok(dump.schema),
146            Err(err) => decode_error.get_or_insert(err),
147        };
148    }
149
150    if let Some(version) = mismatched_version {
151        bail!(
152            "the schema dumper reported dump format version {version}, but this CLI speaks \
153             version {SCHEMA_DUMP_VERSION}; update `toasty-cli` or the project's `toasty` \
154             dependency so the two match"
155        );
156    }
157
158    if let Some(err) = decode_error {
159        return Err(
160            anyhow::Error::new(err).context("failed to decode the schema produced by the dumper")
161        );
162    }
163
164    // Reachability is not linkage: rustc only links crates the target
165    // actually references, so this bin may carry no constructor at all.
166    bail!(
167        "the schema dumper produced no schema; check that `{package}` references the crate \
168         holding the models, and that its `dev` profile does not strip link-time constructors"
169    )
170}

172/// Rejects a schema with no tables: generating from it would `DROP TABLE`
173/// everything in the previous snapshot. The usual cause is a bin that links
174/// `toasty` but never references the crate holding the models.
175fn check_not_empty(schema: &db::Schema, project: &Project, target: &BuildTarget) -> Result<()> {
176    if !schema.tables.is_empty() {
177        return Ok(());
178    }
179
180    let hint = match target {
181        BuildTarget::Bin(name) if project.has_lib => format!(
182            "the bin target `{name}` was built, but the models may live in the lib target; \
183             `--bin <name>` selects a different one"
184        ),
185        BuildTarget::Bin(name) => {
186            format!("check that the bin target `{name}` references the models")
187        }
188        BuildTarget::Cdylib => "check that the lib target defines models".to_string(),
189    };
190
191    bail!(
192        "the schema extracted from `{}` has no tables, so generating would drop every \
193         existing table; {hint}",
194        project.package_name
195    )
196}

198#[cfg(test)]
199mod tests {
200    use super::*;
201
202    #[test]
203    fn version_is_checked_before_the_payload_is_decoded() {
204        // A newer envelope may lay `schema` out differently; decoding it first
205        // would turn the version mismatch into a misleading decode error.
206        let output = br#"{"version":999,"schema":{"shaped":"differently"}}"#;
207
208        let err = parse_dump(output, "app").unwrap_err();
209
210        let message = format!("{err:#}");
211        assert!(message.contains("version 999"), "{message}");
212        assert!(message.contains("toasty-cli"), "{message}");
213    }
214
215    #[test]
216    fn the_dump_is_found_among_other_constructor_output() {
217        // Other constructors may print plain text or JSON, including JSON
218        // with its own `version` field; none of it may mask the real dump.
219        let dump = serde_json::to_string(&SchemaDump {
220            version: SCHEMA_DUMP_VERSION,
221            schema: db::Schema::default(),
222        })
223        .unwrap();
224        let output = format!(
225            "another ctor said hello\n{{\"unrelated\":true}}\n\
226             {{\"version\":\"1.0\",\"schema\":\"public\"}}\n{dump}\n"
227        );
228
229        assert!(parse_dump(output.as_bytes(), "app").is_ok());
230    }
