1use anyhow::{Context, Result};
2use clap::Parser;
3use std::path::PathBuf;
4use toasty::schema_dump::{DUMP_SCHEMA_ENV, DUMP_TABLE_NAME_PREFIX_ENV};
5
6/// Hidden subcommand: loads a cdylib so the schema-dump constructor inside
7/// `toasty` runs. On success the constructor dumps and exits this process
8/// during `Library::new`; the parent captures stdout as it would for a bin.
9#[derive(Parser, Debug)]
10pub struct LoadCdylibCommand {
11    /// Path of the cdylib to load
12    path: PathBuf,
13
14    /// Flavor to dump the schema for
15    #[arg(long)]
16    flavor: String,
17
18    /// Table name prefix to lower the schema with
19    #[arg(long)]
20    table_name_prefix: Option<String>,
21}
22
23impl LoadCdylibCommand {
24    pub(crate) fn run(self) -> Result<()> {
25        // Set here rather than by the parent so a debug build of the CLI,
26        // which links `toasty` and carries the constructor itself, does not
27        // dump its own empty schema at startup.
28        //
29        // SAFETY: `main` runs this before building the async runtime, so the
30        // process is still single-threaded.
31        #[allow(unsafe_code)]
32        unsafe {
33            std::env::set_var(DUMP_SCHEMA_ENV, &self.flavor);
34
35            // Don't let an inherited prefix through; the argument is the only
36            // source for it.
37            match &self.table_name_prefix {
38                Some(prefix) => std::env::set_var(DUMP_TABLE_NAME_PREFIX_ENV, prefix),
39                None => std::env::remove_var(DUMP_TABLE_NAME_PREFIX_ENV),
40            }
41        }
42
43        // SAFETY: loading runs the library's constructors, which is the point:
44        // the dump constructor prints the schema and exits. The library was
45        // just built by the parent from the user's own package.
46        #[allow(unsafe_code)]
47        let library = unsafe { libloading::Library::new(&self.path) };
48
49        library.with_context(|| format!("failed to load `{}`", self.path.display()))?;
50
51        // No constructor fired (the library does not link `toasty`). Exit
52        // cleanly; the parent diagnoses the missing dump.
53        Ok(())
54    }
55}
