3//! Toasty driver for [Turso](https://turso.tech/), an async-native,
4//! SQLite-compatible database engine.
5//!
6//! Speaks the same SQL dialect as the [SQLite driver][toasty-driver-sqlite]
7//! but uses the async Turso client. Supports file-backed and in-memory
8//! databases, an optional concurrent-writes mode that uses Turso's MVCC
9//! journal so transactions don't serialize on a single writer, and per-driver
10//! toggles for Turso's experimental features (`experimental_encryption`,
11//! `experimental_attach`, etc.) that mirror [`turso::Builder`].
12//!
13//! Cargo features only make capabilities available; which engine a driver
14//! opens is decided at runtime by its configuration. With the `sync`
15//! feature, a driver configured with [`Turso::with_remote_url`] (or any
16//! other sync option, or [`Turso::with_sync`]) opens Turso's sync engine;
17//! an unconfigured driver always opens the plain local engine, feature or
18//! not.
19//!
20//! With the `serverless` feature, the driver also connects to a remote
21//! Turso Cloud database over HTTP via the [`turso_serverless`] crate. A
22//! connection URL with a host selects serverless mode; a URL with only a
23//! path (or `:memory:`) selects the embedded engine.
24//!
25//! [toasty-driver-sqlite]: https://docs.rs/toasty-driver-sqlite
26//!
27//! # Examples
28//!
29//! ```
30//! use toasty_driver_turso::Turso;
31//!
32//! // In-memory database
33//! let driver = Turso::in_memory();
34//!
35//! // File-backed database
36//! let driver = Turso::file("path/to/db");
37//!
38//! // Allow transactions to run concurrently instead of serializing writers
39//! let driver = Turso::file("path/to/db").concurrent_writes();
40//! ```
41//!
42//! ```rust,ignore
43//! // Turso Cloud over HTTP (requires the `serverless` feature)
44//! let driver = Turso::new("turso://my-db.aws-us-east-1.turso.io")?
45//!     .with_auth_token("<token>");
46//! ```

108fn create_table_stmts(schema: &db::Schema, table: &Table) -> Vec<String> {
109    let serializer = sql::Serializer::sqlite(schema);
110
111    let mut stmts =
112        vec![serializer.serialize(&sql::Statement::create_table(table, &Capability::SQLITE))];
113
114    for index in &table.indices {
115        if index.primary_key {
116            continue;
117        }
118        stmts.push(serializer.serialize(&sql::Statement::create_index(index)));
119    }
120
121    stmts
122}

124/// Retries an operation while the database reports a retryable
125/// conflict (busy write lock, serialization failure).
126///
127/// A lock-taking `BEGIN` fails fast with "busy" while contended, and the
128/// serverless backend has no server-side busy timeout, so the wait
129/// happens here, with backoff. Retrying is safe for the two callers: a
130/// busy `BEGIN` acquired nothing, and a failed transactional batch is
131/// atomic — it leaves nothing applied on either transport.
132async fn retry_while_busy<T, F, Fut>(mut op: F) -> Result<T>
133where
134    F: FnMut() -> Fut,
135    Fut: std::future::Future<Output = Result<T>>,
136{
137    const RETRY_FOR: Duration = Duration::from_secs(10);
138
139    let deadline = Instant::now() + RETRY_FOR;
140    let mut delay = Duration::from_millis(10);
141    loop {
142        match op().await {
143            Err(err) if err.is_serialization_failure() && Instant::now() < deadline => {
144                tokio::time::sleep(delay).await;
145                delay = (delay * 2).min(Duration::from_millis(250));
146            }
147            res => return res,
148        }
149    }
150}

152async fn exec_ddl(
153    conn: &AnyConn,
154    statements: impl IntoIterator<Item = impl AsRef<str>>,
155) -> Result<()> {
156    let stmts: Vec<(String, Vec<TursoValue>)> = statements
157        .into_iter()
158        .map(|sql| (sql.as_ref().to_string(), vec![]))
159        .collect();
160
161    retry_while_busy(|| conn.transactional_batch(&stmts)).await
162}

267impl LocalBuilderOptions {
268    /// Whether any local-engine option was configured. Used to reject
269    /// configurations that combine local-only options with a remote
270    /// engine (sync or serverless).
271    #[cfg(any(feature = "sync", feature = "serverless"))]
272    fn any_set(&self) -> bool {
273        self.encryption.is_some()
274            || self.attach
275            || self.custom_types
276            || self.generated_columns
277            || self.materialized_views
278            || self.vacuum
279            || self.multiprocess_wal
280            || self.without_rowid
281    }
282
283    fn apply(&self, mut b: Builder) -> Builder {
284        if let Some(opts) = &self.encryption {
285            // Upstream requires *both* the feature flag and the
286            // key/cipher to be set; collapse them into a single call so
287            // callers can't get into a half-configured state.
288            b = b
289                .experimental_encryption(true)
290                .with_encryption(opts.clone());
291        }
292        if self.attach {
293            b = b.experimental_attach(true);
294        }
295        if self.custom_types {
296            b = b.experimental_custom_types(true);
297        }
298        if self.generated_columns {
299            b = b.experimental_generated_columns(true);
300        }
301        if self.materialized_views {
302            b = b.experimental_materialized_views(true);
303        }
304        if self.vacuum {
305            b = b.experimental_vacuum(true);
306        }
307        if self.multiprocess_wal {
308            b = b.experimental_multiprocess_wal(true);
309        }
310        if self.without_rowid {
311            b = b.experimental_without_rowid(true);
312        }
313        b
314    }

380impl BuilderOptions {
381    fn apply_sync(&self, mut b: SyncBuilder) -> SyncBuilder {
382        if let Some(remote_url) = &self.sync_options.remote_url {
383            b = b.with_remote_url(remote_url)
384        }
385        if let Some(provider) = self.sync_options.auth_token.clone() {
386            b = b.with_auth_token_fn(move || provider());
387        }
388        if let Some(client_name) = &self.sync_options.client_name {
389            b = b.with_client_name(client_name)
390        }
391        if let Some(timeout) = self.sync_options.long_poll_timeout {
392            b = b.with_long_poll_timeout(timeout)
393        }
394        if let Some(opts) = &self.sync_options.partial_sync_config_experimental {
395            b = b.with_partial_sync_opts_experimental(opts.clone())
396        }
397        if self.sync_options.remote_encryption
398            && let (Some(base64_key), Some(cipher)) = (
399                &self.sync_options.remote_encryption_key,
400                &self.sync_options.remote_encryption_cipher,
401            )
402        {
403            b = b.with_remote_encryption(base64_key, *cipher)
404        } else if let Some(key) = &self.sync_options.remote_encryption_key {
405            b = b.with_remote_encryption_key(key);
406        }
407        if self.index_method {
408            b = b.experimental_index_method(true);
409        }
410        let bootstrap =
411            self.sync_options.remote_url.is_some() && self.sync_options.bootstrap_if_empty;
412        b = b.bootstrap_if_empty(bootstrap);
413        b
414    }

417/// A Turso [`Driver`] that opens connections to a file or in-memory database.
418///
419/// Experimental Turso features are exposed as `experimental_*` builder
420/// methods that mirror [`turso::Builder`].
421///
422/// # Examples
423///
424/// ```rust,ignore
425/// use toasty_driver_turso::Turso;
426///
427/// // File-backed database
428/// let driver = Turso::file("path/to/db");
429///
430/// // With experimental features
431/// use toasty_driver_turso::EncryptionOpts;
432///
433/// let driver = Turso::file("path/to/db")
434///     .experimental_encryption(EncryptionOpts {
435///         cipher: "aes256gcm".into(),
436///         hexkey: "<64-hex-character-key>".into(),
437///     })
438///     .experimental_attach(true);
439///
440/// // Concurrent writes
441/// let driver = Turso::file("path/to/db").concurrent_writes();
442///
443/// // Syncing with remote server
444/// let driver = Turso::file("path/to/db")
445///     .with_remote_url("<remote-url>")
446///     .with_auth_token("<auth-token>");
447///
448/// driver.push().await?;
449/// ```
450#[derive(Clone)]
451pub struct Turso {
452    path: TursoPath,
453    options: BuilderOptions,
454    concurrent_writes: bool,
455    /// Whether the driver opens the sync engine instead of the plain
456    /// local one. Set by [`Turso::with_sync`] and implied by the
457    /// mode-selecting sync options (`with_remote_url`, `with_client_name`,
458    /// ...) — but never by credentials.
459    #[cfg(feature = "sync")]
460    sync_mode: bool,
461    /// Shared database handle reused across every `connect()` call so that
462    /// all pool slots see the same underlying database. Without this, each
463    /// connection to `:memory:` would open a fresh empty database; even
464    /// file-backed handles open faster after the first builder run.
465    /// Cleared by [`Driver::reset_db`] so the next `connect()` starts fresh.
466    database: Arc<Mutex<Option<AnyDatabase>>>,
467}

470/// driver's configuration: the plain local engine, the sync engine that
471/// replicates to a remote database (`sync` feature), or a remote Turso

482impl Turso {
483    /// Create a new Turso driver from a connection URL.
484    ///
485    /// The URL scheme must be `turso` (e.g. `turso::memory:` or
486    /// `turso:/path/to/db`).
487    ///
488    /// With the `serverless` feature, an authority-form URL with a host
489    /// selects a remote Turso Cloud database reached over HTTP:
490    /// `turso://my-db.turso.io` (the form `turso db show --url` prints).
491    /// The `libsql`, `https` and `http` schemes are also accepted for
492    /// remote URLs, and an `authToken` query parameter is honored as if
493    /// passed to [`Self::with_auth_token`] (and stripped from the URL the
494    /// driver stores and reports). Scheme-relative and path forms
495    /// (`turso:todos.db`, `turso:/path/to/db`) always name local files;
496    /// without the feature, authority form resolves to a file path as
497    /// well.
498    pub fn new(url: impl Into<String>) -> Result<Self> {
499        let url_str = url.into();
500        let url = ConnectionUrl::parse(&url_str)?;
501
502        #[cfg(feature = "serverless")]
503        if let Some(driver) = Self::try_remote(&url) {
504            return Ok(driver);
505        }
506
507        if !url.has_scheme("turso") {
508            return Err(toasty_core::Error::invalid_connection_url(format!(
509                "connection URL does not have a `turso` scheme; url={url_str}"
510            )));
511        }
512
513        let path = url.file_path()?;
514        if path == Path::new(":memory:") {
515            return Ok(Self::with_path(TursoPath::InMemory));
516        }
517
518        Ok(Self::with_path(TursoPath::File(path)))
519    }
520
521    /// Builds a serverless driver when the connection URL names a remote
522    /// database: a remote scheme (`libsql`, `https`, `http`), or a
523    /// `turso` URL whose authority parses as a non-empty host. An
524    /// authority that is not host-shaped (for example `turso://:memory:`)
525    /// falls through to the file interpretation.
526    #[cfg(feature = "serverless")]
527    fn try_remote(url: &ConnectionUrl<'_>) -> Option<Self> {
528        let is_remote_scheme =
529            url.has_scheme("libsql") || url.has_scheme("https") || url.has_scheme("http");
530        let has_host = url.host().ok().flatten().is_some();
531        if !is_remote_scheme && !(url.has_scheme("turso") && has_host) {
532            return None;
533        }
534
535        let auth_token = url
536            .query_pairs()
537            .find(|(key, _)| key == "authToken")
538            .map(|(_, value)| value.into_owned());
539
540        let remote = url.as_str();
541        let remote = remote.split_once('?').map_or(remote, |(base, _)| base);
542
543        let mut driver = Self::with_path(TursoPath::Remote(remote.to_string()));
544        if let Some(token) = auth_token {
545            driver = driver.with_auth_token(token);
546        }
547        Some(driver)
548    }

620    /// Allow transactions to run concurrently instead of serializing on a
621    /// single writer.
622    ///
623    /// When enabled, each new connection switches to Turso's MVCC journal
624    /// (`PRAGMA journal_mode = 'mvcc'`) and a transaction started with
625    /// [`TransactionMode::Default`](toasty_core::driver::operation::TransactionMode::Default)
626    /// — i.e. an unspecified mode — issues `BEGIN CONCURRENT`. Conflicting
627    /// transactions can then fail to commit and must be retried by the
628    /// caller.
629    ///
630    /// Callers can opt out of MVCC concurrency on a per-transaction basis by
631    /// requesting a different
632    /// [`TransactionMode`](toasty_core::driver::operation::TransactionMode):
633    /// `Deferred` falls back to plain `BEGIN`, while `Immediate` and
634    /// `Exclusive` issue `BEGIN IMMEDIATE` / `BEGIN EXCLUSIVE` respectively.
635    ///
636    /// On a serverless (Turso Cloud) database this is a no-op: those
637    /// databases are MVCC-native and transactions already run under
638    /// `BEGIN CONCURRENT` by default.
639    pub fn concurrent_writes(mut self) -> Self {
640        self.concurrent_writes = true;
641        self
642    }

938    /// Returns the cached database handle, opening it on first use.
939    ///
940    /// All connections handed out by [`Driver::connect`] go through the
941    /// same `Database` so that `:memory:` is genuinely shared across pool
942    /// slots (each `Builder::new_local(":memory:").build()` would otherwise
943    /// produce a fresh, empty database).
944    async fn database(&self) -> Result<AnyDatabase> {
945        let mut slot = self.database.lock().await;
946        if let Some(db) = slot.as_ref() {
947            return Ok(db.clone());
948        }
949
950        let db = match &self.path {
951            #[cfg(feature = "serverless")]
952            TursoPath::Remote(url) => {
953                // Sync replicates a local file against a remote; a
954                // serverless URL leaves no local file to replicate. The
955                // combination is contradictory, not something to resolve
956                // by precedence.
957                if self.is_sync() {
958                    return Err(toasty_core::Error::unsupported_feature(
959                        "the driver is configured for sync but the connection URL names \
960                         a remote Turso Cloud database; sync replicates a local file — \
961                         use a file path with with_remote_url() instead",
962                    ));
963                }
964                if self.options.local_options.any_set() || self.options.index_method {
965                    return Err(toasty_core::Error::unsupported_feature(
966                        "local experimental options are not supported for a serverless \
967                         (remote HTTP) database",
968                    ));
969                }
970                let builder = self
971                    .options
972                    .apply_serverless(turso_serverless::Builder::new_remote(url.clone()));
973                AnyDatabase::Serverless(builder.build().await.map_err(classify_serverless_error)?)
974            }
975            _ if self.is_sync() => {
976                #[cfg(feature = "sync")]
977                {
978                    // The sync engine has no counterpart for the local
979                    // experimental toggles; reject the combination instead of
980                    // silently dropping options.
981                    if self.options.local_options.any_set() {
982                        return Err(toasty_core::Error::unsupported_feature(
983                            "local experimental options are not supported when the driver \
984                             is configured for sync",
985                        ));
986                    }
987                    let builder = self
988                        .options
989                        .apply_sync(SyncBuilder::new_remote(self.path_str()));
990                    AnyDatabase::Sync(builder.build().await.map_err(classify_turso_error)?)
991                }
992                #[cfg(not(feature = "sync"))]
993                unreachable!("is_sync() is false without the `sync` feature")
994            }
995            _ => {
996                // A credential (auth token, remote encryption key) is not
997                // a mode request, so it never selects an engine — but a
998                // credential on a plain local database is a configuration
999                // error, not something to drop silently.
1000                if self.has_remote_credentials() {
1001                    return Err(toasty_core::Error::unsupported_feature(
1002                        "a remote credential (auth token or encryption key) is \
1003                         configured but the driver opens a plain local database; a \
1004                         credential requires a sync remote (with_remote_url() or \
1005                         with_sync()) or a Turso Cloud connection URL",
1006                    ));
1007                }
1008                let builder = self
1009                    .options
1010                    .apply_local(Builder::new_local(self.path_str()));
1011                AnyDatabase::Local(builder.build().await.map_err(classify_turso_error)?)
1012            }
1013        };
1014
1015        *slot = Some(db.clone());
1016        Ok(db)
1017    }

1053            TursoPath::File(path) => Cow::Owned(format!("turso:{}", path.display())),

1063    async fn connect(&self, cx: &ConnectContext) -> Result<Box<dyn toasty_core::Connection>> {
1064        let conn = match self.database().await? {
1065            AnyDatabase::Local(db) => {
1066                AnyConn::Embedded(db.connect().map_err(classify_turso_error)?)
1067            }
1068            #[cfg(feature = "sync")]
1069            AnyDatabase::Sync(db) => {
1070                AnyConn::Embedded(db.connect().await.map_err(classify_turso_error)?)
1071            }
1072            #[cfg(feature = "serverless")]
1073            AnyDatabase::Serverless(db) => {
1074                AnyConn::Serverless(db.connect().map_err(classify_serverless_error)?)
1075            }
1076        };
1077
1078        if self.concurrent_writes && !conn.is_serverless() {
1079            // `PRAGMA journal_mode = ...` returns the new mode as a row; the
1080            // `execute` path errors with "unexpected row during execution"
1081            // on any pragma that emits one. Use `pragma_update` so the row
1082            // is consumed.
1083            //
1084            // Serverless databases skip the pragma: Turso Cloud databases
1085            // created with `--tursodb` are MVCC-native and the backend
1086            // rejects the statement ("SQL not allowed"); `BEGIN CONCURRENT`
1087            // alone provides the concurrent-writes semantics there.
1088            conn.pragma_update("journal_mode", "'mvcc'").await?;
1089        }
1090
1091        // Serverless databases are MVCC-native (Turso Cloud, created with
1092        // `--tursodb`), so transactions default to `BEGIN CONCURRENT` —
1093        // the same semantics `concurrent_writes()` opts into for the
1094        // embedded engine, which makes that flag a no-op here. Callers can
1095        // still choose classic locking per transaction with
1096        // `TransactionMode::Deferred`/`Immediate`/`Exclusive`.
1097        let default_begin_sql = if conn.is_serverless() || self.concurrent_writes {
1098            "BEGIN CONCURRENT"
1099        } else {
1100            "BEGIN"
1101        };
1102
1103        Ok(Box::new(Connection {
1104            conn,
1105            default_begin_sql,
1106            query_log: cx.query_log,
1107        }))
1108    }

1121    async fn reset_db(&self) -> Result<()> {
1122        // There is no file to delete on a remote database; drop every user
1123        // table instead (indexes and triggers go down with their table).
1124        #[cfg(feature = "serverless")]
1125        if let TursoPath::Remote(_) = &self.path {
1126            let AnyDatabase::Serverless(db) = self.database().await? else {
1127                unreachable!("a Remote path always opens a serverless database");
1128            };
1129            let conn = AnyConn::Serverless(db.connect().map_err(classify_serverless_error)?);
1130
1131            // `__turso_%` covers tursodb-internal system tables (e.g.
1132            // `__turso_internal_mvcc_meta`), which cannot be dropped.
1133            let mut rows = conn
1134                .query(
1135                    "SELECT name FROM sqlite_master \
1136                     WHERE type = 'table' \
1137                       AND name NOT LIKE 'sqlite_%' \
1138                       AND name NOT LIKE '\\_\\_turso\\_%' ESCAPE '\\'",
1139                    vec![],
1140                )
1141                .await?;
1142
1143            let mut drops = vec![];
1144            while let Some(row) = rows.next().await? {
1145                if let TursoValue::Text(name) = row.get_value(0)? {
1146                    drops.push(format!("DROP TABLE IF EXISTS \"{name}\""));
1147                }
1148            }
1149
1150            return exec_ddl(&conn, &drops).await;
1151        }
1152
1153        // Drop the cached Database so subsequent `connect()` calls open a
1154        // fresh one. For in-memory this is the only way to wipe state;
1155        // for file-backed databases the file is also removed below.
1156        self.database.lock().await.take();
1157
1158        if let TursoPath::File(path) = &self.path
1159            && path.exists()
1160        {
1161            std::fs::remove_file(path).map_err(toasty_core::Error::driver_operation_failed)?;
1162        }
1163
1164        Ok(())
1165    }

1187impl Connection {
1188    async fn exec_sql(
1189        &mut self,
1190        sql_str: &str,
1191        typed_params: Vec<TypedValue>,
1192        ret: SqlReturn,
1193    ) -> Result<ExecResponse> {
1194        let mut log = QueryLog::sql(
1195            &self.query_log,
1196            "turso",
1197            sql_str,
1198            typed_params.iter().map(|tv| &tv.value),
1199        );
1200        let result = self
1201            .exec_sql_inner(sql_str, typed_params, ret, &mut log)
1202            .await;
1203        log.finish(&result);
1204        result
1205    }
1206
1207    async fn exec_sql_inner(
1208        &mut self,
1209        sql_str: &str,
1210        typed_params: Vec<TypedValue>,
1211        ret: SqlReturn,
1212        log: &mut QueryLog<'_>,
1213    ) -> Result<ExecResponse> {
1214        let params: Vec<TursoValue> = typed_params
1215            .iter()
1216            .map(|tv| value::to_turso(&tv.value))
1217            .collect();
1218
1219        let mut stmt = self.conn.prepare_cached(sql_str).await?;
1220
1221        if matches!(ret, SqlReturn::Count) {
1222            let count = stmt.execute(params).await?;
1223
1224            return Ok(ExecResponse::count(count as _));
1225        }
1226
1227        let mut rows = stmt.query(params).await?;
1228
1229        let mut values = vec![];
1230
1231        while let Some(row) = rows.next().await? {
1232            let items = match &ret {
1233                SqlReturn::Count => unreachable!(),
1234                SqlReturn::Infer => {
1235                    let mut items = vec![];
1236                    for index in 0..row.column_count() {
1237                        items.push(value::from_turso_infer(row.get_value(index)?));
1238                    }
1239                    items
1240                }
1241                SqlReturn::Types(ret_tys) => {
1242                    let mut items = Vec::with_capacity(ret_tys.len());
1243                    for (index, ret_ty) in ret_tys.iter().enumerate() {
1244                        items.push(value::from_turso(row.get_value(index)?, ret_ty));
1245                    }
1246                    items
1247                }
1248            };
1249
1250            values.push(stmt::ValueRecord::from_vec(items).into());
1251        }
1252
1253        log.rows(values.len() as u64);
1254        Ok(ExecResponse::value_stream(stmt::ValueStream::from_vec(
1255            values,
1256        )))
1257    }
1258}

1265        let (sql, typed_params, ret_tys) = match op {
1266            Operation::Insert(op) => (sql::Statement::from(op.stmt), op.params, op.ret),
1267            Operation::QuerySql(op) => (sql::Statement::from(op.stmt), op.params, op.ret),
1268            Operation::RawSql(op) => {
1269                let ret = match op.ret {
1270                    RawSqlRet::None => SqlReturn::Count,
1271                    RawSqlRet::Infer => SqlReturn::Infer,
1272                    RawSqlRet::Types(types) => SqlReturn::Types(types),
1273                };
1274                return self.exec_sql(&op.sql, op.params, ret).await;
1275            }
1276            Operation::Transaction(op) => {
1277                if let Transaction::Start { isolation, .. } = &op
1278                    && !matches!(isolation, Some(IsolationLevel::Serializable) | None)
1279                {
1280                    return Err(toasty_core::Error::unsupported_feature(
1281                        "Turso only supports Serializable isolation",
1282                    ));
1283                }
1284                // `default_begin_sql` is the connection's "no opinion" BEGIN
1285                // — `BEGIN` for classic mode, `BEGIN CONCURRENT` for MVCC —
1286                // and the serializer maps the other `TransactionMode`s to
1287                // standard SQLite SQL.
1288                let sql_str =
1289                    sql::Serializer::sqlite_with_default_begin(&schema.db, self.default_begin_sql)
1290                        .serialize_transaction(&op);
1291                // On serverless there is no server-side busy timeout, so a
1292                // lock-taking BEGIN waits here instead of surfacing every
1293                // transient conflict. Embedded connections keep fail-fast
1294                // busy semantics.
1295                if self.conn.is_serverless()
1296                    && matches!(&op, Transaction::Start { .. })
1297                    && sql_str.starts_with("BEGIN")
1298                {
1299                    retry_while_busy(|| self.conn.execute(&sql_str, vec![])).await?;
1300                } else {
1301                    self.conn.execute(&sql_str, vec![]).await?;
1302                }
1303                return Ok(ExecResponse::count(0));
1304            }
1305            _ => todo!("op={:#?}", op),
1306        };
1307
1308        let ret = if sql.returning_len().is_some() {
1309            SqlReturn::Types(ret_tys.unwrap())
1310        } else {
1311            SqlReturn::Count
1312        };
1313
1314        let sql_str = sql::Serializer::sqlite(&schema.db).serialize(&sql);
1315        self.exec_sql(&sql_str, typed_params, ret).await
1316    }

1318    async fn push_schema(&mut self, schema: &Schema) -> Result<()> {
1319        let mut statements = vec![];
1320        for table in &schema.db.tables {
1321            tracing::debug!(table = %table.name, "creating table");
1322            statements.extend(create_table_stmts(&schema.db, table));
1323        }
1324
1325        exec_ddl(&self.conn, &statements).await
1326    }

1328    async fn applied_migrations(
1329        &mut self,
1330    ) -> Result<Vec<toasty_core::schema::db::AppliedMigration>> {
1331        exec_ddl(&self.conn, [CREATE_MIGRATIONS_TABLE]).await?;
1332
1333        let mut rows = self
1334            .conn
1335            .query(
1336                "SELECT id FROM __toasty_migrations ORDER BY applied_at",
1337                vec![],
1338            )
1339            .await?;
1340
1341        let mut migrations = vec![];
1342        while let Some(row) = rows.next().await? {
1343            if let TursoValue::Integer(id) = row.get_value(0)? {
1344                migrations.push(toasty_core::schema::db::AppliedMigration::new(id as u64));
1345            }
1346        }
1347
1348        Ok(migrations)
1349    }

1351    async fn apply_migration(
1352        &mut self,
1353        id: u64,
1354        name: &str,
1355        migration: &toasty_core::schema::db::Migration,
1356    ) -> Result<()> {
1357        tracing::info!(id = id, name = %name, "applying migration");
1358
1359        // The whole migration — DDL plus the parameterized bookkeeping
1360        // INSERT — is one atomic transactional batch: a single HTTP
1361        // request on the serverless transport.
1362        let mut stmts: Vec<(String, Vec<TursoValue>)> =
1363            vec![(CREATE_MIGRATIONS_TABLE.to_string(), vec![])];
1364        for statement in migration.statements() {
1365            stmts.push((statement.to_string(), vec![]));
1366        }
1367        stmts.push((
1368            "INSERT INTO __toasty_migrations (id, name, applied_at) VALUES (?1, ?2, datetime('now'))"
1369                .to_string(),
1370            vec![
1371                TursoValue::Integer(id as i64),
1372                TursoValue::Text(name.to_string()),
1373            ],
1374        ));
1375
1376        retry_while_busy(|| self.conn.transactional_batch(&stmts)).await
1377    }
