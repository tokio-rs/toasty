1//! Schema extraction support for the standalone `toasty` CLI.
2//!
3//! The CLI builds the user's package and runs the artifact with
4//! `TOASTY_DUMP_SCHEMA=<flavor>` set. The constructor below then fires before
5//! `main` (or during `dlopen` for a cdylib), collects every model registered
6//! by `#[derive(Model)]`, lowers the schema for that flavor, writes it to
7//! stdout as JSON, and exits. It only exists in `debug_assertions` builds.
8
9use toasty_core::schema::db;
10
11/// Environment variable that triggers the schema dump. Its value names the
12/// flavor to lower the schema for (e.g. `sqlite`, `postgresql`, `mysql`).
13pub const DUMP_SCHEMA_ENV: &str = "TOASTY_DUMP_SCHEMA";
14
15/// Environment variable carrying the table name prefix to lower the schema
16/// with. The constructor runs before any user code, so the prefix set on
17/// [`Builder::table_name_prefix`](crate::db::Builder::table_name_prefix) is
18/// invisible to it; the CLI passes `migration.table_name_prefix` from
19/// `Toasty.toml` through here instead.
20pub const DUMP_TABLE_NAME_PREFIX_ENV: &str = "TOASTY_DUMP_TABLE_NAME_PREFIX";
21
22/// Version of the JSON envelope written by the dump constructor.
23pub const SCHEMA_DUMP_VERSION: u32 = 1;
24
25/// The JSON envelope exchanged between the schema-dump constructor and the
26/// `toasty` CLI.
27#[derive(serde::Serialize, serde::Deserialize)]
28pub struct SchemaDump {
29    /// Envelope format version; bumped when the layout changes.
30    pub version: u32,
31
32    /// The database-level schema lowered for the requested flavor.
33    pub schema: db::Schema,
34}
35
36#[cfg(debug_assertions)]
37mod dump {
38    use super::*;
39    use crate::schema::DiscoverItem;
40    use toasty_core::driver::Capability;
41    use toasty_core::schema::app::ModelSet;
42
43    /// Flavor names accepted in [`DUMP_SCHEMA_ENV`]; the CLI's `Flavor::as_str`
44    /// must match.
45    const FLAVORS: &[(&str, &Capability)] = &[
46        ("sqlite", &Capability::SQLITE),
47        ("postgresql", &Capability::POSTGRESQL),
48        ("mysql", &Capability::MYSQL),
49        ("mariadb", &Capability::MARIADB),
50        ("turso", &Capability::TURSO),
51    ];
52
53    #[linktime::ctor(unsafe)]
54    fn __toasty_maybe_dump_schema() {
55        let Some(flavor) = std::env::var_os(DUMP_SCHEMA_ENV) else {
56            return;
57        };
58
59        // Never fall through to the user's `main`.
60        match dump_schema(&flavor.to_string_lossy()) {
61            Ok(json) => {
62                println!("{json}");
63                std::process::exit(0);
64            }
65            Err(message) => {
66                eprintln!("error: {message}");
67                std::process::exit(1);
68            }
69        }
70    }
71
72    fn dump_schema(flavor: &str) -> Result<String, String> {
73        let Some(&(_, capability)) = FLAVORS.iter().find(|(name, _)| *name == flavor) else {
74            let names: Vec<_> = FLAVORS.iter().map(|(name, _)| *name).collect();
75            return Err(format!(
76                "`{DUMP_SCHEMA_ENV}={flavor}` names an unknown flavor; expected one of: {}",
77                names.join(", ")
78            ));
79        };
80
81        // Sort by name so the schema is stable across rebuilds; `inventory`
82        // iteration order depends on link order.
83        let mut model_set = ModelSet::new();
84        for item in crate::schema::inventory::iter::<DiscoverItem> {
85            item.add_to(&mut model_set);
86        }
87
88        let mut models: Vec<_> = model_set.into_iter().collect();
89        models.sort_by(|a, b| a.name().parts.cmp(&b.name().parts));
90
91        let app_schema = toasty_core::schema::app::Schema::from_macro(models)
92            .map_err(|err| format!("failed to build schema from registered models: {err}"))?;
93
94        let mut builder = toasty_core::schema::Builder::new();
95
96        if let Some(prefix) = std::env::var_os(DUMP_TABLE_NAME_PREFIX_ENV) {
97            builder.table_name_prefix(&prefix.to_string_lossy());
98        }
99
100        let schema = builder
101            .build(app_schema, capability)
102            .map_err(|err| format!("failed to lower schema for flavor `{flavor}`: {err}"))?;
103
104        serde_json::to_string(&SchemaDump {
105            version: SCHEMA_DUMP_VERSION,
106            schema: schema.db,
107        })
108        .map_err(|err| format!("failed to serialize schema: {err}"))
109    }
110}
