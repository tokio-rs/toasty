34/// The expression-level type that a create/update setter accepts for a field
35/// of type `F`. This is just [`Field::ExprTarget`], but naming it keeps the
36/// long `<F as Field>::ExprTarget` projection out of generated setter
37/// signatures (and out of the compiler errors they produce).
38pub type FieldExprTarget<F> = <F as Field>::ExprTarget;
39
40/// Internal constructors used by generated model field accessors.
41pub trait ModelCodegen: Model {
42    /// Construct the field accessor for a singular relation to this model.
43    fn new_one_field<Origin>(path: Path<Origin, Self>) -> Self::OneField<Origin>;
44
45    /// Encode referenced fields so the engine can resolve a relation key.
46    fn to_relation_expr(&self, fields: &[&str]) -> core::stmt::Expr;
47}

49/// Infer the [`Scope`] type from a scope expression and return its fields
50/// path.
51///
52/// The `create!` macro uses this in the scoped form (`in expr { ... }`) to
53/// obtain the field struct for nested builders. Because the macro has no
54/// type information, it cannot call `S::new_path_root()` directly — this function
55/// lets Rust infer `S` from the scope argument.
56pub fn scope_fields<S: Scope>(_scope: &S) -> S::Path<S::Item> {
57    S::new_path_root()
58}
59
60/// Create a record inside a scoped create target.
61pub fn create_in_scope<S: Scope>(scope: S) -> S::Create {
62    S::create_in_scope(scope)
63}

65/// Convert a value into an untyped [`core::stmt::Expr`] via the typed
66/// [`IntoExpr<T>`] trait.
67///
68/// Generated code (`#[derive(Model)]`, `#[derive(Embed)]`) splices this in
69/// instead of inlining the `let expr: Expr<T> = value.into_expr(); expr.into()`
70/// pattern at every field site. The explicit `T` type parameter
71/// disambiguates which `IntoExpr` impl to use.
72pub fn into_untyped_expr<T, V: IntoExpr<T>>(value: V) -> core::stmt::Expr {
73    let expr: stmt::Expr<T> = value.into_expr();
74    expr.into()
75}

77/// A value usable as the parent of an embedded relation in a write.
78///
79/// The trait parameter `F` is the relation field's *declared* type
80/// (`M`, `Option<M>`, `Deferred<M>`, or `Deferred<Option<M>>`); the accepted value is the target
81/// model itself (by value or reference) or a value of the declared shape.
82/// Keying by `F` rather than by the model keeps inference working when the
83/// setter receives `Deferred::default()` — the setter signature pins `F`, so
84/// the deferred wrapper's inner type resolves uniquely.
85///
86/// `model_ref` resolves to the parent model when one is present: an unloaded
87/// `Deferred` (or a loaded `Deferred<Option<M>>` holding `None`) resolves to
88/// `None`, meaning the write has no parent value to take keys from and the
89/// explicitly set key fields stand.
90pub trait EmbeddedRelationValue<F> {
91    type Model;
92
93    fn model_ref(&self) -> Option<&Self::Model>;
94}

102                Some(self)

104        }

112        }

176/// Encode a loaded relation's key, leaving unloaded relations unset.
177pub fn embedded_relation_value_expr<F, V>(value: &V, fields: &[&str]) -> core::stmt::Expr
178where
179    V: EmbeddedRelationValue<F>,
180    V::Model: ModelCodegen,
181{
182    value
183        .model_ref()
184        .map_or_else(core::stmt::Expr::null, |model| {
185            model.to_relation_expr(fields)
186        })
187}

189/// Continue a `has_many` traversal from `query` along `path`.
190///
191/// If `query` was already scoped from a relation traversal, append
192/// `field_offset` to that traversal's path so the result remains a single
193/// chained association. Otherwise build a fresh `Association::many` rooted
194/// at `query`. Used by generated `.field()` chain methods on the per-model
195/// `Query<List<Source>>` for `has_many` relations.
196pub fn chain_or_build_many<Source, Target>(
197    mut query: stmt::Query<List<Source>>,
198    field_offset: usize,
199    path: stmt::Path<Source, List<Target>>,
200) -> stmt::Association<List<Target>>
201where
202    Source: Model,
203    Target: Model,
204{
205    match query.take_via_assoc() {
206        Some(untyped) => stmt::Association::<List<Source>>::from_untyped(untyped)
207            .chain_field::<Target>(field_offset),
208        None => stmt::Association::many(query, path),
209    }
210}

212/// Continue a `belongs_to` / `has_one` traversal from `query` along `path`.
213///
214/// Mirrors [`chain_or_build_many`] for singular relation steps: when no
215/// existing traversal is present the fresh association is built via
216/// [`stmt::Association::many_via_one`].
217pub fn chain_or_build_many_via_one<Source, Target>(
218    mut query: stmt::Query<List<Source>>,
219    field_offset: usize,
220    path: stmt::Path<Source, Target>,
221) -> stmt::Association<List<Target>>
222where
223    Source: Model,
224    Target: Model,
225{
226    match query.take_via_assoc() {
227        Some(untyped) => stmt::Association::<List<Source>>::from_untyped(untyped)
228            .chain_field::<Target>(field_offset),
229        None => stmt::Association::many_via_one(query, path),
230    }
