1//! Compile-time validation that a `belongs_to` key field has the same type as
2//! the target field it references.
3//!
4//! For each `(key, references)` pair of a `belongs_to` declared inside a
5//! `#[derive(Embed)]` enum variant, the generated code passes the target
6//! model's field accessor to [`target_field`], which recovers the referenced
7//! field's type through the accessor's [`FieldPath`] impl, then calls
8//! [`TargetField::check_key`] with the key field's type. The [`SameKeyType`]
9//! obligation fails with a readable message when the two disagree.
10//!
11//! Root models get the same guarantee from the generated relation accessor,
12//! whose filter compares the key field to the target path.
13
14use crate::stmt::Path;
15use std::marker::PhantomData;
16
17/// Recovers the Rust type a field accessor points at.
18///
19/// Implemented for [`Path<Origin, T>`] (primitive fields) and for the
20/// generated `{Type}Fields<Origin>` structs of embedded types, so
21/// `Target::fields().id()` resolves to the type of `Target::id` without
22/// spelling it out.
23pub trait FieldPath {
24    /// The referenced field's type.
25    type Ty;
26}
27
28impl<Origin, T> FieldPath for Path<Origin, T> {
29    type Ty = T;
30}
31
32/// Asserts that a `belongs_to` key field can hold the value of the field it
33/// references.
34///
35/// Implemented only reflexively (`impl<T> SameKeyType<T> for T`), so
36/// `A: SameKeyType<B>` holds iff `A` and `B` are the same type.
37#[diagnostic::on_unimplemented(
38    message = "`belongs_to` key field has type `{Self}`, but the field it references has type `{Target}`",
39    label = "this key field must have type `{Target}`",
40    note = "a foreign key stores the referenced field's value, so both fields must have the same type"
41)]
42pub trait SameKeyType<Target> {}
43
44impl<T> SameKeyType<T> for T {}
45
46/// The type of a `belongs_to` target field, recovered from its accessor by
47/// [`target_field`].
48pub struct TargetField<T>(PhantomData<T>);
49
50/// Resolves the type of the field `target` points at.
51pub fn target_field<P: FieldPath>(_target: P) -> TargetField<P::Ty> {
52    TargetField(PhantomData)
53}
54
55impl<T> TargetField<T> {
56    /// Type-checks one `(key, references)` pair. `Key` is the key field's
57    /// type with any `Option` wrapper removed. Never called at runtime.
58    ///
59    /// A method on an already-resolved `TargetField<T>` rather than a free
60    /// function generic over both types: with `T` still an inference variable,
61    /// rustc would infer `T = Key` from the sole `SameKeyType` impl and report
62    /// a bare type mismatch instead of the `on_unimplemented` message.
63    pub fn check_key<Key>(self)
64    where
65        Key: SameKeyType<T>,
66    {
67    }
68}
