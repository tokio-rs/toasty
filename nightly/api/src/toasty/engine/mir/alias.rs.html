5/// Passes another node's output through unchanged.
6///
7/// Fills a reserved node slot (see [`Store::reserve`](super::Store::reserve))
8/// once planning knows which node actually produces the slot's value. Other
9/// statements reference the reserved slot before its producer is planned; the
10/// alias makes that reference concrete without rewriting their edges. At
11/// execution the pass-through is a move between variable slots — rows are
12/// never touched. A future copy-propagation pass may eliminate these.
13#[derive(Debug)]
14pub(crate) struct Alias {
15    /// The node whose output is passed through.
16    pub(crate) input: mir::NodeId,
17
18    /// The output type (same as input).
19    pub(crate) ty: stmt::Type,
20}
21
22impl From<Alias> for mir::Node {
23    fn from(value: Alias) -> Self {
24        mir::Operation::Alias(value).into()
25    }
26}
