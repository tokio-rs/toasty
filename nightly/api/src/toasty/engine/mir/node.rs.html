2use toasty_core::stmt;

4use super::{NodeId, Operation};

6/// A single node in the MIR operation graph.
7///
8/// Each [`Node`] represents one operation to execute. It contains the
9/// operation itself, its dependencies on other nodes, and its execution
10/// guard.
11#[derive(Debug)]
12pub(crate) struct Node {
13    /// The operation this node performs.
14    pub(crate) op: Operation,
15
16    /// Nodes that must execute before this one.
17    ///
18    /// This is a superset of the node's data inputs; it may include additional
19    /// ordering dependencies (e.g., an `UPDATE` depending on a prior `INSERT`).
20    pub(crate) deps: IndexSet<NodeId>,
21
22    /// When set, this node only executes if the referenced node produced at
23    /// least one row, evaluated with a non-consuming peek. Assigned by the
24    /// guard-annotation pass in [`LogicalPlan::new`], which proves the
25    /// guarded node's output unobservable when the condition fails; only
26    /// pure (non-effectful) nodes may carry a guard.
27    ///
28    /// [`LogicalPlan::new`]: super::LogicalPlan::new
29    pub(crate) guard: Option<NodeId>,
30
31    /// Number of variable loads this node's output receives during execution,
32    /// plus one exit use when this is the completion node.
33    ///
34    /// [`LogicalPlan::new`](super::LogicalPlan::new) sets this after planning
35    /// finishes. The executor uses it for reference counting and frees the
36    /// output after the last use.
37    pub(crate) num_uses: usize,
38}
39
40impl Node {
41    pub(crate) fn ty(&self) -> &stmt::Type {
42        match &self.op {
43            Operation::Alias(m) => &m.ty,
44            Operation::Const(m) => &m.ty,
45            Operation::DeleteByKey(m) => &m.ty,
46            Operation::Eval(m) => &m.body.ret,
47            Operation::ExecStatement(m) => &m.ty,
48            Operation::Filter(m) => &m.ty,
49            Operation::FindPkByIndex(m) => &m.ty,
50            Operation::GetByKey(m) => &m.ty,
51            Operation::QueryPk(m) => &m.ty,
52            Operation::Repeat(m) => &m.ty,
53            Operation::Scan(m) => &m.ty,
54            Operation::UpdateByKey(m) => &m.ty,
55            Operation::Upsert(m) => &m.ty,
56            Operation::NestedMerge(m) => &m.ty,
57            Operation::ReadModifyWrite(m) => &m.ty,
58        }
59    }
60}
