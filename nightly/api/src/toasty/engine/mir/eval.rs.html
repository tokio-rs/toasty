4use crate::engine::{eval, mir};

6/// Evaluates `body` once over whole input values: with no `row_input`,
7/// `arg(i)` is `inputs[i]`'s complete output; with a `row_input`, `arg(0)` is
8/// `row_input`'s complete output and `arg(1 + i)` is `inputs[i]`'s.
9///
10/// A `row_input` marks the operation as per-row: the body is a `map` over
11/// `arg(0)` (built by [`Eval::map_over`]), so the output has one element per
12/// input row and pagination metadata forwards from `row_input`. Zero input
13/// rows means the map body never runs, so the other inputs are read only when
14/// `row_input` returned rows — the guard pass reads this from
15/// [`Operation::input_reads`](mir::Operation::input_reads).
16#[derive(Debug)]
17pub(crate) struct Eval {
18    /// When set, this operation returns one result for each row from this input.
19    pub(crate) row_input: Option<mir::NodeId>,
20
21    /// Nodes whose whole outputs the body reads.
22    pub(crate) inputs: IndexSet<mir::NodeId>,
23
24    /// The function to evaluate, over whole input values ordered
25    /// `[row_input?, inputs...]`. Its return type is the operation's output
26    /// type.
27    pub(crate) body: eval::Func,
28}
29
30impl Eval {
31    /// Evaluates `body` once over the whole outputs of `inputs`:
32    /// `arg(i)` = `inputs[i]`.
33    pub(crate) fn compute(inputs: IndexSet<mir::NodeId>, body: eval::Func) -> Self {
34        Eval {
35            row_input: None,
36            inputs,
37            body,
38        }
39    }
40
41    /// Evaluates the per-row `body` once per row of `row_input`: `arg(0)` =
42    /// current row, `arg(1 + i)` = `inputs[i]`.
43    ///
44    /// The executor evaluates one function over whole input values, so the
45    /// per-row structure is erased here into a single `map` expression over
46    /// input 0, with inputs ordered `[row_input, inputs...]`.
47    pub(crate) fn map_over(
48        store: &mir::Store,
49        row_input: mir::NodeId,
50        inputs: IndexSet<mir::NodeId>,
51        body: eval::Func,
52    ) -> Self {
53        debug_assert_eq!(body.args.len(), 1 + inputs.len());
54        debug_assert!(!inputs.contains(&row_input));
55
56        let mut arg_tys = vec![store[row_input].ty().clone()];
57        for input in &inputs {
58            arg_tys.push(store[input].ty().clone());
59        }
60
61        // Inside the map, the body's `arg(0)` (the row) resolves to the map's
62        // element scope unchanged, while references to other inputs must
63        // climb one extra scope — past the map — to reach the function
64        // inputs. A body arg references a body parameter when its nesting
65        // equals the number of scopes around it.
66        let ty = stmt::Type::list(body.ret.clone());
67        let mut map_body = body.into_expr();
68        visit_mut::walk_expr_scoped_mut(&mut map_body, 0, |expr, scope_depth| {
69            if let stmt::Expr::Arg(arg) = expr
70                && arg.nesting == scope_depth
71                && arg.position >= 1
72            {
73                arg.nesting += 1;
74            }
75            true
76        });
77        let expr = stmt::Expr::map(stmt::Expr::arg(0), map_body);
78
79        Eval {
80            row_input: Some(row_input),
81            inputs,
82            body: eval::Func::from_stmt_typed(expr, arg_tys, ty),
83        }
84    }
85}

87impl From<Eval> for mir::Node {
88    fn from(value: Eval) -> Self {
89        mir::Operation::Eval(value).into()
90    }
