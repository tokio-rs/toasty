1use crate::{
2    Result,

10impl Exec<'_> {
11    pub(super) async fn exec_eval(&mut self, action: &mir::Eval) -> Result<ExecResponse> {
12        match action.row_input {
13            Some(row_input) => self.exec_eval_map_over(action, row_input).await,
14            None => self.exec_eval_compute(action).await,
15        }
16    }
17
18    async fn exec_eval_compute(&mut self, action: &mir::Eval) -> Result<ExecResponse> {
19        // This form evaluates the body once with each complete input.
20        // For example, two input nodes produce this call:
21        //
22        //     body.eval([input_0, input_1])
23        let mut input = Vec::with_capacity(action.inputs.len());
24
25        for &node_id in &action.inputs {
26            let response = self.vars.load(node_id).await?;
27            // Only a row input can pass page cursors to an `Eval` result.
28            debug_assert!(response.is_unpaginated());
29            input.push(response.values.collect_as_value().await?);
30        }
31
32        let result = action.body.eval(&self.engine.schema, &input)?;
33
34        Ok(ExecResponse::from_rows(Rows::Value(result)))
35    }
36
37    async fn exec_eval_map_over(
38        &mut self,
39        action: &mir::Eval,
40        row_input: mir::NodeId,
41    ) -> Result<ExecResponse> {
42        // This form maps the body over the rows from `row_input`.
43        // For example, two rows produce this result:
44        //
45        //     [row_body.eval(row_input[0], inputs...),
46        //      row_body.eval(row_input[1], inputs...)]
47        //
48        // Load `row_input` first because it decides whether the other inputs
49        // are needed.
50        let ExecResponse {
51            values,
52            next_cursor,
53            prev_cursor,
54        } = self.vars.load(row_input).await?;
55        let input_rows = values.collect_as_value().await?;
56
57        if input_rows.is_list_empty() {
58            return Ok(self.exec_eval_map_over_empty_input(action, next_cursor, prev_cursor));
59        }
60
61        let mut input = Vec::with_capacity(1 + action.inputs.len());
62        input.push(input_rows);
63
64        for &node_id in &action.inputs {
65            let input_response = self.vars.load(node_id).await?;
66            // Only a row input can pass page cursors to an `Eval` result.
67            debug_assert!(input_response.is_unpaginated());
68            input.push(input_response.values.collect_as_value().await?);
69        }
70
71        let result = action.body.eval(&self.engine.schema, &input)?;
72
73        // The output has one item for each input row. Its cursors must match
74        // the input cursors so the caller can fetch the next or previous page.
75        Ok(ExecResponse {
76            values: Rows::Value(result),
77            next_cursor,
78            prev_cursor,
79        })
80    }
81
82    fn exec_eval_map_over_empty_input(
83        &mut self,
84        action: &mir::Eval,
85        next_cursor: Option<Box<Value>>,
86        prev_cursor: Option<Box<Value>>,
87    ) -> ExecResponse {
88        // There are no rows to evaluate. Do not load the other inputs.
89        // Loading a value consumes one recorded use. Since this path skips
90        // those loads, release the uses instead.
91        for &node_id in &action.inputs {
92            self.vars.release(node_id);
93        }
94
95        ExecResponse {
96            values: Rows::Value(Value::List(vec![])),
97            next_cursor,
98            prev_cursor,
99        }
100    }
101}
