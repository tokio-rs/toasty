11/// How to interpret a statement's output rows.
12///
13/// A conditional write (the SQL `#[version]` / OCC path compiled as a single
14/// CTE statement) prefixes its result with two probe columns: the number of
15/// rows matching the filter and, of those, the number satisfying the condition.
16/// The write applied only when the two agree; a mismatch is a condition
17/// failure, and zero matched rows means the record no longer exists.
18#[derive(Debug, Clone, Copy, PartialEq, Eq)]
19pub(crate) enum ConditionalOutput {
20    /// Not a conditional write. Output is passed through unchanged.
21    None,
22
23    /// Conditional write with no `RETURNING`. The two probe columns are the
24    /// only output; the action reports the matched-row count.
25    Count,
26
27    /// Conditional write with a `RETURNING`. The two probe columns are followed
28    /// by the changed rows' columns, which become the action's output.
29    Returning,
30}
31
32/// Configuration for pagination at the execution level.
33#[derive(Debug, Clone)]
34pub(crate) struct PaginationConfig {
35    /// Number of items per page
36    pub page_size: i64,
37    /// Whether the query starts after a cursor and can have a previous page.
38    pub has_previous_page: bool,
39    /// Function to extract cursor from a row (SQL only).
40    /// For NoSQL drivers, this is None (driver provides cursor).
41    pub extract_cursor: Option<eval::Func>,
42}

44/// Information about a MySQL UPDATE with RETURNING that needs special handling.
45///
46/// MySQL doesn't support `RETURNING` on `UPDATE`. The workaround is to strip
47/// the returning, run the UPDATE, then run a follow-up `SELECT` over the same
48/// table and filter to fetch the post-update column values. The two
49/// statements are not atomic relative to concurrent writers — see #881 for
50/// the broader design discussion.
51#[derive(Debug)]
52pub(super) struct MySQLUpdateReturning {
53    /// The `SELECT` statement that returns the post-update values. Carries
54    /// the same filter as the original `UPDATE` plus the projected
55    /// returning expression.
56    select_stmt: stmt::Statement,
57}

59impl Exec<'_> {
60    pub(super) async fn exec_statement(
61        &mut self,
62        action: &mir::ExecStatement,
63    ) -> Result<ExecResponse> {
64        // Databases always return rows as a vec of values; this specifies the
65        // type of each value. `None` means the statement returns only a count.
66        let output_ty = mir::row_field_types(&action.ty);
67
68        let mut stmt = action.stmt.clone();
69
70        // Collect input values and substitute into the statement
71        if !action.inputs.is_empty() {
72            let input_values = self.collect_input(action.inputs.iter().copied()).await?;
73            stmt.substitute(&input_values);
74
75            self.engine.simplify_stmt(&mut stmt);
76        }
77
78        debug_assert!(
79            stmt.returning()
80                .and_then(|returning| returning.as_project())
81                .map(|expr| expr.is_record())
82                .unwrap_or(true),
83            "stmt={stmt:#?}"
84        );
85
86        // MySQL does not support `RETURNING` on `UPDATE`. Strip the returning
87        // and capture an equivalent `SELECT` to run after the UPDATE.
88        let mysql_update_returning = self.process_stmt_update_with_returning_on_mysql(&mut stmt);
89
90        // Short circuit if we can statically determine there are no results
91        if let stmt::Statement::Query(query) = &stmt
92            && let stmt::ExprSet::Values(values) = &query.body
93            && values.is_empty()
94        {
95            assert_eq!(action.conditional, ConditionalOutput::None);
96
97            let rows = if output_ty.is_some() {
98                Rows::Stream(stmt::ValueStream::default())
99            } else {
100                Rows::Count(0)
101            };
102
103            return Ok(ExecResponse::from_rows(rows));
104        }
105
106        // Legalize the statement for the target backend and extract bind
107        // parameters (SQL drivers only; key-value drivers read values
108        // directly from the statement).
109        let params = self.engine.prepare_for_driver(&mut stmt);
110
111        let ret = match action.conditional {
112            // A conditional write prefixes its result with two `I64` probe
113            // counts; the `Returning` variant follows them with the changed
114            // rows' columns.
115            ConditionalOutput::Count => Some(vec![stmt::Type::I64, stmt::Type::I64]),
116            ConditionalOutput::Returning => {
117                let mut tys = vec![stmt::Type::I64, stmt::Type::I64];
118                tys.extend(
119                    output_ty
120                        .clone()
121                        .expect("conditional write with RETURNING has output columns"),
122                );
123                Some(tys)
124            }
125            ConditionalOutput::None if mysql_update_returning.is_some() => {
126                // The UPDATE has had its RETURNING stripped; the driver runs
127                // a plain UPDATE that returns no rows. The follow-up SELECT
128                // below produces the returning values.
129                None
130            }
131            ConditionalOutput::None => output_ty.clone(),
132        };
133
134        let op: toasty_core::driver::Operation = if stmt.is_insert() {
135            operation::Insert { stmt, params, ret }.into()
136        } else {
137            operation::QuerySql { stmt, params, ret }.into()
138        };
139
140        let mut res = self.connection.exec(&self.engine.schema, op).await?;
141
142        match action.conditional {
143            ConditionalOutput::None => {
144                if let Some(mysql_update) = mysql_update_returning {
145                    res = self
146                        .run_mysql_update_returning_select(mysql_update, output_ty.clone())
147                        .await?;
148                }
149            }
150            ConditionalOutput::Count | ConditionalOutput::Returning => {
151                let rows = collect_conditional_probe(res.values).await?;
152                let (matched, conditioned) = conditional_probe_counts(&rows[0])?;
153
154                // A conditional write targets a row the caller holds an
155                // instance of: zero matched rows means it has since been
156                // deleted.
157                if matched == 0 {
158                    return Err(toasty_core::Error::record_not_found(
159                        "conditional write matched no rows",
160                    ));
161                }
162                if matched != conditioned {
163                    return Err(toasty_core::Error::condition_failed(
164                        "write condition did not match",
165                    ));
166                }
167
168                res.values = match action.conditional {
169                    ConditionalOutput::Count => Rows::Count(matched as u64),
170                    _ => {
171                        // The probe locked the matched rows, so the write
172                        // applied to exactly those rows and every result row is
173                        // a real changed row — strip the two leading probe
174                        // columns.
175                        let changed = rows
176                            .into_iter()
177                            .map(|row| {
178                                let stmt::Value::Record(record) = row else {
179                                    return Err(toasty_core::Error::invalid_result(
180                                        "conditional write expected Record",
181                                    ));
182                                };
183                                Ok(stmt::Value::record_from_vec(
184                                    record.fields.into_iter().skip(2).collect(),
185                                ))
186                            })
187                            .collect::<Result<Vec<_>>>()?;
188
189                        Rows::value_stream(changed)
190                    }
191                };
192            }
193        }
194
195        // Apply pagination if configured
196        if let Some(pagination) = &action.pagination {
197            assert!(res.is_unpaginated());
198            res.values.buffer().await?;
199            self.apply_sql_pagination(&mut res, pagination)?;
200        }
201
202        Ok(res)
203    }
204
205    /// Apply SQL pagination by extracting cursor from last row.
206    /// If we got a full page (page_size rows), extract cursor for potential next page.
207    /// The client will naturally discover there's no more data when the next request returns empty.
208    ///
209    /// The response values must already be buffered (via `Rows::buffer()`).
210    fn apply_sql_pagination(
211        &mut self,
212        res: &mut ExecResponse,
213        pagination: &PaginationConfig,
214    ) -> Result<()> {
215        let Some(extract_cursor) = &pagination.extract_cursor else {
216            return Ok(());
217        };
218
219        let Rows::Value(stmt::Value::List(ref row_vec)) = res.values else {
220            return Ok(());
221        };
222
223        let page_size = pagination.page_size as usize;
224
225        // Extract cursors for potential next/prev pages
226        res.next_cursor = if row_vec.len() == page_size {
227            let cursor_row = &row_vec[page_size - 1];
228            Some(Box::new(extract_cursor.eval(
229                &self.engine.schema,
230                std::slice::from_ref(cursor_row),
231            )?))
232        } else {
233            // Got fewer than page_size rows, no more data
234            None
235        };
236
237        // Extract a previous cursor only when this query started after another page.
238        res.prev_cursor = if pagination.has_previous_page
239            && !row_vec.is_empty()
240            && self.engine.capability().backward_pagination
241        {
242            let cursor_row = &row_vec[0];
243            Some(Box::new(extract_cursor.eval(
244                &self.engine.schema,
245                std::slice::from_ref(cursor_row),
246            )?))
247        } else {
248            None
249        };
250
251        Ok(())
252    }
253}

255impl Exec<'_> {
256    /// Detects an UPDATE with a non-empty `RETURNING` on a MySQL backend
257    /// and rewrites the statement for the workaround path:
258    ///
259    /// - The returning clause is stripped from the UPDATE so the SQL
260    ///   serializer doesn't reject it.
261    /// - An equivalent `SELECT` over the same table + filter is captured,
262    ///   carrying the original returning expression as its projection.
263    ///
264    /// Returns `None` when the backend supports `RETURNING` natively (PG,
265    /// SQLite) or when the statement is not an UPDATE with a returning
266    /// project. The two-statement path is not atomic relative to concurrent
267    /// writers — see #881.
268    pub(super) fn process_stmt_update_with_returning_on_mysql(
269        &self,
270        stmt: &mut stmt::Statement,
271    ) -> Option<MySQLUpdateReturning> {
272        if self.engine.capability().returning_from_mutation || !self.engine.capability().sql() {
273            return None;
274        }
275
276        let stmt::Statement::Update(update) = stmt else {
277            return None;
278        };
279
280        let table_id = match &update.target {
281            stmt::UpdateTarget::Table(table_id) => *table_id,
282            _ => return None,
283        };
284
285        let returning = update.returning.take()?;
286
287        let select = stmt::Select {
288            returning,
289            source: stmt::Source::table(table_id),
290            filter: update.filter.clone(),
291            distinct: false,
292        };
293        let select_stmt =
294            stmt::Statement::Query(stmt::Query::new(stmt::ExprSet::Select(Box::new(select))));
295
296        Some(MySQLUpdateReturning { select_stmt })
297    }
298
299    /// Runs the follow-up `SELECT` for a MySQL UPDATE with stripped
300    /// `RETURNING`. The driver receives a plain query whose result rows
301    /// take the place of the original RETURNING output.
302    pub(super) async fn run_mysql_update_returning_select(
303        &mut self,
304        mysql_update: MySQLUpdateReturning,
305        ret_ty: Option<Vec<stmt::Type>>,
306    ) -> Result<toasty_core::driver::ExecResponse> {
307        let mut select_stmt = mysql_update.select_stmt;
308        let select_params = self.engine.prepare_for_driver(&mut select_stmt);
309
310        let op = operation::QuerySql {
311            stmt: select_stmt,
312            params: select_params,
313            ret: ret_ty,
314        };
315
316        self.connection.exec(&self.engine.schema, op.into()).await
317    }
318}

320/// Collects a conditional write's result rows. The probe (a `COUNT` aggregate)
321/// always yields at least one row, so an empty result is a driver bug.
322async fn collect_conditional_probe(rows: Rows) -> Result<Vec<stmt::Value>> {
323    let Rows::Stream(rows) = rows else {
324        return Err(toasty_core::Error::invalid_result(format!(
325            "conditional write expected Stream, got {rows:?}"
326        )));
327    };
328
329    let rows = rows.collect().await?;
330    if rows.is_empty() {
331        return Err(toasty_core::Error::invalid_result(
332            "conditional write probe returned no rows",
333        ));
334    }
335
336    Ok(rows)
337}

339/// Reads the two leading probe counts (`matched`, `conditioned`) from a
340/// conditional write's result row.
341fn conditional_probe_counts(row: &stmt::Value) -> Result<(i64, i64)> {
342    let stmt::Value::Record(record) = row else {
343        return Err(toasty_core::Error::invalid_result(format!(
344            "conditional write expected Record, got {row:?}"
345        )));
346    };
347
348    match (record.fields.first(), record.fields.get(1)) {
349        (Some(stmt::Value::I64(matched)), Some(stmt::Value::I64(conditioned))) => {
350            Ok((*matched, *conditioned))
351        }
352        _ => Err(toasty_core::Error::invalid_result(format!(
353            "conditional write probe columns are not I64; row={row:?}"
354        ))),
355    }
