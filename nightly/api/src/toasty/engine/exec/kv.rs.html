9impl Exec<'_> {
10    /// Bind a key-value row filter to its operation's input values.
11    ///
12    /// Returns `false` when the bound filter matches no row, so the caller
13    /// can skip the driver call. A filter that binds to `true` is removed.
14    pub(super) fn bind_row_filter(
15        &self,
16        filter: &mut Option<stmt::Expr>,
17        input: &[stmt::Value],
18        table: TableId,
19    ) -> bool {
20        let Some(expr) = filter else {
21            return true;
22        };
23
24        // An absent candidate key matches no row. Dropping it keeps the
25        // key-value comparison two-valued, so a negated membership still
26        // admits every other row. Only lists bound from the input carry
27        // candidate keys; constant lists in the filter are left as written.
28        stmt::visit_mut::for_each_expr_mut(expr, |expr| {
29            if let stmt::Expr::InList(in_list) = expr
30                && is_input_arg(&in_list.list)
31            {
32                in_list.list.substitute(input);
33
34                if let stmt::Expr::Value(stmt::Value::List(values)) = &mut *in_list.list {
35                    values.retain(|value| !is_absent_key(value));
36                }
37            }
38        });
39
40        expr.substitute(input);
41
42        // Bound inputs can reduce the filter to a constant, e.g. an empty
43        // candidate list turns `x IN ()` into `false`.
44        let db_table = self.engine.schema.db.table(table);
45        let cx = self.engine.expr_cx_for(db_table);
46        simplify::simplify_expr(cx, self.engine.capability, expr);
47
48        if expr.is_unsatisfiable() {
49            return false;
50        }
51
52        if expr.is_true() {
53            *filter = None;
54        }
55
56        true
57    }

59    /// Split a composite filter into individual key predicates.
60    ///
61    /// Recognizes these forms and decomposes them:
62    /// - `ANY(MAP(Value::List([v1, v2, ...]), pred))` — substitutes each vi
63    ///   into pred
64    /// - `InList(expr, Value::List([v1, v2, ...]))` — produces `expr == vi`
65    ///   for each value
66    ///
67    /// For any other form (including a single equality), simplifies and returns
68    /// it as a single-element vec (or empty if unsatisfiable).
69    ///
70    /// Each returned predicate has been simplified and is guaranteed
71    /// satisfiable.
72    pub(super) fn split_filter(&self, filter: stmt::Expr, table: TableId) -> Vec<stmt::Expr> {
73        let db_table = self.engine.schema.db.table(table);
74        let cx = self.engine.expr_cx_for(db_table);
75
76        let capability = self.engine.capability;
77        match filter {
78            stmt::Expr::Any(any) => Self::split_filter_any_map(*any.expr, cx, capability),
79            stmt::Expr::InList(in_list) => {
80                Self::split_filter_in_list(*in_list.expr, *in_list.list, cx, capability)
81            }
82            mut other => {
83                simplify::simplify_expr(cx, self.engine.capability, &mut other);
84                if other.is_unsatisfiable() {
85                    vec![]
86                } else {
87                    vec![other]
88                }
89            }
90        }
91    }
92
93    /// `ANY(MAP(Value::List([v1, v2, ...]), pred))` — substitutes each value
94    /// into the predicate template.
95    ///
96    /// Duplicate values are collapsed: each kv-layer fan-out becomes one
97    /// driver call per partition key, and a downstream `HashIndex` build
98    /// over the merged rows requires unique keys.
99    fn split_filter_any_map(
100        map_expr: stmt::Expr,
101        cx: ExprContext<'_>,
102        capability: &Capability,
103    ) -> Vec<stmt::Expr> {
104        let stmt::Expr::Map(map) = map_expr else {
105            unreachable!()
106        };
107        let stmt::Expr::Value(stmt::Value::List(items)) = *map.base else {
108            unreachable!()
109        };
110
111        let mut seen = ValueSet::with_capacity(items.len());
112        items
113            .into_iter()
114            .filter(|item| !item.is_null())
115            .filter(|item| seen.insert(item.clone()))
116            .filter_map(|item| {
117                let mut pred = *map.map.clone();
118                // Unpack Record fields so arg(i) binds to field i.
119                match item {
120                    stmt::Value::Record(r) => pred.substitute(&r.fields[..]),
121                    item => pred.substitute([item]),
122                }
123                simplify::simplify_expr(cx, capability, &mut pred);
124                (!pred.is_unsatisfiable()).then_some(pred)
125            })
126            .collect()
127    }
128
129    /// `InList(expr, Value::List([v1, v2, ...]))` — produces `expr == vi` for
130    /// each value.
131    ///
132    /// Duplicate values are collapsed; see `split_filter_any_map`.
133    fn split_filter_in_list(
134        expr: stmt::Expr,
135        list: stmt::Expr,
136        cx: ExprContext<'_>,
137        capability: &Capability,
138    ) -> Vec<stmt::Expr> {
139        let stmt::Expr::Value(stmt::Value::List(values)) = list else {
140            unreachable!()
141        };
142
143        let mut seen = ValueSet::with_capacity(values.len());
144        values
145            .into_iter()
146            .filter(|v| !is_absent_key(v))
147            .filter(|v| seen.insert(v.clone()))
148            .filter_map(|v| {
149                let mut pred = stmt::Expr::binary_op(expr.clone(), stmt::BinaryOp::Eq, v);
150                simplify::simplify_expr(cx, capability, &mut pred);
151                (!pred.is_unsatisfiable()).then_some(pred)
152            })
153            .collect()
154    }
155}

157/// Whether `expr` is a reference to the operation's input, possibly projected.
158fn is_input_arg(expr: &stmt::Expr) -> bool {
159    match expr {
160        stmt::Expr::Arg(_) => true,
161        stmt::Expr::Project(project) => matches!(&*project.base, stmt::Expr::Arg(_)),
162        _ => false,
163    }
164}

166/// Whether a key value is absent: null itself, or a composite key with any
167/// null component. An absent key identifies no row, and key-value drivers
168/// reject null key attributes, so key lookups skip it.
169pub(super) fn is_absent_key(value: &stmt::Value) -> bool {
170    match value {
171        stmt::Value::Null => true,
172        stmt::Value::Record(record) => record.fields.iter().any(is_absent_key),
173        _ => false,
174    }
