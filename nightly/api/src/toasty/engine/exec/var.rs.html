12/// Runtime storage for node outputs: one slot per MIR node, keyed by the
13/// node's [`NodeId`].
14#[derive(Debug)]
15pub(crate) struct VarStore {
16    slots: IndexVec<NodeId, Option<Entry>>,
17    /// Resolves `Type::Model` (`#[document]`) layouts for the value type-checks.
18    schema: Arc<Schema>,
19}
20
21#[derive(Debug)]
22struct Entry {
23    value: EntryValue,
24    count: usize,
25}
26
27#[derive(Debug)]
28enum EntryValue {
29    Response(ExecResponse),
30    Skipped,
31}
32
33impl VarStore {
34    pub(crate) fn new(node_count: usize, schema: Arc<Schema>) -> Self {
35        Self {
36            slots: IndexVec::from_vec((0..node_count).map(|_| None).collect()),
37            schema,
38        }
39    }
40
41    pub(crate) async fn load(&mut self, node: NodeId) -> crate::Result<ExecResponse> {
42        let Some(entry) = &mut self.slots[node] else {
43            panic!("no stream at slot {node:?}; store={self:#?}")
44        };
45
46        if matches!(&entry.value, EntryValue::Skipped) {
47            panic!("load of skipped slot {node:?}")
48        }
49
50        if entry.count == 1 {
51            let entry = self.slots[node].take().unwrap();
52            let EntryValue::Response(response) = entry.value else {
53                unreachable!()
54            };
55            return Ok(response);
56        }
57
58        entry.count -= 1;
59        let EntryValue::Response(response) = &mut entry.value else {
60            unreachable!()
61        };
62        Ok(ExecResponse {
63            values: response.values.dup().await?,
64            next_cursor: response.next_cursor.clone(),
65            prev_cursor: response.prev_cursor.clone(),
66        })
67    }
68
69    /// Decrements a slot's use count without observing its value, dropping
70    /// the entry at zero. Called on paths that decline a load the use
71    /// counting expects (a skipped `If` arm).
72    #[track_caller]
73    pub(crate) fn release(&mut self, node: NodeId) {
74        let Some(entry) = self.slots[node].as_mut() else {
75            panic!("release of unset slot {node:?}; store={self:#?}")
76        };
77
78        if entry.count == 1 {
79            self.slots[node] = None;
80        } else {
81            entry.count -= 1;
82        }
83    }
84
85    /// Returns whether the slot holds at least one row, without consuming a
86    /// use. A skipped slot is empty, allowing a guard to depend on a node that
87    /// an earlier guard skipped. A stream-backed slot is buffered in place so
88    /// the peek does not disturb later loads.
89    pub(crate) async fn peek_non_empty(&mut self, node: NodeId) -> crate::Result<bool> {
90        let Some(entry) = self.slots[node].as_mut() else {
91            panic!("no stream at slot {node:?}; store={self:#?}")
92        };
93
94        let EntryValue::Response(response) = &mut entry.value else {
95            return Ok(false);
96        };
97
98        response.values.buffer().await?;
99
100        Ok(match &response.values {
101            Rows::Count(count) => *count > 0,
102            Rows::Value(stmt::Value::List(items)) => !items.is_empty(),
103            Rows::Value(stmt::Value::Null) => false,
104            Rows::Value(_) => true,
105            Rows::Stream(_) => unreachable!("stream was buffered above"),
106        })
107    }
108
109    /// Debug-asserts every slot has been drained. With exact use counts this
110    /// holds after the final load of the plan's returning variable, on the
111    /// success path only — a mid-plan failure legitimately leaves loads
112    /// unperformed. Undercounting already panics loudly on a load of a freed
113    /// slot; this converts the silent overcounting direction into a loud one.
114    pub(crate) fn assert_empty(&self) {
115        debug_assert!(
116            self.slots.iter().all(Option::is_none),
117            "variable slots not drained at plan completion; store={self:#?}"
118        );
119    }
120
121    /// Marks an escaping output from a skipped `If` arm. The slot participates
122    /// in use counting but has no value and cannot be loaded.
123    pub(crate) fn store_skipped(&mut self, node: NodeId, count: usize) {
124        if count == 0 {
125            return;
126        }
127
128        self.slots[node] = Some(Entry {
129            value: EntryValue::Skipped,
130            count,
131        });
132    }
133
134    #[track_caller]
135    pub(crate) fn store(
136        &mut self,
137        node: NodeId,
138        ty: &stmt::Type,
139        count: usize,
140        response: ExecResponse,
141    ) {
142        // A zero-use output is never observed; don't occupy a slot.
143        if count == 0 {
144            return;
145        }
146
147        let values = match response.values {
148            Rows::Count(_) => {
149                assert!(ty.is_unit());
150                response.values
151            }
152            Rows::Value(value) => {
153                assert!(
154                    value.is_a(&self.schema.app, ty),
155                    "type mismatch: {value:?} is not a {ty:?}",
156                );
157                Rows::Value(value)
158            }
159            Rows::Stream(value_stream) => {
160                let stmt::Type::List(item_tys) = ty else {
161                    todo!("ty={ty:#?}")
162                };
163                let item_ty = (**item_tys).clone();
164
165                Rows::Stream(value_stream.typed(self.schema.clone(), item_ty))
166            }
167        };
168        let response = ExecResponse {
169            values,
170            next_cursor: response.next_cursor,
171            prev_cursor: response.prev_cursor,
172        };
173
174        self.slots[node] = Some(Entry {
175            value: EntryValue::Response(response),
176            count,
177        });
178    }
179}
