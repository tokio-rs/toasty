6/// A conditionally executed block of nodes — the exec program's first
7/// control-flow construct.
8///
9/// The `then` arm holds pure nodes whose outputs are only consumed when the
10/// condition holds; mutations never appear in it. Every node in the arm
11/// carries the same guard, which is the block's condition. Skipping the arm
12/// is pure bookkeeping, declared as data: `skipped_inputs` and
13/// `skipped_outputs` keep variable slots consistent so consumers outside the
14/// block never see an unset slot and use counts stay exact on both paths.
15#[derive(Debug)]
16pub(crate) struct If {
17    /// Nodes run when the condition holds. Non-empty; the condition is the
18    /// nodes' shared guard.
19    pub(crate) then: Vec<mir::NodeId>,
20
21    /// Nodes produced outside the block whose outputs the `then` arm would
22    /// have loaded — one entry per declined load. Released when the arm is
23    /// skipped.
24    pub(crate) skipped_inputs: Vec<mir::NodeId>,
25
26    /// The `then` arm's escaping outputs, each with its external use count.
27    /// When the arm is skipped, each receives a non-loadable slot used only
28    /// for use counting.
29    pub(crate) skipped_outputs: Vec<(mir::NodeId, usize)>,
30}
31
32impl If {
33    fn guard(&self, logical_plan: &LogicalPlan) -> mir::NodeId {
34        // Every node in the branch has the same guard, so the first node's
35        // guard is the branch's guard.
36        logical_plan[self.then[0]]
37            .guard
38            .expect("`If` block nodes carry the block's guard")
39    }
40}
41
42impl Exec<'_> {
43    pub(super) async fn action_if(
44        &mut self,
45        logical_plan: &LogicalPlan,
46        action: &If,
47    ) -> Result<()> {
48        let cond = action.guard(logical_plan);
49
50        // A non-consuming peek: buffers the condition variable's stream in
51        // place, leaving its use count untouched.
52        let pass = self.vars.peek_non_empty(cond).await?;
53
54        if pass {
55            for &node_id in &action.then {
56                self.exec_node(logical_plan, node_id).await?;
57            }
58        } else {
59            for &node_id in &action.skipped_inputs {
60                self.vars.release(node_id);
61            }
62            for &(node_id, num_uses) in &action.skipped_outputs {
63                self.vars.store_skipped(node_id, num_uses);
64            }
65        }
66
67        Ok(())
68    }
