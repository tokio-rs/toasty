1use crate::{
2    Result,
3    engine::{exec::Exec, mir},
4};
5use toasty_core::{
6    driver::{ExecResponse, Rows},

10impl Exec<'_> {
11    pub(super) async fn exec_repeat(&mut self, action: &mir::Repeat) -> Result<ExecResponse> {
12        let ExecResponse {
13            values,
14            next_cursor,
15            prev_cursor,
16        } = self.vars.load(action.input).await?;
17
18        // Only the input's cardinality is observed; a count-only driver
19        // response works the same as one that returned rows.
20        let count = match values {
21            Rows::Count(count) => count as usize,
22            Rows::Value(stmt::Value::List(items)) => items.len(),
23            Rows::Value(value) => todo!("value={value:#?}"),
24            Rows::Stream(mut value_stream) => {
25                let mut count = 0;
26                while let Some(res) = value_stream.next().await {
27                    res?;
28                    count += 1;
29                }
30                count
31            }
32        };
33
34        let rows = vec![action.value.clone(); count];
35
36        // The output has one item for each input row. Its cursors must match
37        // the input cursors so the caller can fetch the next or previous page.
38        Ok(ExecResponse {
39            values: Rows::value_stream(rows),
40            next_cursor,
41            prev_cursor,
42        })
43    }
44}
