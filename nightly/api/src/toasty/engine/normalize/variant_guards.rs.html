1//! Attaches the variant checks a predicate over an enum variant requires.
2//!
3//! A path into an enum variant (`contact().email().address()`) converts to an
4//! [`ExprVariant`] selection: the variant's payload, without a check that the
5//! value holds that variant. A predicate over such an operand only means
6//! something for rows of that variant, so normalization conjoins an
7//! `is_variant` check for every selection a predicate's operands reach — on
8//! either operand, and for every enclosing variant of a nested selection,
9//! outermost first.
10//!
11//! Guards attach at each predicate boundary: the comparison, null check,
12//! membership test, or other boolean operator whose operands hold the
13//! selection. Boolean connectives and subqueries are boundaries of their own,
14//! so `a.eq(b).not()` becomes `NOT (is_variant AND a = b)` and negates the
15//! guarded comparison as a whole, while `a.is_some()` — the `IS NOT NULL`
16//! predicate — becomes `is_variant AND NOT (a IS NULL)`. A selection used as
17//! a value only, such as an ordering expression, stays unguarded.
18//!
19//! Guards a conjunction already states are not repeated below it, which keeps
20//! the rewrite idempotent and lets generated `matches()` filters keep their
21//! explicit `is_variant AND body` shape. Conjunctions produced by the rewrite
22//! flatten into the enclosing conjunction so that a guard and the comparison
23//! it scopes stay siblings, which relation lifting relies on to fold them
24//! into one subquery.
25

32impl Normalize<'_> {
33    /// Normalizes the operands of a conjunction and flattens conjunctions
34    /// they produce into it.
35    ///
36    /// The `is_variant` operands are in force for every operand, including
37    /// those the flattening adds ahead of the operands that follow.
38    pub(super) fn normalize_conjunction(&mut self, and: &mut ExprAnd) {
39        let depth = self.guards.len();
40        self.guards
41            .extend(and.operands.iter().filter(|e| is_guard(e)).cloned());
42
43        for mut operand in std::mem::take(&mut and.operands) {
44            self.visit_expr_mut(&mut operand);
45
46            match operand {
47                Expr::And(inner) => {
48                    for operand in inner.operands {
49                        if is_guard(&operand) && !self.guards.contains(&operand) {
50                            self.guards.push(operand.clone());
51                        }
52                        and.operands.push(operand);
53                    }
54                }
55                operand => and.operands.push(operand),
56            }
57        }
58
59        self.guards.truncate(depth);
60    }
61
62    /// Conjoins the variant checks a predicate's operands require with the
63    /// predicate, and expands a negated predicate into its `NOT` form.
64    ///
65    /// The operands have already been normalized: a nested predicate carries
66    /// its own guards, and connectives and subqueries are not searched.
67    pub(super) fn normalize_predicate_guards(&mut self, expr: &mut Expr) {
68        let mut guards = vec![];
69        let mut collect = CollectGuards {
70            guards: &mut guards,
71            existing_guards: &self.guards,
72        };
73
74        match expr {
75            Expr::BinaryOp(e) if e.op.is_arithmetic() => return,
76            Expr::AllOp(_)
77            | Expr::AnyOp(_)
78            | Expr::Between(_)
79            | Expr::BinaryOp(_)
80            | Expr::InList(_)
81            | Expr::InSubquery(_)
82            | Expr::Intersects(_)
83            | Expr::IsNull(_)
84            | Expr::IsSuperset(_)
85            | Expr::IsVariant(_)
86            | Expr::Like(_)
87            | Expr::StartsWith(_) => collect.visit_expr(expr),
88            Expr::Not(e) => collect.visit_expr(&e.expr),
89            _ => return,
90        }
91
92        let mut predicate = expr.take();
93        let negated = match &mut predicate {
94            Expr::IsNull(e) => std::mem::take(&mut e.negated),
95            Expr::InSubquery(e) => std::mem::take(&mut e.negated),
96            _ => false,
97        };
98        if negated {
99            predicate = Expr::not(predicate);
100        }
101
102        *expr = if guards.is_empty() {
103            predicate
104        } else {
105            guards.push(predicate);
106            Expr::and_from_vec(guards)
107        };
108    }
109}
110
111fn is_guard(expr: &Expr) -> bool {
112    matches!(expr, Expr::IsVariant(_))
113}
114
115/// Collects the `is_variant` checks for the selections an operand reaches.
116///
117/// Outer selections come before the selections nested inside them, without
118/// duplicates and without the checks already in force. Connectives and
119/// subqueries are predicate scopes of their own and are not entered.
120struct CollectGuards<'a> {
121    guards: &'a mut Vec<Expr>,
122    existing_guards: &'a [Expr],
123}
124
125impl Visit for CollectGuards<'_> {
126    fn visit_expr_variant(&mut self, i: &ExprVariant) {
127        // The base holds the enclosing selections, so its guards come first.
128        self.visit_expr(&i.base);
129
130        let guard = Expr::is_variant((*i.base).clone(), i.variant);
131        if !self.existing_guards.contains(&guard) && !self.guards.contains(&guard) {
132            self.guards.push(guard);
133        }
134    }
135
136    fn visit_expr_and(&mut self, _: &ExprAnd) {}
137
138    fn visit_expr_or(&mut self, _: &ExprOr) {}
139
140    fn visit_expr_not(&mut self, _: &ExprNot) {}
141
142    fn visit_stmt(&mut self, _: &Statement) {}
143
144    fn visit_stmt_query(&mut self, _: &Query) {}
145}
