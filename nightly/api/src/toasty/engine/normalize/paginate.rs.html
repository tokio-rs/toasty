9impl Normalize<'_> {
10    /// Appends missing physical primary-key columns to ambiguous SQL cursor
11    /// ordering.
12    ///
13    /// The public query retains only the user-provided ordering. Normalization
14    /// runs on the cloned statement passed to the engine, making these
15    /// tie-breakers internal to execution and cursor generation. Physical
16    /// column references keep embedded newtype keys scalar after lowering.
17    pub(super) fn normalize_cursor_order(&mut self, query: &mut stmt::Query) {
18        if !self.capability.sql() || !matches!(query.limit, Some(stmt::Limit::Cursor(_))) {
19            return;
20        }
21
22        let Some(order_by) = &mut query.order_by else {
23            return;
24        };
25        if order_by.exprs.is_empty() {
26            return;
27        }
28        let Some(select) = query.body.as_select() else {
29            return;
30        };
31        let Some(model_id) = select.source.model_id() else {
32            return;
33        };
34
35        let mapping = self.schema.mapping.model(model_id);
36        let table = self.schema.db.table(mapping.table);
37        let mut ordered_columns: HashSet<_> = order_by
38            .exprs
39            .iter()
40            .filter_map(|order| resolve_order_column(mapping, &order.expr))
41            .collect();
42
43        let unambiguous = table.indices.iter().any(|index| {
44            (index.unique || index.primary_key)
45                && index
46                    .columns
47                    .iter()
48                    .all(|index_column| ordered_columns.contains(&index_column.column))
49                && index
50                    .columns
51                    .iter()
52                    .all(|index_column| !table.column(index_column.column).nullable)
53        });
54        if unambiguous {
55            return;
56        }
57
58        let direction = order_by.exprs.last().and_then(|order| order.order);
59        for column in &table.primary_key.columns {
60            if !ordered_columns.insert(*column) {
61                continue;
62            }
63
64            order_by.exprs.push(stmt::OrderByExpr {
65                expr: stmt::Expr::column(*column),
66                order: direction,
67            });
68        }
69    }
70}

72/// Resolves a model field path used by `order_by` to its single physical
73/// column. Expressions that do not resolve to exactly one column cannot prove
74/// that an order is unique.
75fn resolve_order_column(
76    mapping: &toasty_core::schema::mapping::Model,
77    expr: &stmt::Expr,
78) -> Option<ColumnId> {
79    if let stmt::Expr::Reference(ExprReference::Column(column)) = expr {
80        return (column.nesting == 0 && column.table == 0).then_some(ColumnId {
81            table: mapping.table,
82            index: column.column,
83        });
84    }
85
86    let projection = field_projection(expr)?;
87    let field = mapping.resolve_field_mapping(&projection)?;
88    let mut columns = field.columns();
89    let (column, _) = columns.next()?;
90    columns.next().is_none().then_some(column)
91}
92
93fn field_projection(expr: &stmt::Expr) -> Option<stmt::Projection> {
94    match expr {
95        stmt::Expr::Reference(ExprReference::Field { nesting: 0, index }) => {
96            Some(stmt::Projection::single(*index))
97        }
98        stmt::Expr::Project(project) => {
99            let mut projection = field_projection(&project.base)?;
100            for step in &project.projection {
101                projection.push(step);
102            }
103            Some(projection)
104        }
105        _ => None,
106    }
107}
