50struct Exec<'a> {
51    engine: &'a Engine,
52    connection: &'a mut dyn Connection,
53    vars: VarStore,
54    /// True when an outer transaction is active on this connection. Used by
55    /// ReadModifyWrite to decide between savepoints (nested) and its own
56    /// BEGIN/COMMIT (standalone).
57    in_transaction: bool,
58}
59
60impl Engine {
61    pub(crate) async fn exec_plan(
62        &self,
63        connection: &mut dyn Connection,
64        plan: ExecPlan,
65        in_transaction: bool,
66    ) -> Result<ExecResponse> {
67        let ExecPlan {
68            plan: logical_plan,
69            vars,
70            steps,
71            returning,
72            needs_transaction,
73        } = plan;
74
75        let mut exec = Exec {
76            engine: self,
77            connection,
78            vars,
79            in_transaction,
80        };
81
82        // When nested inside an outer transaction use savepoints so the outer
83        // transaction can still commit or roll back as a whole. When standalone,
84        // start our own transaction (MySQL requires an active BEGIN before
85        // SAVEPOINT can be used, so we can't use savepoints here).
86        let (begin, commit, rollback) = if exec.in_transaction {
87            let name = "statement";
88            (
89                Transaction::Savepoint(name.to_owned()),
90                Transaction::ReleaseSavepoint(name.to_owned()),
91                Transaction::RollbackToSavepoint(name.to_owned()),
92            )
93        } else {
94            (
95                Transaction::start(),
96                Transaction::Commit,
97                Transaction::Rollback,
98            )
99        };
100
101        if needs_transaction {
102            tracing::trace!("beginning plan transaction");
103            exec.connection.exec(&self.schema, begin.into()).await?;
104            exec.in_transaction = true;
105        }
106
107        for (i, step) in steps.iter().enumerate() {
108            tracing::trace!(step = i, action = %step.name(&logical_plan), "executing action");
109            // Debug, not error: the failure propagates to the caller, who
110            // decides whether it is an application error. A handled unique
111            // violation should not error-spam production logs.
112            if let Err(e) = exec.exec_step(&logical_plan, step).await {
113                tracing::debug!(step = i, action = %step.name(&logical_plan), error = %e, "action failed");
114                if needs_transaction {
115                    tracing::trace!("rolling back plan transaction due to error");
116                    // Best effort: ignore rollback errors so the original error is returned
117                    let _ = exec.connection.exec(&self.schema, rollback.into()).await;
118                }
119                return Err(e);
120            }
121        }
122
123        if needs_transaction {
124            tracing::trace!("committing plan transaction");
125            exec.connection.exec(&self.schema, commit.into()).await?;
126        }
127
128        let response = exec.vars.load(returning).await?;
129        tracing::trace!("final result from var {:?}:\n{:#?}", returning, response);
130
131        // With exact use counts, the returning load was the last use of
132        // the last live slot (success path only — the loop above returned
133        // early on failure).
134        exec.vars.assert_empty();
135
136        let value_stream = match response.values {
137            Rows::Count(_) => ValueStream::default(),
138            Rows::Value(stmt::Value::List(items)) => ValueStream::from_vec(items),
139            // TODO have the public API be able to handle single rows
140            Rows::Value(value) => ValueStream::from_vec(vec![value]),
141            Rows::Stream(value_stream) => value_stream,
142        };
143
144        Ok(ExecResponse {
145            values: Rows::Stream(value_stream),
146            next_cursor: response.next_cursor,
147            prev_cursor: response.prev_cursor,
148        })
149    }
150}

152impl Exec<'_> {
153    async fn exec_step(&mut self, logical_plan: &mir::LogicalPlan, step: &Step) -> Result<()> {
154        match step {
155            Step::Run(node_id) => self.exec_node(logical_plan, *node_id).await,
156            Step::If(action) => self.action_if(logical_plan, action).await,
157        }
158    }
159
160    /// Executes one node's operation and stores its output in the node's
161    /// variable slot.
162    async fn exec_node(
163        &mut self,
164        logical_plan: &mir::LogicalPlan,
165        node_id: mir::NodeId,
166    ) -> Result<()> {
167        use mir::Operation;
168
169        let node = &logical_plan[node_id];
170
171        let response = match &node.op {
172            // A pass-through: the response (stream included) relocates
173            // between slots without buffering.
174            Operation::Alias(op) => self.vars.load(op.input).await?,
175            Operation::Const(op) => ExecResponse::from_rows(Rows::Value(op.value.clone())),
176            Operation::DeleteByKey(op) => self.exec_delete_by_key(op).await?,
177            Operation::Eval(op) => self.exec_eval(op).await?,
178            Operation::ExecStatement(op) => self.exec_statement(op).await?,
179            Operation::Filter(op) => self.exec_filter(op).await?,
180            Operation::FindPkByIndex(op) => self.exec_find_pk_by_index(op).await?,
181            Operation::GetByKey(op) => self.exec_get_by_key(op).await?,
182            Operation::NestedMerge(op) => self.exec_nested_merge(op).await?,
183            Operation::ReadModifyWrite(op) => self.exec_read_modify_write(op).await?,
184            Operation::Repeat(op) => self.exec_repeat(op).await?,
185            Operation::QueryPk(op) => self.exec_query_pk(op).await?,
186            Operation::Scan(op) => self.exec_scan(op).await?,
187            Operation::UpdateByKey(op) => self.exec_update_by_key(op).await?,
188            Operation::Upsert(op) => self.exec_upsert(op).await?,
189        };
190
191        self.vars.store(node_id, node.ty(), node.num_uses, response);
192
193        Ok(())
194    }
195
196    async fn collect_input(
197        &mut self,
198        input: impl IntoIterator<Item = mir::NodeId>,
199    ) -> Result<Vec<stmt::Value>> {
200        let mut ret = Vec::new();
201
202        for node_id in input {
203            let response = self.vars.load(node_id).await?;
204            let value = response.values.collect_as_value().await?;
205            ret.push(value);
206        }
207
208        Ok(ret)
209    }
210}
