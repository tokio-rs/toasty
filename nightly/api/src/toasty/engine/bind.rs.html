1//! Bind a fully-resolved statement's inline values into typed parameters.
2//!
3//! Three phases:
4//! 1. **Extract** ([`extract`]): replace scalar `Value` nodes with `Arg(n)`
5//!    placeholders, initializing each param's type from the value itself.
6//! 2. **Synthesize** (bottom-up): compute each node's inferred type from its
7//!    children (column refs get their storage type from the schema, records get
8//!    a tuple of field types, etc.).
9//! 3. **Check** (top-down): push refined types into `Arg(n)` nodes, upgrading a
10//!    param when context is more precise (e.g. `Enum` over `Text`).
11//!
12//! Synthesize and check ([`infer`]) run together in a single recursive walk.
13//! Types carry **provenance** (`Column` vs `Inferred`) so schema-authoritative
14//! column types win over value-inferred guesses when merging.
15//!
16//! `#[document]` values are named into their `Value::Object` form before this
17//! runs: the mapping's lowering casts convert them during statement
18//! lowering/simplification, and document *paths* are resolved by legalization
19//! ([`super::legalize`]) — `Engine::prepare_for_driver` runs both in order.
20

33/// Bind a statement's inline values: replace scalar `Value` nodes with
34/// `Expr::Arg(n)` placeholders and infer a precise `db::Type` for each. The
35/// returned `Vec<TypedValue>` is indexed by the `n` in each placeholder.
36///
37/// The statement must already be legalized ([`super::legalize`]): document
38/// values named, document paths resolved. Runs via
39/// `Engine::prepare_for_driver`, as the final engine step before a driver
40/// serializes the statement.
41pub(crate) fn run(
42    stmt: &mut stmt::Statement,
43    db_schema: &db::Schema,
44    capability: &Capability,
45) -> Vec<TypedValue> {
46    let mut params: Vec<Param> = Vec::new();
47
48    // Phase 0: transpose a multi-row INSERT into per-column array params
49    // where the driver can bind them as `unnest(...)`.
50    transpose_insert_unnest(stmt, db_schema, capability, &mut params);
51
52    // Phase 1: mechanical extraction — replace values with Arg(n).
53    extract::extract_values(stmt, &mut params, capability);
54
55    // Phases 2+3: bidirectional type inference — refine param types.
56    infer::refine_param_types(stmt, db_schema, &mut params);
57
58    // Materialize the final TypedValues. `finalize_ty` panics if any param is
59    // still unresolved — synthesize/check is expected to type every param.
60    params
61        .into_iter()
62        .map(|p| {
63            let Param { value, ty } = p;
64            TypedValue {
65                ty: finalize_ty(&value, ty),
66                value,
67            }
68        })
69        .collect()
70}

72/// Rewrite a multi-row `INSERT ... VALUES` into parallel `unnest` table
73/// functions over per-column array params, serialized as
74/// `SELECT * FROM ROWS FROM (unnest($1), unnest($2))`.
75/// No-op for single-row inserts, non-scalar target columns, non-value rows,
76/// or when [`Capability::insert_values_unnest`] is off.
77///
78/// Each column array is bound as a param here, not left for extract: extract
79/// refuses lists containing NULL, but a NULL cell must ride inside the array
80/// bind. Must run after lowering (returning constantization assumes the
81/// per-row `VALUES` shape) and before serialization.
82fn transpose_insert_unnest(
83    stmt: &mut stmt::Statement,
84    db_schema: &db::Schema,
85    capability: &Capability,
86    params: &mut Vec<Param>,
87) {
88    if !capability.insert_values_unnest {
89        return;
90    }
91
92    let stmt::Statement::Insert(insert) = stmt else {
93        return;
94    };
95    let stmt::InsertTarget::Table(table) = &insert.target else {
96        return;
97    };
98    let stmt::ExprSet::Values(values) = &mut insert.source.body else {
99        return;
100    };
101
102    if values.rows.len() < 2 {
103        return;
104    }
105
106    let num_cols = table.columns.len();
107    if num_cols == 0 {
108        return;
109    }
110
111    // A list/document column's per-row value is itself a collection, not a
112    // single array element. `Json`/`Jsonb` are the native spellings of the
113    // same shape and are excluded for the same reason: PostgreSQL has no
114    // array OID registered for them, so transposing panics the driver.
115    let db_table = &db_schema.tables[table.table.0];
116    let all_scalar = table
117        .columns
118        .iter()
119        .all(|col_id| column_supports_unnest(&db_table.columns[col_id.index].storage_ty));
120    if !all_scalar {
121        tracing::debug!(
122            table = %db_table.name,
123            "multi-row INSERT not transposed to unnest: non-scalar target column"
124        );
125        return;
126    }
127
128    let rows_ok = values
129        .rows
130        .iter()
131        .all(|row| row_is_transposable(row, num_cols));
132    if !rows_ok {
133        tracing::debug!(
134            table = %db_table.name,
135            "multi-row INSERT not transposed to unnest: non-value row shape"
136        );
137        return;
138    }
139
140    // Matched directly rather than via `Expr::into_record_items`: this is the
141    // bulk-insert hot path and the generic iterator boxes per row.
142    let mut columns: Vec<Vec<stmt::Value>> = (0..num_cols)
143        .map(|_| Vec::with_capacity(values.rows.len()))
144        .collect();
145    for row in std::mem::take(&mut values.rows) {
146        match row {
147            stmt::Expr::Value(stmt::Value::Record(record)) => {
148                for (col, value) in record.fields.into_iter().enumerate() {
149                    columns[col].push(value);
150                }
151            }
152            stmt::Expr::Record(record) => {
153                for (col, field) in record.fields.into_iter().enumerate() {
154                    let stmt::Expr::Value(value) = field else {
155                        panic!("row validated as plain values above; field={field:?}");
156                    };
157                    columns[col].push(value);
158                }
159            }
160            _ => unreachable!("row validated as a record above"),
161        }
162    }
163
164    let funcs = columns
165        .into_iter()
166        .zip(&table.columns)
167        .map(|(cells, col_id)| {
168            let value = stmt::Value::List(cells);
169            let ty = Ty::List(Box::new(infer::ty_from_column(
170                db_table.columns[col_id.index].storage_ty.clone(),
171            )));
172            let position = params.len();
173            params.push(Param { value, ty });
174            stmt::FuncUnnest {
175                arg: Box::new(stmt::Expr::arg(position)),
176            }
177            .into()
178        })
179        .collect::<Vec<_>>();
180
181    let source = stmt::Source::table(stmt::TableRef::RowsFrom(funcs));
182    insert.source.body = stmt::ExprSet::Select(Box::new(stmt::Select {
183        returning: stmt::Returning::Star,
184        source,
185        filter: stmt::Filter::ALL,
186        distinct: false,
187    }));
188}

190/// A row is transposable if it is a record of exactly `num_cols` plain values.
191fn row_is_transposable(row: &stmt::Expr, num_cols: usize) -> bool {
192    if row.record_len() != Some(num_cols) {
193        return false;
194    }
195    match row {
196        stmt::Expr::Value(_) => true,
197        stmt::Expr::Record(record) => record
198            .fields
199            .iter()
200            .all(|field| matches!(field, stmt::Expr::Value(_))),
201        _ => false,
202    }
203}

205fn column_supports_unnest(ty: &db::Type) -> bool {
206    !matches!(
207        ty,
208        db::Type::List(_) | db::Type::Document { .. } | db::Type::Json | db::Type::Jsonb
209    )
210}

212/// A bind parameter being inferred. Once inference completes, the `Ty` is
213/// converted to a concrete `db::Type` for the `TypedValue`.
214struct Param {
215    value: stmt::Value,
216    ty: Ty,
217}
218
219/// Resolve a `Ty` to a concrete `db::Type`. Panics on `Unknown` / `Record` —
220/// every param should be fully inferred by the synthesize/check pass; if a
221/// statement reaches here with an unresolved param, that's a bug worth
222/// surfacing so we can evaluate the specific case.
223fn finalize_ty(value: &stmt::Value, ty: Ty) -> db::Type {
224    match ty {
225        Ty::Column(t) | Ty::Inferred(t) => t,
226        Ty::List(elem) => db::Type::list(finalize_ty(value, *elem)),
227        Ty::Unknown => panic!("bind left {value:?} with unresolved type"),
228        Ty::Record(_) => panic!(
229            "bind left {value:?} typed as a record; only scalars and lists are extracted as params"
230        ),
231    }
232}

238/// The inferred database-level type of an expression node.
239///
240/// Each scalar type carries **provenance**: `Column` means the type came from
241/// the schema (authoritative), `Inferred` means it was guessed from the value.
242/// Column types always win when merging.
243#[derive(Debug, Clone)]
244enum Ty {
245    /// Type from a column reference or schema (authoritative).
246    Column(db::Type),
247    /// Type inferred from a value (initial guess — may be less specific).
248    Inferred(db::Type),
249    /// A tuple of types (one per field).
250    Record(Vec<Ty>),
251    /// A homogeneous list where all elements share a type.
252    List(Box<Ty>),
253    /// Type could not be determined.
254    Unknown,
255}
256
257impl Ty {
258    /// Extract the `db::Type`, regardless of provenance.
259    fn db_type(&self) -> Option<&db::Type> {
260        match self {
261            Ty::Column(ty) | Ty::Inferred(ty) => Some(ty),
262            _ => None,
263        }
264    }
265
266    /// Returns true if this type comes from the schema (authoritative).
267    #[cfg(test)]
268    fn is_column(&self) -> bool {
269        matches!(self, Ty::Column(_))
270    }
271}
