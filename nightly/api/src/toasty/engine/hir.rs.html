14/// High-level Intermediate Representation of a query.
15///
16/// [`HirStatement`] is a collection of related statements with tracked
17/// dependencies, produced by the lowering phase. It captures the structure of
18/// a query after model-to-table transformation but before operation graph
19/// construction.
20///
21/// The HIR tracks which statements depend on results from others, enabling the
22/// planner to determine execution order and identify opportunities for batching.
23/// The dependency graph may contain cycles when preloading associations (e.g.,
24/// users → todos → users), which the planning phase must break when building
25/// the final DAG.
26#[derive(Debug)]
27pub(super) struct HirStatement {
28    /// Storage for all statement metadata, indexed by [`StmtId`].
29    store: IndexVec<StmtId, StatementInfo>,
30}
31
32/// Metadata for a single statement within the HIR.
33///
34/// Each [`StatementInfo`] represents one database operation that will execute
35/// as part of the query. Not every sub-expression in the original query becomes
36/// a [`StatementInfo`]; only those that execute as separate database operations
37/// (e.g., the root query and any `include()` subqueries).
38///
39/// It tracks the statement's dependencies on other statements, arguments passed
40/// from parent or child statements, and references to the MIR nodes created
41/// during planning.
42#[derive(Debug)]
43pub(super) struct StatementInfo {
44    /// The lowered statement to execute.
45    ///
46    /// Initially `None`, populated during lowering after all transformations
47    /// are complete. Contains the table-level statement ready for planning.
48    pub(super) stmt: Option<Box<stmt::Statement>>,
49
50    /// Whether cursor pagination resumed this query after an earlier page.
51    pub(super) has_pagination_cursor: bool,
52
53    /// Ordering edges: statements that must execute, to the degree the
54    /// [`DepKind`] demands, before this one.
55    ///
56    /// These are pure ordering constraints — value flow between statements is
57    /// carried by [`Self::args`] and [`Self::back_refs`], never by this map.
58    /// One entry per target: inserting via [`StatementInfo::add_dep`]
59    /// upgrades an [`Effect`](DepKind::Effect) entry to
60    /// [`Statement`](DepKind::Statement), which subsumes it. An `IndexMap`
61    /// keeps iteration — and therefore planning order — deterministic.
62    pub(super) deps: IndexMap<StmtId, DepKind>,
63
64    /// Arguments that flow into this statement from other statements.
65    ///
66    /// Each [`Arg`] describes data passed from a sub-statement ([`Arg::Sub`]) or
67    /// referenced from a parent statement ([`Arg::Ref`]). During planning, these
68    /// arguments become edges in the operation graph.
69    pub(super) args: Vec<Arg>,
70
71    /// Column references from child statements that point back to this one.
72    ///
73    /// When a child statement references columns from this statement (via
74    /// [`Arg::Ref`]), those columns must be included in this statement's
75    /// batch-load query. The key is the child's [`StmtId`], and the value
76    /// tracks which columns are referenced.
77    pub(super) back_refs: HashMap<StmtId, BackRef>,
78
79    /// MIR node that executes this statement's database query.
80    ///
81    /// Set during planning when the statement is converted to an operation.
82    /// Used to wire up dependencies between operations.
83    pub(super) load_data_statement: Cell<Option<mir::NodeId>>,
84
85    /// Items selected by the `exec_statement` operation.
86    ///
87    /// Populated during planning to track which items are fetched from the
88    /// database. Used to resolve column references in child statements.
89    pub(super) load_data_select_items: OnceCell<SelectItems>,
90
91    /// MIR node representing this statement's final output.
92    ///
93    /// May differ from `exec_statement` when post-processing is required
94    /// (e.g., filtering, projection, or nested merge). Dependent statements
95    /// reference this node to establish execution order.
96    pub(super) output: Cell<Option<mir::NodeId>>,
97
98    /// When true, the statement is independent. An independent statement does
99    /// not depend on any anestors itself nor do any of its sub-dependencies.
100    pub(super) independent: bool,
101
102    /// True once the statement's planning pass has started. Guards against
103    /// re-entrant planning when a statement is reached again through a
104    /// dependency cycle that is acyclic at the operation level.
105    pub(super) planning: Cell<bool>,
106}

112/// The kind of ordering edge between two statements.
113#[derive(Debug, Clone, Copy, PartialEq, Eq)]
114pub(super) enum DepKind {
115    /// The target's full result must exist first. During planning the edge
116    /// anchors on the target's output node, keeping everything the target's
117    /// returning consumed ordered and reachable. Targets are never
118    /// ancestors.
119    Statement,
120
121    /// The target's database write must have happened first. The target may
122    /// be an ancestor still being planned — e.g. a `belongs_to`
123    /// returning-load subquery ordered after its enclosing INSERTs. Such an
124    /// edge is cyclic at statement granularity (the ancestor's output
125    /// transitively consumes this statement's result) but acyclic at the
126    /// operation level, so it anchors on the target's data-loading node —
127    /// reserved up front so the reference resolves regardless of planning
128    /// order.
129    Effect,
130}

132impl StatementInfo {
133    /// Creates a new [`StatementInfo`] with the given dependencies.
134    ///
135    /// All other fields are initialized to empty or `None` and populated
136    /// later during lowering and planning.
137    pub(super) fn new(deps: IndexMap<StmtId, DepKind>) -> StatementInfo {
138        StatementInfo {
139            stmt: None,
140            has_pagination_cursor: false,
141            deps,
142            args: vec![],
143            back_refs: HashMap::new(),
144            load_data_statement: Cell::new(None),
145            load_data_select_items: OnceCell::new(),
146            output: Cell::new(None),
147            independent: true,
148            planning: Cell::new(false),
149        }
150    }
151
152    pub(super) fn stmt(&self) -> &stmt::Statement {
153        self.stmt.as_deref().unwrap()
154    }
155
156    /// Records an ordering edge on `target`. A [`DepKind::Statement`] edge
157    /// subsumes an [`DepKind::Effect`] edge to the same target, so inserting
158    /// the stronger kind upgrades an existing entry and inserting the weaker
159    /// kind over the stronger is a no-op.
160    pub(super) fn add_dep(&mut self, target: StmtId, kind: DepKind) {
161        let entry = self.deps.entry(target).or_insert(kind);
162        if kind == DepKind::Statement {
163            *entry = DepKind::Statement;
164        }
165    }
166}

168/// Tracks columns referenced by a child statement.
169///
170/// When a child statement (e.g., an `include()` subquery) references columns
171/// from its parent, the parent must include those columns in its result set.
172/// [`BackRef`] records which columns are needed and the MIR node that projects
173/// them for the child's batch-load operation.
174#[derive(Debug, Default)]
175pub(super) struct BackRef {
176    /// Column expressions from this statement that are referenced by a child statement.
177    ///
178    /// When a child statement references columns from this statement (via
179    /// [`Arg::Ref`]), those columns must be included in this statement's
180    /// batch-load query. This set tracks which columns need to be loaded so they
181    /// can be used during nested merge.
182    pub(super) exprs: IndexSet<stmt::ExprReference>,
183
184    /// Node ID of the projection operation that extracts these back-ref columns.
185    ///
186    /// After executing this statement, a projection node is created to extract just
187    /// the columns needed by child statements. This projection's output is used as
188    /// input to the child statement's batch-load operation.
189    pub(super) node_id: Cell<Option<mir::NodeId>>,
190}

192/// An argument that flows between statements in the HIR.
193///
194/// Arguments represent data dependencies between statements. They describe how
195/// results from one statement are used by another, enabling the planner to wire
196/// up the operation graph correctly.
197#[derive(Debug)]
198pub(super) enum Arg {
199    /// Data from a sub-statement that feeds into this statement.
200    Sub {
201        /// The statement ID that provides the data for this argument.
202        stmt_id: StmtId,
203
204        /// True when the sub-statement is used in the returning clause, false when used in filters.
205        ///
206        /// Determines how the sub-statement is handled during planning:
207        /// - `true`: Data is merged with parent rows via [`NestedMerge`](crate::engine::mir::NestedMerge)
208        /// - `false`: Data is used as input to filter expressions
209        returning: bool,
210
211        /// Index in the operation's inputs list. Set during planning.
212        input: Cell<Option<usize>>,
213
214        /// Depending on the statement type, this is used in different ways. For
215        /// Query types, it is the index of the TableRef that will provide the
216        /// data for this ref. For Insert statements, it is the offset at which
217        /// the data should be fetched.
218        batch_load_index: Cell<Option<usize>>,
219    },
220
221    /// A reference to a parent statement's columns.
222    Ref {
223        /// The expression reference relative to the target statement.
224        target_expr_ref: stmt::ExprReference,
225
226        /// The parent statement that provides the data for this reference.
227        stmt_id: StmtId,
228
229        /// Number of nesting levels between this statement and the referenced parent.
230        ///
231        /// A value of 1 means the immediate parent, 2 means the grandparent, etc.
232        nesting: usize,
233
234        /// The MIR node input when the ref is used during the data loading phase
235        data_load_input: Cell<Option<usize>>,
236
237        /// The MIR node input when the ref is used during the returing phase
238        returning_input: Cell<Option<usize>>,
239
240        /// Depending on the statement type, this is used in different ways. For
241        /// Query types, it is the index of the TableRef that will provide the
242        /// data for this ref. For Insert statements, it is the offset at which
243        /// the data should be fetched.
244        batch_load_index: Cell<Option<usize>>,
245    },
246}

248impl HirStatement {
249    /// Creates an empty [`HirStatement`] with no statements.
250    pub(super) fn new() -> HirStatement {
251        HirStatement {
252            store: IndexVec::new(),
253        }
254    }
255
256    /// Inserts a [`StatementInfo`] and returns its assigned [`StmtId`].
257    pub(super) fn insert(&mut self, info: StatementInfo) -> StmtId {
258        self.store.push(info)
259    }
260
261    /// Creates and inserts a new [`StatementInfo`] with the given dependencies.
262    pub(super) fn new_statement_info(&mut self, deps: IndexMap<StmtId, DepKind>) -> StmtId {
263        self.insert(StatementInfo::new(deps))
264    }
265
266    /// Returns the [`StmtId`] of the root statement.
267    ///
268    /// The root statement is always the first one inserted (index 0) and
269    /// represents the top-level query that produces the final result.
270    pub(super) fn root_id(&self) -> StmtId {
271        StmtId::from(0)
272    }
273
274    /// Returns a reference to the root [`StatementInfo`].
275    pub(super) fn root(&self) -> &StatementInfo {
276        let root_id = self.root_id();
277        &self.store[root_id]
278    }
279
280    /// Iterates all statements.
281    pub(super) fn statements(&self) -> impl Iterator<Item = &StatementInfo> {
282        self.store.iter()
283    }
284}
