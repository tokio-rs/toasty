4impl Simplify<'_> {
5    /// Combine `IN` subqueries joined by `AND` when they test the same row.
6    ///
7    /// For example, if `users.id` is a primary key, this filter:
8    ///
9    /// ```sql
10    /// user_id IN (SELECT id FROM users WHERE name = 'alice')
11    /// AND user_id IN (SELECT id FROM users WHERE active)
12    /// ```
13    /// becomes:
14    ///
15    /// ```sql
16    /// user_id IN (SELECT id FROM users WHERE name = 'alice' AND active)
17    /// ```
18    /// Both checks refer to the same user. That user must satisfy both
19    /// filters. The subqueries can come from relation paths, explicit queries,
20    /// or a mix of the two. Their operands need not be next to each other.
21    ///
22    /// The filters can contain `OR` or nested subqueries. For example, combining
23    /// a filter for either Alice or Bob with a filter for active users gives:
24    ///
25    /// ```sql
26    /// user_id IN (
27    ///     SELECT id FROM users
28    ///     WHERE (name = 'alice' OR name = 'bob') AND active
29    /// )
30    /// ```
31    ///
32    /// Both queries must test the same value and read the same columns from
33    /// the same source. The returned columns must include a complete unique
34    /// key, and none may be nullable. A secondary unique key works too. For a
35    /// key made of `(tenant_id, id)`, returning only `tenant_id` is not enough.
36    ///
37    /// A query that returns a non-unique value stays separate. For example:
38    ///
39    /// ```sql
40    /// parent_id IN (SELECT parent_id FROM children WHERE color = 'red')
41    /// AND parent_id IN (SELECT parent_id FROM children WHERE color = 'blue')
42    /// ```
43    ///
44    /// A parent can have one red child and another blue child. Combining the
45    /// filters would require one child to be both red and blue. This is why
46    /// separate `has_many` checks cannot usually be combined.
47    ///
48    /// Queries with limits also stay separate. For example, adding `active`
49    /// to `WHERE name = 'alice' LIMIT 1` can select a different user from the
50    /// one chosen before the merge. This rule also skips ordering, locks,
51    /// CTEs, joins, set operations, and queries marked as returning one row.
52    /// Those query restrictions also apply to nested subqueries. Queries that
53    /// contain function calls or references to an outer query stay separate
54    /// too. This rule combines only positive `IN` checks, not `NOT IN` checks.
55    ///
56    /// SQL nulls need care. If the left value is null, each `IN` check can be
57    /// null even when the subqueries have no rows in common. The merged query
58    /// then returns no rows, making `IN` false. Both results are rejected by a
59    /// `WHERE` filter, including through `AND` or `OR`. They differ under `NOT` or when returned
60    /// as a value, so those cases require a proven non-null left value. NoSQL
61    /// membership always returns a boolean and does not need this restriction.
62    ///
63    /// This runs after relation lifting and before lowering extracts NoSQL
64    /// subqueries into separate statements. Newly combined filters are
65    /// simplified too, so matching subqueries inside them can also combine.
66    pub(super) fn merge_in_subqueries(&self, operands: &mut Vec<Expr>) {
67        for i in 0..operands.len() {
68            // Earlier merges can shorten the operand list.
69            if i >= operands.len() {
70                break;
71            }
72            let Expr::InSubquery(first) = &operands[i] else {
73                continue;
74            };
75            if !self.can_merge_membership(first) {
76                continue;
77            }
78
79            let mut j = i + 1;
80            while j < operands.len() {
81                let Expr::InSubquery(first) = &operands[i] else {
82                    unreachable!();
83                };
84                let compatible = match &operands[j] {
85                    Expr::InSubquery(other) if self.can_merge_membership(other) => {
86                        let a = first.query.body.as_select_unwrap();
87                        let b = other.query.body.as_select_unwrap();
88                        first.expr.is_equivalent_to(&other.expr)
89                            && a.source == b.source
90                            && a.returning == b.returning
91                            && a.distinct == b.distinct
92                    }
93                    _ => false,
94                };
95                if !compatible {
96                    j += 1;
97                    continue;
98                }
99
100                let Expr::InSubquery(other) = operands.remove(j) else {
101                    unreachable!();
102                };
103                let Expr::InSubquery(first) = &mut operands[i] else {
104                    unreachable!();
105                };
106                let select = first.query.body.as_select_mut_unwrap();
107                select.add_filter(other.query.body.as_select_unwrap().filter.clone());
108                // Combining filters can expose further memberships at a deeper
109                // relation hop. Simplify them before subquery extraction.
110                self.scope(&select.source)
111                    .visit_filter_mut(&mut select.filter);
112            }
113        }
114    }

116    fn can_merge_membership(&self, membership: &stmt::ExprInSubquery) -> bool {
117        if membership.negated || !membership.expr.is_stable() {
118            return false;
119        }
120        // In SQL, NULL IN two nonempty disjoint sets is NULL AND NULL, while
121        // NULL IN their intersection is false. They select the same rows in
122        // a positive filter, but are not interchangeable as values or under NOT.
123        // NoSQL membership uses value equality and always returns a boolean.
124        if self.capability.sql()
125            && !self.positive_filter
126            && !self.non_nullable_key(&membership.expr)
127        {
128            return false;
129        }
130        let mut safe = SafeQuery(true);
131        safe.visit_stmt_query(&membership.query);
132        if !safe.0 {
133            return false;
134        }
135        let select = membership.query.body.as_select_unwrap();
136        let stmt::Returning::Project(projection) = &select.returning else {
137            return false;
138        };
139        let fields = projection
140            .as_record()
141            .map_or(std::slice::from_ref(projection), |record| &record.fields);
142
143        match &select.source {
144            Source::Model(source) => {
145                let model = self.cx.schema().app.model(source.id).as_root_unwrap();
146                let Some(indices) = fields
147                    .iter()
148                    .map(|expr| match expr {
149                        Expr::Reference(ExprReference::Field { nesting: 0, index })
150                            if !model.fields[*index].nullable =>
151                        {
152                            Some(*index)
153                        }
154                        _ => None,
155                    })
156                    .collect::<Option<Vec<_>>>()
157                else {
158                    return false;
159                };
160                model.indices.iter().any(|index| {
161                    index.unique
162                        && !index.fields.is_empty()
163                        && index
164                            .fields
165                            .iter()
166                            .all(|field| indices.contains(&field.field.index))
167                })
168            }
169            Source::Table(source) => {
170                let stmt::TableRef::Table(id) = source.tables[0] else {
171                    unreachable!();
172                };
173                let table = self.cx.schema().db.table(id);
174                let Some(indices) = fields
175                    .iter()
176                    .map(|expr| match expr {
177                        Expr::Reference(ExprReference::Column(column))
178                            if column.nesting == 0
179                                && column.table == 0
180                                && !table.columns[column.column].nullable =>
181                        {
182                            Some(column.column)
183                        }
184                        _ => None,
185                    })
186                    .collect::<Option<Vec<_>>>()
187                else {
188                    return false;
189                };
190                table.indices.iter().any(|index| {
191                    index.unique
192                        && !index.columns.is_empty()
193                        && index
194                            .columns
195                            .iter()
196                            .all(|column| indices.contains(&column.column.index))
197                })
198            }
199        }
200    }

202    fn non_nullable_key(&self, expr: &Expr) -> bool {
203        match expr {
204            Expr::Reference(reference @ ExprReference::Field { .. }) => !self
205                .cx
206                .resolve_expr_reference(reference)
207                .as_field_unwrap()
208                .nullable(),
209            Expr::Record(record) => record
210                .fields
211                .iter()
212                .all(|field| self.non_nullable_key(field)),
213            Expr::Value(value) => !value.is_null() && !matches!(value, stmt::Value::Record(_)),
214            // Columns may be nullable through an outer join even when their
215            // schema declaration is not. Other expressions need a nullability proof.
216            _ => false,
217        }
218    }

221/// Restrict merging to plain reads. Walk nested subqueries too: their filters
222/// may contain arbitrary boolean expressions, but must not introduce effects,
223/// unstable functions, or references to an outer query scope.
224struct SafeQuery(bool);
225
226impl Visit for SafeQuery {
227    fn visit_stmt_query(&mut self, query: &stmt::Query) {
228        if query.with.is_some()
229            || query.limit.is_some()
230            || query.order_by.is_some()
231            || !query.locks.is_empty()
232            || query.single
233        {
234            self.0 = false;
235            return;
236        }
237        let Some(select) = query.body.as_select() else {
238            self.0 = false;
239            return;
240        };
241        let plain_source = match &select.source {
242            Source::Model(source) => source.via.is_none(),
243            Source::Table(source) => {
244                matches!(source.tables.as_slice(), [stmt::TableRef::Table(_)])
245                    && matches!(source.from.as_slice(), [from]
246                    if matches!(from.relation, stmt::TableFactor::Table(stmt::SourceTableId(0)))
247                        && from.joins.is_empty())
248            }
249        };
250        if !plain_source || !matches!(select.returning, stmt::Returning::Project(_)) {
251            self.0 = false;
252            return;
253        }
254        stmt::visit::visit_stmt_query(self, query);
255    }
256
257    fn visit_expr(&mut self, expr: &Expr) {
258        if matches!(
259            expr,
260            Expr::Func(_) | Expr::Stmt(_) | Expr::Default | Expr::Ident(_) | Expr::Error(_)
261        ) {
262            self.0 = false;
263        } else {
264            stmt::visit::visit_expr(self, expr);
265        }
266    }
267
268    fn visit_expr_reference(&mut self, reference: &ExprReference) {
269        let nesting = match reference {
270            ExprReference::Field { nesting, .. } | ExprReference::Model { nesting } => *nesting,
271            ExprReference::Column(column) => column.nesting,
272        };
273        self.0 &= nesting == 0;
274    }
