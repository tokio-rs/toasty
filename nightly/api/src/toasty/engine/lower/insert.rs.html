6/// Process the scope component of an insert statement.
7struct ApplyInsertScope<'a> {
8    expr: &'a mut stmt::Expr,
9}
10
11/// How to encode an auto-generated value for a given field. Each newtype
12/// embed layer adds one record wrapping around the leaf primitive value.
13struct AutoTarget<'a> {
14    ty: &'a stmt::Type,
15    wrap_depth: usize,
16}
17
18impl AutoTarget<'_> {
19    fn wrap(&self, mut value: stmt::Value) -> stmt::Value {
20        for _ in 0..self.wrap_depth {
21            value = stmt::Value::record_from_vec(vec![value]);
22        }
23        value
24    }
25
26    fn wrap_expr(&self, mut expr: stmt::Expr) -> stmt::Expr {
27        for _ in 0..self.wrap_depth {
28            expr = stmt::Expr::record_from_vec(vec![expr]);
29        }
30        expr
31    }
32}

35    // First, apply the insertion scope to the insertion values
36    pub(super) fn apply_insert_scope(
37        &mut self,
38        target: &mut stmt::InsertTarget,
39        source: &mut stmt::Query,
40    ) {
41        let stmt::InsertTarget::Scope(scope) = target else {
42            // Insertion is not targetting a scope
43            return;
44        };
45
46        let stmt::ExprSet::Values(values) = &mut source.body else {
47            todo!()
48        };
49
50        let scope = &scope.body.as_select_unwrap();
51
52        if let Some(filter) = &scope.filter.expr {
53            for expr in &mut values.rows {
54                ApplyInsertScope { expr }.apply_expr(filter)
55            }
56        }
57
58        *target = stmt::InsertTarget::Model(scope.source.model_id_unwrap());
59    }

61    pub(super) fn preprocess_insert_values(
62        &mut self,
63        source: &mut stmt::Query,
64        returning: &mut Option<stmt::Returning>,
65        preserve_returning_projection: bool,
66    ) {
67        let stmt::ExprSet::Values(values) = &mut source.body else {
68            todo!()
69        };
70
71        let Some(model) = self.expr_cx.target_as_model() else {
72            return;
73        };
74
75        let mut set_fields: BitSet<usize> = BitSet::default();
76
77        // First, apply any defaults while also tracking all the fields that are set.
78        for (index, row) in values.rows.iter_mut().enumerate() {
79            self.lower_insert_with_row(index, |lower| {
80                lower.apply_app_level_insertion_defaults(model, row, &mut set_fields);
81            });
82        }
83
84        // If there are any has_n associations included in the insertion, the
85        // statement returning has to be transformed to accomodate the nested
86        // structure.
87        if !preserve_returning_projection {
88            self.convert_returning_for_insert(values, returning, source.single);
89        }
90
91        for (index, row) in values.rows.iter_mut().enumerate() {
92            self.lower_insert_with_row(index, |lower| {
93                lower.plan_stmt_insert_relations(row, returning, index);
94                lower.plan_insert_returning_belongs_to(
95                    row,
96                    returning,
97                    index,
98                    preserve_returning_projection,
99                );
100                lower.verify_field_constraints(model, row);
101            });
102        }
103    }

105    /// Fill each eager `belongs_to` slot of an INSERT's returning with a
106    /// per-row load subquery.
107    ///
108    /// Runs after [`Self::plan_stmt_insert_relations`], which resolves
109    /// relation values into the row's FK source fields. The subquery
110    /// correlates on those FK fields, and running per row (under
111    /// `lower_insert_with_row`) captures the row index the planner needs to
112    /// bind the subquery's parent reference to this row of the INSERT.
113    fn plan_insert_returning_belongs_to(
114        &mut self,
115        row: &stmt::Expr,
116        returning: &mut Option<stmt::Returning>,
117        index: usize,
118        preserve_returning_projection: bool,
119    ) {
120        let Some(model) = self.expr_cx.target_as_model() else {
121            return;
122        };
123        let Some(record) =
124            Self::insert_returning_record_mut(returning, index, preserve_returning_projection)
125        else {
126            return;
127        };
128
129        for field in &model.fields {
130            let app::FieldTy::BelongsTo(rel) = &field.ty else {
131                continue;
132            };
133            if field.deferred {
134                continue;
135            }
136            if Self::belongs_to_fk_is_unset(row, rel) {
137                continue;
138            }
139
140            record[field.id.index] = self.plan_insert_belongs_to_load(
141                field.id.index,
142                index,
143                preserve_returning_projection,
144            );
145        }
146    }

148    /// Return the model record for one row of an INSERT's returning value.
149    fn insert_returning_record_mut(
150        returning: &mut Option<stmt::Returning>,
151        index: usize,
152        preserve_returning_projection: bool,
153    ) -> Option<&mut stmt::ExprRecord> {
154        match returning {
155            Some(stmt::Returning::Expr(stmt::Expr::List(rows)))
156                if !preserve_returning_projection =>
157            {
158                Some(rows.items[index].as_record_mut_unwrap())
159            }
160            Some(stmt::Returning::Expr(stmt::Expr::Record(record)))
161                if !preserve_returning_projection =>
162            {
163                Some(record)
164            }
165            Some(stmt::Returning::Project(stmt::Expr::Record(record)))
166                if preserve_returning_projection =>
167            {
168                Some(record)
169            }
170            _ => None,
171        }
172    }

174    /// Return whether the row has no usable foreign key for this relation.
175    fn belongs_to_fk_is_unset(row: &stmt::Expr, rel: &app::BelongsTo) -> bool {
176        rel.foreign_key.fields.iter().any(|fk_field| {
177            row.entry(fk_field.source.index).is_none_or(|entry| {
178                let expr = entry.to_expr();
179                expr.is_value_null() || expr.is_default()
180            })
181        })
182    }

184    /// Build the expression that loads one eager `belongs_to` relation.
185    fn plan_insert_belongs_to_load(
186        &mut self,
187        field_index: usize,
188        row_index: usize,
189        preserve_returning_projection: bool,
190    ) -> stmt::Expr {
191        // A preserved projection is shared across rows and cannot hold
192        // different subqueries for different VALUES rows.
193        assert!(
194            !preserve_returning_projection || row_index == 0,
195            "eager belongs_to in a multi-row insert with a preserved returning projection"
196        );
197
198        let load = self.build_relation_subquery(field_index);
199        self.order_relation_load_after_enclosing_inserts(&load);
200        Self::single_relation_from_load(load)
201    }
202
203    /// Make a relation load wait for the database writes that can create its row.
204    fn order_relation_load_after_enclosing_inserts(&mut self, load: &stmt::Expr) {
205        let stmt::Expr::Arg(expr_arg) = load else {
206            unreachable!("belongs_to subquery lowers to a sub-statement arg");
207        };
208        let crate::engine::hir::Arg::Sub {
209            stmt_id: sub_id, ..
210        } = self.curr_stmt_info().args[expr_arg.position]
211        else {
212            unreachable!("subquery arg refers to a sub-statement");
213        };
214
215        let state = &mut *self.state;
216        let (insert_stmts, hir) = (&state.insert_stmts, &mut state.hir);
217        for &target in insert_stmts {
218            hir[sub_id].add_dep(target, crate::engine::hir::DepKind::Effect);
219        }
220    }

222    /// Convert a relation query's row list into one nullable record.
223    fn single_relation_from_load(load: stmt::Expr) -> stmt::Expr {
224        stmt::Expr::match_expr(
225            load.clone(),
226            vec![stmt::MatchArm {
227                pattern: stmt::Value::List(vec![]),
228                expr: stmt::Expr::null(),
229            }],
230            stmt::Expr::project(load, [0usize]),
231        )
232    }

234    // Checks all fields of a record and handles nulls
235    fn apply_app_level_insertion_defaults(
236        &mut self,
237        model: &app::ModelRoot,
238        expr: &mut stmt::Expr,
239        set_fields: &mut BitSet<usize>,
240    ) {
241        // First, we pad the record to account for all fields
242        if let stmt::Expr::Record(expr_record) = expr {
243            // TODO: get rid of this
244            assert_eq!(expr_record.len(), model.fields.len());
245        }
246
247        // Next, we have to find all belongs-to fields and normalize them to FK
248        // values
249        for field in &model.fields {
250            if let app::FieldTy::BelongsTo(rel) = &field.ty {
251                let mut field_expr = expr.entry_mut(field.id.index);
252
253                if field_expr.is_value_null() {
254                    continue;
255                }
256
257                let fk_fields = &rel.foreign_key.fields;
258
259                // Distribute the BelongsTo value into its FK source column(s):
260                //   - composite FK (`fk_fields.len() > 1`): the value is a
261                //     record (`Expr::Record` from the tuple `IntoExpr` for
262                //     composite-PK targets, or `Expr::Value(Value::Record)`
263                //     from a literal); one item per FK column via
264                //     `into_record_items`.
265                //   - single FK: the value goes whole into the single FK
266                //     column, whether it is a scalar or a record (an
267                //     `Embed`-typed PK produces an `Expr::Record` whose shape
268                //     matches the embed-typed FK source).
269                if fk_fields.len() > 1 && field_expr.is_record() {
270                    let items = field_expr.take().into_record_items().unwrap();
271                    let mut count = 0;
272                    for (fk_field, item) in fk_fields.iter().zip(items) {
273                        expr.entry_mut(fk_field.source.index).insert(item);
274                        count += 1;
275                    }
276                    assert_eq!(count, fk_fields.len());
277                } else {
278                    if !field_expr.is_value() && !field_expr.is_record() {
279                        continue;
280                    }
281                    let [fk_field] = &fk_fields[..] else {
282                        panic!("composite FK expected a record value, got scalar");
283                    };
284                    let e = field_expr.take();
285                    expr.entry_mut(fk_field.source.index).insert(e);
286                }
287            }
288        }
289
290        // Initialize version fields to 1 if not already set by the user.
291        // For embedded newtypes (e.g. `Version(u64)`), wrap in one Record per
292        // embed layer so the value round-trips through the embed's `Load` impl.
293        for field in &model.fields {
294            if field.is_versionable() {
295                let mut field_expr = expr.entry_mut(field.id.index);
296                if field_expr.is_default() || field_expr.is_value_null() {
297                    let target = self.auto_target(field.id);
298                    field_expr.insert(target.wrap(stmt::Value::U64(1)).into());
299                }
300            }
301        }
302
303        // We have to handle auto fields first because they are often the
304        // identifier which may be referenced to handle associations.
305        for field in &model.fields {
306            let mut field_expr = expr.entry_mut(field.id.index);
307
308            if field_expr.is_default() {
309                // If the field is defined to be auto-populated, then populate
310                // it here.
311                if let Some(auto) = &field.auto {
312                    // For an embedded newtype, the auto value is generated as
313                    // the inner primitive and wrapped in a single-element
314                    // record so it round-trips through the embed `Load` impl.
315                    let target = self.auto_target(field.id);
316                    match auto {
317                        app::AutoStrategy::Uuid(version) => {
318                            let id = match version {
319                                app::UuidVersion::V4 => uuid::Uuid::new_v4(),
320                                app::UuidVersion::V7 => uuid::Uuid::now_v7(),
321                            };
322                            let primitive = match &target.ty {
323                                stmt::Type::String => stmt::Value::String(id.to_string()),
324                                stmt::Type::Uuid => stmt::Value::Uuid(id),
325                                other => panic!(
326                                    "auto-generated UUID cannot be inserted into column of type {other:?}"
327                                ),
328                            };
329                            field_expr.insert(target.wrap(primitive).into());
330                        }
331                        app::AutoStrategy::Increment => {
332                            // Leave value as `Expr::Default` for primitives so
333                            // the database fills it in. For embedded newtypes
334                            // the column-projection step needs a Record of the
335                            // right shape (one wrapping per nested embed) so
336                            // it can extract `Default` per column.
337                            if target.wrap_depth > 0 {
338                                field_expr.insert(target.wrap_expr(stmt::Expr::Default));
339                            }
340                        }
341                    }
342                }
343            }
344
345            if !field_expr.is_value_null() {
346                set_fields.insert(field.id.index);
347            }
348        }
349    }

351    fn convert_returning_for_insert(
352        &mut self,
353        values: &stmt::Values,
354        returning: &mut Option<stmt::Returning>,
355        single: bool,
356    ) {
357        // If there is no returning statement, there is nothing to convert
358        let Some(stmt::Returning::Project(projection)) = returning else {
359            return;
360        };
361
362        #[derive(Debug)]
363        struct Input(usize);
364
365        impl stmt::Input for Input {
366            fn resolve_arg(
367                &mut self,
368                expr_arg: &stmt::ExprArg,
369                projection: &stmt::Projection,
370            ) -> Option<stmt::Expr> {
371                let _ = (expr_arg, projection);
372                None
373            }
374
375            fn resolve_ref(
376                &mut self,
377                expr_reference: &stmt::ExprReference,
378                projection: &stmt::Projection,
379            ) -> Option<stmt::Expr> {
380                let expr_column = expr_reference.as_expr_column()?;
381
382                assert!(
383                    expr_column.nesting == 0 && expr_column.table == 0,
384                    "expr_reference={expr_reference:#?}"
385                );
386                assert!(projection.is_identity(), "TODO");
387
388                Some(stmt::Expr::project(*expr_reference, self.0))
389            }
390        }
391
392        let mut converted = vec![];
393
394        for i in 0..values.rows.len() {
395            let mut converted_row = projection.clone();
396            converted_row.substitute(Input(i));
397            converted.push(converted_row);
398        }
399
400        *returning = Some(stmt::Returning::Expr(if single {
401            assert!(converted.len() == 1);
402            converted.into_iter().next().unwrap()
403        } else {
404            stmt::Expr::list_from_vec(converted)
405        }));
406    }
407
408    fn verify_field_constraints(&mut self, model: &app::ModelRoot, expr: &mut stmt::Expr) {
409        for field in &model.fields {
410            if field.nullable && field.constraints.is_empty() {
411                continue;
412            }
413
414            let field_expr = expr.entry(field.id.index).unwrap();
415
416            if !field.nullable && field_expr.is_value_null() {
417                // Relations are handled differently
418                if !field.ty.is_relation() && field.auto.is_none() {
419                    self.state
420                        .errors
421                        .push(toasty_core::Error::validation_failed(format!(
422                            "insert missing non-nullable field `{}` in model `{}`",
423                            field.name,
424                            model.name.upper_camel_case()
425                        )));
426                }
427            }
428
429            for constraint in &field.constraints {
430                if let Err(err) = constraint.check(&field_expr) {
431                    self.state.errors.push(err);
432                }
433            }
434        }
435    }

438impl<'b> LowerStatement<'_, 'b> {
439    /// Returns the primitive type the auto value should be encoded as, plus
440    /// how many record wrappings to apply for the embed chain. Walks down
441    /// through nested newtype embeds (`Outer(Inner(u64))`) until it reaches
442    /// the leaf primitive.
443    fn auto_target(&self, field_id: app::FieldId) -> AutoTarget<'b> {
444        let mut ty = &self.schema().app.field(field_id).ty;
445        let mut wrap_depth = 0;
446        loop {
447            match ty {
448                app::FieldTy::Primitive(primitive) => {
449                    return AutoTarget {
450                        ty: &primitive.ty,
451                        wrap_depth,
452                    };
453                }
454                app::FieldTy::Embedded(embedded) => {
455                    let target = self.schema().app.model(embedded.target);
456                    let app::Model::EmbeddedStruct(es) = target else {
457                        panic!(
458                            "#[auto] on embedded enum is not supported (target {:?})",
459                            embedded.target
460                        );
461                    };
462                    let [inner] = es.fields.as_slice() else {
463                        panic!(
464                            "#[auto] on embedded type with {} fields; expected exactly one",
465                            es.fields.len()
466                        );
467                    };
468                    ty = &inner.ty;
469                    wrap_depth += 1;
470                }
471                _ => panic!("#[auto] not allowed on non-primitive fields"),
472            }
473        }
474    }
475
476    /// The target projection and per-row delta for a versionable field's update
477    /// assignment, used to build an atomic `version = version + 1` via
478    /// [`Assignment::Add`](stmt::Assignment::Add).
479    ///
480    /// The projection walks past any embed-newtype layers (`Version(u64)`) to
481    /// the leaf primitive column — `[field, 0, …]` — and the delta is the bare
482    /// literal `1`. Arithmetic operators apply only to primitive columns, so
483    /// unlike the whole-record `Set` the instance path uses, the increment must
484    /// reach the leaf rather than sit on the embed field.
485    pub(super) fn version_increment_target(
486        &self,
487        field_id: app::FieldId,
488    ) -> (stmt::Projection, stmt::Expr) {
489        let target = self.auto_target(field_id);
490
491        // `[field_id.index]` then one `0` per embed layer, reaching the leaf.
492        let mut steps = vec![field_id.index];
493        steps.resize(1 + target.wrap_depth, 0);
494
495        (
496            stmt::Projection::from(steps.as_slice()),
497            stmt::Expr::Value(stmt::Value::U64(1)),
498        )
499    }
500
501    /// Adds an atomic version increment unless the statement already assigns
502    /// the version field.
503    pub(super) fn inject_version_increment(
504        &self,
505        assignments: &mut stmt::Assignments,
506    ) -> Option<stmt::Projection> {
507        let version_id = self.model().and_then(|m| m.version_field()).map(|f| f.id)?;
508
509        let already_assigned = assignments
510            .keys()
511            .any(|projection| projection.as_slice().first() == Some(&version_id.index));
512        if already_assigned {
513            return None;
514        }
515
516        let (projection, delta) = self.version_increment_target(version_id);
517        assignments.add(projection.clone(), delta);
518        Some(projection)
519    }
520}

522impl ApplyInsertScope<'_> {
523    fn apply_expr(&mut self, stmt: &stmt::Expr) {
524        match stmt {
525            stmt::Expr::And(exprs) => {
526                for expr in exprs {
527                    self.apply_expr(expr);
528                }
529            }
530            stmt::Expr::BinaryOp(e) if e.op.is_eq() => match (&*e.lhs, &*e.rhs) {
531                (
532                    stmt::Expr::Reference(expr_ref @ stmt::ExprReference::Field { .. }),
533                    rhs @ stmt::Expr::Value(..),
534                ) => {
535                    self.apply_eq_constraint(expr_ref, rhs);
536                }
537                (
538                    lhs @ stmt::Expr::Value(..),
539                    stmt::Expr::Reference(expr_ref @ stmt::ExprReference::Field { .. }),
540                ) => {
541                    self.apply_eq_constraint(expr_ref, lhs);
542                }
543                (
544                    lhs_expr @ stmt::Expr::Reference(
545                        lhs @ stmt::ExprReference::Field {
546                            nesting: nesting_lhs,
547                            ..
548                        },
549                    ),
550                    rhs_expr @ stmt::Expr::Reference(
551                        rhs @ stmt::ExprReference::Field {
552                            nesting: nesting_rhs,
553                            ..
554                        },
555                    ),
556                ) => match (nesting_lhs, nesting_rhs) {
557                    (0, _) if *nesting_rhs > 0 => self.apply_eq_constraint(lhs, rhs_expr),
558                    (_, 0) if *nesting_lhs > 0 => self.apply_eq_constraint(rhs, lhs_expr),
559                    _ => panic!("exactly one field must reference parent"),
560                },
561                _ => todo!("EXPR = {:#?}", stmt),
562            },
563            // Constants are ignored
564            stmt::Expr::Value(_) => {}
565            _ => todo!("EXPR = {:#?}", stmt),
566        }
567    }
568
569    fn apply_eq_constraint(&mut self, expr_ref: &stmt::ExprReference, val: &stmt::Expr) {
570        let stmt::ExprReference::Field { nesting, index } = expr_ref else {
571            todo!("handle non-field reference");
572        };
573
574        assert!(*nesting == 0, "TODO: handle references to parent scopes");
575
576        let mut existing = self.expr.entry_mut(*index);
577
578        if !existing.is_value_null() && !existing.is_default() {
579            if let stmt::EntryMut::Value(existing) = existing {
580                if let stmt::Expr::Value(val) = val {
581                    assert_eq!(existing, val);
582                } else {
583                    todo!()
584                }
585            } else {
586                todo!()
587            }
588        } else {
589            existing.insert(val.clone());
590        }
591    }
