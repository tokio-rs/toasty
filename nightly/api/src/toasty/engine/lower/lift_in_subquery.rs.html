1//! App-level rewrites that lift relation references out of `IN` subqueries
2//! into direct foreign-key comparisons.
3//!
4//! [`LiftInSubquery`] runs as a whole-statement pre-pass before the main
5//! lowering walk.  The visitor overrides `visit_expr_mut` to fire two
6//! rewrites pre-children:
7//!
8//! - [`lift_in_subquery`] fires on `Expr::InSubquery` where the LHS is a
9//!   relation field reference (`BelongsTo`/`HasOne`/`HasMany`).  For
10//!   `BelongsTo` it tries to lift the subquery's filter into FK comparisons
11//!   on the parent, falling back to a re-targeted IN subquery on the
12//!   foreign-key column.  For `HasOne`/`HasMany` it always rewrites to a
13//!   foreign-key IN subquery against the related table.
14//!
15//! - [`try_lift_relation_path_comparison`] fires on `Expr::BinaryOp` where
16//!   one side walks a relation field. For example, filtering profiles by
17//!   their user's name (`profile.user.name = 'alice'`) rewrites to:
18//!
19//!   ```text
20//!   Profile.user_id IN (SELECT User.id FROM User WHERE User.name = 'alice')
21//!   ```
22//!
23//! - [`try_lift_relation_path_like`] does the same for `LIKE`/`ILIKE`
24//!   (`profile.user.name LIKE 'al%'`). These are `Expr::Like`, not
25//!   `Expr::BinaryOp`, so they need their own entry point. Both rewrites
26//!   share [`lift_relation_path_predicate`].
27//!
28//! A pre-pass is necessary (rather than folding into
29//! `LowerStatement::visit_expr_mut` per #823's pattern) because not every
30//! expression that contains an `IN` subquery flows through
31//! `LowerStatement::visit_expr_mut`; `ApplyInsertScope::apply_expr` in
32//! particular walks insert-scope constraint expressions through its own
33//! recursion and would panic on an unlifted relation `IN` subquery.
34//!
35//! The free functions [`lift_in_subquery`] and
36//! [`try_lift_relation_path_comparison`] are exposed on `&ExprContext` so
37//! the visitor and the unit tests can both call them without constructing
38//! a `LiftInSubquery`.
39
40use toasty_core::{
41    schema::app::{self, BelongsTo, FieldId, FieldTy, ModelId},
42    stmt::{self, Expr, ExprContext, IntoExprTarget, ResolvedRef, Visit, VisitMut},
43};

45/// Pre-lowering pass that lifts relation references out of `IN`-subquery
46/// and projection comparisons into direct foreign-key forms.  Runs as a
47/// whole-statement visitor before the main lowering walk: code paths that
48/// process expressions outside the lowering walk (notably
49/// `ApplyInsertScope::apply_expr`) see the already-lifted form.
50pub(super) struct LiftInSubquery<'a> {
51    cx: ExprContext<'a>,
52    exclude_nulls: bool,
53}
54
55impl<'a> LiftInSubquery<'a> {
56    pub(super) fn new(cx: ExprContext<'a>, exclude_nulls: bool) -> Self {
57        Self { cx, exclude_nulls }
58    }
59
60    pub(super) fn rewrite(&mut self, stmt: &mut stmt::Statement) {
61        self.visit_mut(stmt);
62    }
63
64    fn scope<'scope>(&'scope self, target: impl IntoExprTarget<'scope>) -> LiftInSubquery<'scope> {
65        LiftInSubquery {
66            cx: self.cx.scope(target),
67            exclude_nulls: self.exclude_nulls,
68        }
69    }
70
71    fn exclude_nulls(&self, expr: &mut stmt::Expr) {
72        if !self.exclude_nulls {
73            return;
74        }
75
76        let stmt::Expr::InSubquery(expr) = expr else {
77            return;
78        };
79        let select = expr.query.body.as_select_mut_unwrap();
80        let target = select.source.model_id_unwrap();
81        let returning = select.returning.as_project_unwrap().clone();
82
83        for field in returning
84            .as_record()
85            .map_or(std::slice::from_ref(&returning), |record| &record.fields)
86        {
87            let stmt::Expr::Reference(stmt::ExprReference::Field { index, .. }) = field else {
88                unreachable!();
89            };
90            if self.cx.schema().app.field(target.field(*index)).nullable {
91                select.add_filter(stmt::Expr::is_not_null(field.clone()));
92            }
93        }
94    }
95}

97impl VisitMut for LiftInSubquery<'_> {
98    fn visit_expr_mut(&mut self, expr: &mut stmt::Expr) {
99        // Apply lifts pre-children: the rewrites pattern-match on app-
100        // level `Reference::Field` to a relation, and the children walk
101        // would not introduce relation references that were not there.
102        match expr {
103            stmt::Expr::InSubquery(e) => {
104                if let Some(mut lifted) = lift_in_subquery(&self.cx, &e.expr, &e.query) {
105                    self.exclude_nulls(&mut lifted);
106                    *expr = lifted;
107                }
108            }
109            stmt::Expr::BinaryOp(e) => {
110                if let Some(mut lifted) =
111                    try_lift_relation_path_comparison(&self.cx, e.op, &e.lhs, &e.rhs)
112                {
113                    self.exclude_nulls(&mut lifted);
114                    *expr = lifted;
115                } else if let Some(commuted) = e.op.commute()
116                    && let Some(mut lifted) =
117                        try_lift_relation_path_comparison(&self.cx, commuted, &e.rhs, &e.lhs)
118                {
119                    self.exclude_nulls(&mut lifted);
120                    *expr = lifted;
121                }
122            }
123            stmt::Expr::Like(e) => {
124                if let Some(mut lifted) = try_lift_relation_path_like(&self.cx, e) {
125                    self.exclude_nulls(&mut lifted);
126                    *expr = lifted;
127                }
128            }
129            _ => {}
130        }
131
132        // Walk children (which may themselves be expressions needing
133        // lifts on subtrees).
134        stmt::visit_mut::visit_expr_mut(self, expr);
135    }
136
137    fn visit_stmt_delete_mut(&mut self, stmt: &mut stmt::Delete) {
138        self.visit_source_mut(&mut stmt.from);
139
140        let mut s = self.scope(&stmt.from);
141
142        s.visit_filter_mut(&mut stmt.filter);
143
144        if let Some(returning) = &mut stmt.returning {
145            s.visit_returning_mut(returning);
146        }
147    }
148
149    fn visit_stmt_insert_mut(&mut self, stmt: &mut stmt::Insert) {
150        self.visit_insert_target_mut(&mut stmt.target);
151
152        let mut s = self.scope(&stmt.target);
153
154        s.visit_stmt_query_mut(&mut stmt.source);
155
156        if let Some(returning) = &mut stmt.returning {
157            s.visit_returning_mut(returning);
158        }
159    }
160
161    fn visit_stmt_select_mut(&mut self, stmt: &mut stmt::Select) {
162        self.visit_source_mut(&mut stmt.source);
163
164        let mut s = self.scope(&stmt.source);
165
166        s.visit_filter_mut(&mut stmt.filter);
167        s.visit_returning_mut(&mut stmt.returning);
168    }
169
170    fn visit_stmt_update_mut(&mut self, stmt: &mut stmt::Update) {
171        self.visit_update_target_mut(&mut stmt.target);
172
173        let mut s = self.scope(&stmt.target);
174
175        s.visit_assignments_mut(&mut stmt.assignments);
176        s.visit_filter_mut(&mut stmt.filter);
177
178        if let Some(expr) = &mut stmt.condition.expr {
179            s.visit_expr_mut(expr);
180        }
181
182        if let Some(returning) = &mut stmt.returning {
183            s.visit_returning_mut(returning);
184        }
185    }
186}

197/// Lift `expr IN (subquery)` into a foreign-key-based comparison when
198/// `expr` is a relation field reference and `subquery` targets the
199/// relation's target model.
200///
201/// Returns `None` when the LHS is not a relation field reference or when
202/// the lift cannot apply.  When the lift succeeds, the returned
203/// expression may itself contain unlowered references and the caller is
204/// expected to re-visit it through the lowering walk.
205pub(super) fn lift_in_subquery(
206    cx: &ExprContext,
207    expr: &stmt::Expr,
208    query: &stmt::Query,
209) -> Option<stmt::Expr> {
210    // The expression is a path expression referencing a relation.
211    let field = match expr {
212        // `Project(Ref(rel), [head, ...tail])` — the path traverses through a
213        // relation field (`rel`) before reaching the relation the subquery
214        // targets. Re-root the path at `rel`'s target model and rebuild as a
215        // nested IN-subquery on that model, then recurse to lift the outer
216        // `rel` hop.
217        stmt::Expr::Project(project_expr) => {
218            return lift_projection_in_subquery(cx, project_expr, query);
219        }
220        stmt::Expr::Reference(expr_reference @ stmt::ExprReference::Field { .. }) => {
221            cx.resolve_expr_reference(expr_reference).as_field_unwrap()
222        }
223        _ => {
224            return None;
225        }
226    };
227
228    // If the field is not a relation, abort. Direct relations lift through
229    // their paired foreign keys. A `via` has no single pair, so normalize its
230    // relation chain into a projected path and use the projection lift.
231    match &field.ty {
232        FieldTy::BelongsTo(belongs_to) => lift_belongs_to_in_subquery(cx, belongs_to, query),
233        FieldTy::Has(has) => lift_has_n_in_subquery(has.target, has.pair(&cx.schema().app), query),
234        FieldTy::Via(via) => lift_via_in_subquery(cx, via, query),
235        _ => None,
236    }
237}

239/// Lift an `IN` subquery whose left side names a `via` relation.
240///
241/// For `User.groups.any(Group.name == "Rust")`, where `groups` follows
242/// `User.memberships.group`, this builds:
243///
244/// ```text
245/// User.id IN (
246///     SELECT Membership.user_id FROM Membership
247///     WHERE Membership.group_id IN (
248///         SELECT Group.id FROM Group WHERE Group.name == "Rust"
249///     )
250/// )
251/// ```
252///
253/// The path is expanded into direct relation fields, converted to the same
254/// projected expression produced by an explicit relation chain, and then
255/// handled by the regular projection lift.
256fn lift_via_in_subquery(
257    cx: &ExprContext,
258    via: &app::Via,
259    query: &stmt::Query,
260) -> Option<stmt::Expr> {
261    if via.is_scalar() {
262        return None;
263    }
264
265    let fields = super::relation_path::flatten_via_path(cx.schema(), via)?;
266    let target = fields
267        .last()
268        .and_then(|field| cx.schema().app.field(*field).relation_target_id())?;
269
270    if target != via.target || target != query.body.as_select_unwrap().source.model_id_unwrap() {
271        return None;
272    }
273
274    let (base, projection) = fields.split_first()?;
275    let base = stmt::Expr::ref_self_field(*base);
276    let path = if projection.is_empty() {
277        base
278    } else {
279        let projection = projection
280            .iter()
281            .map(|field| field.index)
282            .collect::<Vec<_>>();
283        stmt::Expr::project(base, projection.as_slice())
284    };
285
286    lift_in_subquery(cx, &path, query)
287}

289/// Lifts an `IN`-subquery whose left side is a path through a relation field.
290///
291/// `.any()` on a relation chain — for example, `Release.project.topics.any(name
292/// == "rust")` — builds an `InSubquery` whose LHS projects through the
293/// chain:
294///
295/// ```text
296/// InSubquery {
297///     expr:  Project(Ref(Release.project), [Project.topics_idx]),
298///     query: SELECT FROM Topic WHERE name == "rust",
299/// }
300/// ```
301///
302/// Two paths handle this:
303///
304/// **Fused.** [`try_fuse_paired_relations`] recognizes the common
305/// `BelongsTo → Has` chain over a shared primary key and emits a single
306/// FK-on-FK `IN` that bypasses the intermediate model:
307///
308/// ```text
309/// Release.project_id IN (SELECT Topic.project_id FROM Topic WHERE name == "rust")
310/// ```
311///
312/// **General.** For chains the fast path doesn't match (multi-hop
313/// projections, non-paired relations), re-root the projection at the
314/// relation's target model and wrap the original subquery in a nested `IN`
315/// against that model, then recurse on the outer hop. For the same input,
316/// the fallback produces:
317///
318/// ```text
319/// Release.project IN (
320///     SELECT FROM Project
321///     WHERE Project.topics IN (SELECT FROM Topic WHERE name == "rust")
322/// )
323/// ```
324///
325/// The recursive call lifts the outer `Release.project` hop via the standard
326/// `BelongsTo` branch; the inner `Project.topics` hop is lifted on the next
327/// visitor pass when [`LiftInSubquery`]'s children walk reaches it.
328fn lift_projection_in_subquery(
329    cx: &ExprContext,
330    project_expr: &stmt::ExprProject,
331    query: &stmt::Query,
332) -> Option<stmt::Expr> {
333    let Expr::Reference(expr_ref) = &*project_expr.base else {
334        return None;
335    };
336    let ResolvedRef::Field(field) = cx.resolve_expr_reference(expr_ref) else {
337        return None;
338    };
339
340    let target_model_id = field.relation_target_id()?;
341
342    let (head_idx, tail) = project_expr.projection.as_slice().split_first()?;
343
344    if tail.is_empty()
345        && let Some(direct) =
346            try_fuse_paired_relations(cx, field, target_model_id, *head_idx, query)
347    {
348        return Some(direct);
349    }
350
351    let target_field = Expr::ref_self_field(FieldId {
352        model: target_model_id,
353        index: *head_idx,
354    });
355    let inner_lhs = if tail.is_empty() {
356        target_field
357    } else {
358        Expr::project(target_field, stmt::Projection::from(tail))
359    };
360
361    let new_subquery = stmt::Query::new_select(
362        stmt::Source::from(target_model_id),
363        Expr::in_subquery(inner_lhs, query.clone()),
364    );
365
366    lift_in_subquery(cx, &project_expr.base, &new_subquery)
367}

369/// Fuses a `BelongsTo → Has` chain into a single FK-on-FK `IN` when both
370/// relations meet at the same primary key.
371///
372/// # Example
373///
374/// Given the schema:
375///
376/// ```text
377/// Todo     { category_id: Category.id, ... }   // BelongsTo  Todo.category
378/// Category { todos: HasMany Todo, ... }        // Has, paired with Todo.category
379/// ```
380///
381/// The path `Todo.category.todos` lifts as follows:
382///
383/// ```text
384/// // Input
385/// InSubquery {
386///     expr:  Project(Ref(Todo.category), [Category.todos_idx]),
387///     query: SELECT FROM Todo WHERE title == "salad",
388/// }
389///
390/// // Output
391/// InSubquery {
392///     expr:  Ref(Todo.category_id),
393///     query: SELECT Todo.category_id FROM Todo WHERE title == "salad",
394/// }
395/// ```
396///
397/// The outer relation's FK source columns become the LHS; the inner
398/// relation's paired-BelongsTo FK source columns become the subquery's
399/// returning list. The user's original filter is preserved verbatim.
400///
401/// # Why the fusion is sound
402///
403/// Composing the two hops without fusion routes through the intermediate
404/// model:
405///
406/// ```text
407/// Todo.category_id IN (
408///     SELECT Category.id FROM Category
409///     WHERE Category.id IN (SELECT Todo.category_id FROM Todo WHERE title == "salad")
410/// )
411/// ```
412///
413/// Both FKs target `Category.id`, so every value the innermost subquery
414/// returns exists in `Category.id` under FK integrity (which the engine
415/// already assumes). The middle filter therefore admits every row the inner
416/// returns, and the middle `SELECT Category.id` simply re-emits them. The
417/// `Category` scan is a no-op and can be dropped.
418///
419/// Returns `None` when the chain doesn't match the pattern — outer isn't a
420/// `BelongsTo`, inner isn't a `Has`, or the FK columns don't line up.
421fn try_fuse_paired_relations(
422    cx: &ExprContext,
423    outer_field: &app::Field,
424    target_model_id: ModelId,
425    head_idx: usize,
426    query: &stmt::Query,
427) -> Option<stmt::Expr> {
428    let outer_belongs_to = match &outer_field.ty {
429        FieldTy::BelongsTo(rel) => rel,
430        _ => return None,
431    };
432
433    let target_model = cx.schema().app.model(target_model_id).as_root_unwrap();
434    let head_field = target_model.fields.get(head_idx)?;
435    let inner_has = match &head_field.ty {
436        FieldTy::Has(has) => has,
437        _ => return None,
438    };
439    let inner_pair = inner_has.pair(&cx.schema().app);
440
441    // Both FKs must reference the same PK columns in the same order.
442    if outer_belongs_to.foreign_key.fields.len() != inner_pair.foreign_key.fields.len() {
443        return None;
444    }
445    for (outer_fk, inner_fk) in outer_belongs_to
446        .foreign_key
447        .fields
448        .iter()
449        .zip(inner_pair.foreign_key.fields.iter())
450    {
451        if outer_fk.target != inner_fk.target {
452            return None;
453        }
454    }
455
456    lift_fk_in_subquery(
457        inner_has.target,
458        super::key_field_refs(
459            0,
460            outer_belongs_to
461                .foreign_key
462                .fields
463                .iter()
464                .map(|fk| fk.source),
465        ),
466        super::key_field_refs(0, inner_pair.foreign_key.fields.iter().map(|fk| fk.source)),
467        query,
468    )
469}

471/// Rewrites a comparison that walks a relation field into a foreign-key
472/// subquery.
473///
474/// `Profile::filter(Profile::fields().user().name().eq("alice"))` starts as
475/// the filter `profile.user.name = 'alice'`, where the left side projects
476/// through the `user` relation. This moves the comparison into a subquery on
477/// the target model and defers to [`lift_in_subquery`]:
478///
479/// ```text
480/// Profile.user_id IN (SELECT User.id FROM User WHERE User.name = 'alice')
481/// ```
482///
483/// Returns `None` when `project_side` does not walk a relation field.
484pub(super) fn try_lift_relation_path_comparison(
485    cx: &ExprContext,
486    op: stmt::BinaryOp,
487    project_side: &stmt::Expr,
488    other_side: &stmt::Expr,
489) -> Option<stmt::Expr> {
490    lift_relation_path_predicate(cx, project_side, |target_lhs| {
491        Expr::binary_op(target_lhs, op, other_side.clone())
492    })
493}

495/// Rewrites a `LIKE`/`ILIKE` that walks a relation field into a foreign-key
496/// subquery — [`try_lift_relation_path_comparison`] for pattern matches.
497///
498/// `Profile::filter(Profile::fields().user().name().like("al%"))` rewrites to:
499///
500/// ```text
501/// Profile.user_id IN (SELECT User.id FROM User WHERE User.name LIKE 'al%')
502/// ```
503///
504/// `LIKE`/`ILIKE` are `Expr::Like`, not `Expr::BinaryOp`, so they never reach
505/// [`try_lift_relation_path_comparison`] and need this entry point. Without
506/// it the `user.name` path stays a projection through the relation, which the
507/// rest of lowering cannot turn into a column and so panics.
508///
509/// Returns `None` when the pattern's subject does not walk a relation field.
510pub(super) fn try_lift_relation_path_like(
511    cx: &ExprContext,
512    like: &stmt::ExprLike,
513) -> Option<stmt::Expr> {
514    lift_relation_path_predicate(cx, &like.expr, |target_lhs| {
515        stmt::ExprLike {
516            expr: Box::new(target_lhs),
517            pattern: like.pattern.clone(),
518            escape: like.escape,
519            case_insensitive: like.case_insensitive,
520        }
521        .into()
522    })
523}

525/// Shared core of the relation-path lifts.
526///
527/// `project_side` is the side of the predicate that walks a relation — the
528/// `profile.user.name` in `profile.user.name = 'alice'`. It is a projection
529/// whose first step names the `user` relation field on `Profile` and whose
530/// remaining steps index into `User`. This re-roots the path at the target
531/// model (so it reads `User.name`), builds a `SELECT` over `User` whose filter
532/// `make_filter` produces from the re-rooted path, and defers to
533/// [`lift_in_subquery`] to turn the relation reference into the foreign-key
534/// `IN` form.
535///
536/// `make_filter` is the only difference between callers: a binary op for
537/// comparisons, a `LIKE` for pattern matches.
538///
539/// Returns `None` when `project_side` does not walk a relation field.
540fn lift_relation_path_predicate(
541    cx: &ExprContext,
542    project_side: &stmt::Expr,
543    make_filter: impl FnOnce(stmt::Expr) -> stmt::Expr,
544) -> Option<stmt::Expr> {
545    let Expr::Project(project_expr) = project_side else {
546        return None;
547    };
548    let Expr::Reference(expr_ref) = &*project_expr.base else {
549        return None;
550    };
551    let ResolvedRef::Field(field) = cx.resolve_expr_reference(expr_ref) else {
552        return None;
553    };
554
555    let target_model_id = field.relation_target_id()?;
556
557    // The first step names the relation field on the source model; the
558    // rest index into the target model. Drop the first step and re-root the
559    // remainder at the target model.
560    let (head_idx, tail) = project_expr.projection.as_slice().split_first()?;
561    let target_field = Expr::ref_self_field(FieldId {
562        model: target_model_id,
563        index: *head_idx,
564    });
565    let target_lhs = if tail.is_empty() {
566        target_field
567    } else {
568        Expr::project(target_field, stmt::Projection::from(tail))
569    };
570
571    let subquery =
572        stmt::Query::new_select(stmt::Source::from(target_model_id), make_filter(target_lhs));
573
574    lift_in_subquery(cx, &project_expr.base, &subquery)
575}

577/// Build a foreign-key `IN` subquery for one direct relation edge.
578///
579/// `lhs` references key fields on the current model. `returning` references
580/// the matching key fields on `target`, which is also the subquery source.
581fn lift_fk_in_subquery(
582    target: ModelId,
583    lhs: stmt::Expr,
584    returning: stmt::Expr,
585    query: &stmt::Query,
586) -> Option<stmt::Expr> {
587    if target != query.body.as_select_unwrap().source.model_id_unwrap() {
588        return None;
589    }
590
591    let mut subquery = query.clone();
592    subquery.body.as_select_mut_unwrap().returning = stmt::Returning::Project(returning);
593
594    Some(stmt::Expr::in_subquery(lhs, subquery))
595}

597/// BelongsTo branch: try to lift the subquery's filter into direct FK
598/// comparisons.  When the filter references only FK-mapped fields, return
599/// the AND of per-FK equalities.  Otherwise, fall back to an IN subquery
600/// on the foreign-key column(s) — a tuple-form IN for composite FKs.
601///
602/// Returns `None` when the subquery does not target the BelongsTo's
603/// target model.
604fn lift_belongs_to_in_subquery(
605    cx: &ExprContext,
606    belongs_to: &BelongsTo,
607    query: &stmt::Query,
608) -> Option<stmt::Expr> {
609    if belongs_to.target != query.body.as_select_unwrap().source.model_id_unwrap() {
610        return None;
611    }
612
613    let select = query.body.as_select_unwrap();
614
615    let mut lift = LiftBelongsTo {
616        cx: cx.scope(&select.source),
617        belongs_to,
618        fk_field_matches: vec![false; belongs_to.foreign_key.fields.len()],
619        operands: vec![],
620        fail: false,
621    };
622
623    lift.visit_filter(&select.filter);
624
625    // Fall back to the IN-subquery form whenever we couldn't account for
626    // every FK column with a direct equality from the filter. This covers
627    // both `fail=true` (a binary op referenced something that isn't on the
628    // FK) and the case where the filter contained no liftable binary ops at
629    // all — e.g. when it's a nested `InSubquery` that the LiftBelongsTo
630    // visitor deliberately skips (see `visit_expr_in_subquery`).
631    let all_fks_matched = lift.fk_field_matches.iter().all(|m| *m);
632
633    if lift.fail || !all_fks_matched {
634        lift_fk_in_subquery(
635            belongs_to.target,
636            super::key_field_refs(0, belongs_to.foreign_key.fields.iter().map(|fk| fk.source)),
637            super::key_field_refs(0, belongs_to.foreign_key.fields.iter().map(|fk| fk.target)),
638            query,
639        )
640    } else {
641        Some(if lift.operands.len() == 1 {
642            lift.operands.into_iter().next().unwrap()
643        } else {
644            stmt::ExprAnd {
645                operands: lift.operands,
646            }
647            .into()
648        })
649    }
650}
651
652/// HasOne/HasMany branch: rewrite to a foreign-key IN subquery against
653/// the related table. Single-column FKs produce a scalar IN; composite
654/// FKs produce a tuple-form IN that the SQL serializer renders as
655/// `(a, b) IN (SELECT a, b FROM ...)`.
656fn lift_has_n_in_subquery(
657    target: ModelId,
658    pair: &BelongsTo,
659    query: &stmt::Query,
660) -> Option<stmt::Expr> {
661    lift_fk_in_subquery(
662        target,
663        super::key_field_refs(0, pair.foreign_key.fields.iter().map(|fk| fk.target)),
664        super::key_field_refs(0, pair.foreign_key.fields.iter().map(|fk| fk.source)),
665        query,
666    )
667}
668
669impl Visit for LiftBelongsTo<'_> {
670    fn visit_expr_in_subquery(&mut self, _i: &stmt::ExprInSubquery) {
671        // Stop the walk at a nested IN-subquery boundary. Field references
672        // inside resolve in the nested query's own scope, but the visitor's
673        // `cx` is scoped to the current (outer) subquery's source — recursing
674        // would misresolve inner refs as outer-model fields and produce
675        // bogus FK matches. The main LiftInSubquery walker handles the
676        // nested subquery on its own pass, with the correct scope.
677    }
678
679    fn visit_expr_binary_op(&mut self, i: &stmt::ExprBinaryOp) {
680        match (&*i.lhs, &*i.rhs) {
681            (stmt::Expr::Reference(expr_reference), other)
682            | (other, stmt::Expr::Reference(expr_reference)) => {
683                assert!(i.op.is_eq() || i.op.is_ne());
684
685                if i.op.is_eq() || i.op.is_ne() {
686                    let field = self
687                        .cx
688                        .resolve_expr_reference(expr_reference)
689                        .as_field_unwrap();
690
691                    self.lift_fk_constraint(field.id, i.op, other);
692                } else {
693                    self.fail = true;
694                }
695            }
696            // Constraints we can't lift to a direct FK comparison (e.g. a
697            // projection through an embedded field).  Bail to the IN-subquery
698            // form so the filter is preserved verbatim; without this, the
699            // empty `operands` list silently produced an empty AND (= true)
700            // and the subquery returned every row.
701            _ => {
702                self.fail = true;
703            }
704        }
705    }
706}
707
708impl LiftBelongsTo<'_> {
709    fn lift_fk_constraint(&mut self, field: FieldId, op: stmt::BinaryOp, expr: &stmt::Expr) {
710        for (i, fk_field) in self.belongs_to.foreign_key.fields.iter().enumerate() {
711            if fk_field.target == field {
712                if self.fk_field_matches[i] {
713                    todo!("not handled");
714                }
715
716                self.operands.push(stmt::Expr::binary_op(
717                    stmt::Expr::ref_self_field(fk_field.source),
718                    op,
719                    expr.clone(),
720                ));
721                self.fk_field_matches[i] = true;
722
723                return;
724            }
725        }
726
727        self.fail = true;
728    }
729}
