16/// Normalizes application-level statements in one recursive traversal.
17///
18/// Node-specific helpers may inspect or mutate the node they receive, but they
19/// must not recurse. The [`VisitMut`] implementation owns traversal so adding a
20/// normalization rule does not add another full AST walk.
21struct Normalize<'a> {
22    schema: &'a Schema,
23    capability: &'a Capability,
24    error: Option<Error>,
25
26    /// Variant guards the enclosing conjunctions state, innermost last.
27    ///
28    /// A predicate inside `is_variant(x, v) AND ..` need not repeat that
29    /// guard. The stack is scoped to the statement being walked: a subquery
30    /// starts empty.
31    guards: Vec<stmt::Expr>,
32}
33
34impl Engine {
35    pub(super) fn normalize_stmt(&self, stmt: &mut stmt::Statement) -> Result<()> {
36        let mut normalize = Normalize {
37            schema: &self.schema,
38            capability: self.capability,
39            error: None,
40            guards: vec![],
41        };
42        normalize.visit_stmt_mut(stmt);
43
44        match normalize.error {
45            Some(error) => Err(error),
46            None => Ok(()),
47        }
48    }
49}
50
51impl Normalize<'_> {
52    fn record(&mut self, error: Error) {
53        if self.error.is_none() {
54            self.error = Some(error);
55        }
56    }
57
58    /// Walks a nested statement with its own guard scope.
59    fn scoped(&mut self, walk: impl FnOnce(&mut Self)) {
60        let guards = std::mem::take(&mut self.guards);
61        walk(self);
62        self.guards = guards;
63    }
64}
65
66impl VisitMut for Normalize<'_> {
67    fn visit_expr_mut(&mut self, expr: &mut stmt::Expr) {
68        if let stmt::Expr::And(and) = expr {
69            self.normalize_conjunction(and);
70        } else {
71            stmt::visit_mut::visit_expr_mut(self, expr);
72            self.normalize_predicate_guards(expr);
73        }
74    }
75
76    fn visit_stmt_mut(&mut self, stmt: &mut stmt::Statement) {
77        self.scoped(|this| stmt::visit_mut::visit_stmt_mut(this, stmt));
78    }
79
80    fn visit_stmt_insert_mut(&mut self, insert: &mut stmt::Insert) {
81        self.normalize_upsert_defaults(insert);
82        if self.error.is_none() {
83            stmt::visit_mut::visit_stmt_insert_mut(self, insert);
84        }
85    }
86
87    fn visit_stmt_query_mut(&mut self, query: &mut stmt::Query) {
88        self.normalize_cursor_order(query);
89        if self.error.is_none() {
90            self.scoped(|this| stmt::visit_mut::visit_stmt_query_mut(this, query));
91        }
92    }
93}
