9impl ExecPlanner<'_> {
10    /// Converts the logical plan's execution order into the exec program's
11    /// step sequence, returning the steps and whether the plan needs to be
12    /// wrapped in a transaction.
13    pub(super) fn plan_execution(mut self) -> (Vec<Step>, bool) {
14        self.plan_steps();
15
16        let needs_transaction = self.needs_transaction();
17
18        (self.steps, needs_transaction)
19    }
20
21    fn plan_steps(&mut self) {
22        // Group each maximal run of consecutive same-guard nodes into one
23        // `If` block, emitted at the run's existing position. An unguarded
24        // chain interleaved between two same-guard runs yields several `If`
25        // blocks with the same condition; the guard rules guarantee an
26        // interleaved unguarded node never consumes a guarded output, and the
27        // skip-path variable classification below is per block.
28        let logical_plan = self.logical_plan;
29        let mut if_block = vec![];
30
31        for &node_id in logical_plan.execution_order() {
32            self.plan_node(node_id, &mut if_block);
33        }
34
35        self.flush_if_block(&mut if_block);
36    }
37
38    fn plan_node(&mut self, node_id: mir::NodeId, if_block: &mut Vec<mir::NodeId>) {
39        let guard = self.logical_plan[node_id].guard;
40
41        if let Some(&first_node_id) = if_block.first()
42            && guard != self.logical_plan[first_node_id].guard
43        {
44            self.flush_if_block(if_block);
45        }
46
47        if guard.is_some() {
48            if_block.push(node_id);
49        } else {
50            self.steps.push(Step::Run(node_id));
51        }
52    }
53
54    fn flush_if_block(&mut self, block: &mut Vec<mir::NodeId>) {
55        if block.is_empty() {
56            return;
57        }
58
59        let step = self.emit_if_block(std::mem::take(block));
60        self.steps.push(step);
61    }
62
63    fn needs_transaction(&self) -> bool {
64        self.use_transactions
65            && self
66                .steps
67                .iter()
68                .map(|step| step.db_op_count(self.logical_plan))
69                .sum::<usize>()
70                > 1
71    }
72
73    /// Wraps a run of same-guard nodes in an `If`, deriving the skip
74    /// bookkeeping from a static classification of the variables the `then`
75    /// arm touches:
76    ///
77    /// - **External inputs** (produced outside, loaded inside): released on
78    ///   skip, one entry per load the `then` arm would have performed,
79    ///   keeping use counts exact on both paths.
80    /// - **Escaping outputs** (produced inside, consumed outside): assigned a
81    ///   non-loadable slot on skip with the node's external use count, so
82    ///   outside consumers can release the skipped output.
83    /// - **Internal variables** (produced and consumed inside): untouched —
84    ///   on the skip path their slots are never created.
85    fn emit_if_block(&self, block: Vec<mir::NodeId>) -> Step {
86        debug_assert!(
87            block
88                .iter()
89                .all(|id| !self.logical_plan[id].op.is_effectful()),
90            "effectful node inside an `If` arm"
91        );
92
93        let skipped_inputs = self.collect_skipped_inputs(&block);
94        let skipped_outputs = self.collect_skipped_outputs(&block);
95
96        Step::If(exec::If {
97            then: block,
98            skipped_inputs,
99            skipped_outputs,
100        })
101    }
102
103    /// Collects loads from variables produced outside the block. The skip arm
104    /// releases each load that the `then` arm would have performed.
105    fn collect_skipped_inputs(&self, block: &[mir::NodeId]) -> Vec<mir::NodeId> {
106        let in_block: IndexSet<mir::NodeId> = block.iter().copied().collect();
107
108        block
109            .iter()
110            .flat_map(|&node_id| self.logical_plan[node_id].op.input_loads())
111            .filter(|load| !in_block.contains(load))
112            .collect()
113    }
114
115    /// Collects variables produced inside the block and consumed outside it.
116    fn collect_skipped_outputs(&self, block: &[mir::NodeId]) -> Vec<(mir::NodeId, usize)> {
117        block
118            .iter()
119            .filter_map(|&node_id| {
120                let external_uses = self.external_use_count(block, node_id);
121                (external_uses > 0).then_some((node_id, external_uses))
122            })
123            .collect()
124    }
125
126    fn external_use_count(&self, block: &[mir::NodeId], node_id: mir::NodeId) -> usize {
127        let in_block_loads = block
128            .iter()
129            .flat_map(|&consumer_id| self.logical_plan[consumer_id].op.input_loads())
130            .filter(|&load| load == node_id)
131            .count();
132
133        self.logical_plan[node_id].num_uses - in_block_loads
134    }
